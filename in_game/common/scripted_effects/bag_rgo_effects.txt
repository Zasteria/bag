# Mark the mod active for cross-mod detection and register its CMM settings.
# Scope: country
bag_rgo_register = {
	cmf_register_mod = { mod_id = bag_rgo }

	cmm_register_bool_setting = {
		mod_id = bag_rgo
		setting_id = location_only
		tab_id = filter
		group_id = scope
		default_value = 0
	}

	bag_rgo_sync_settings = yes
}

# CMM keeps setting values in the `cmm` variable map, which a filter trigger
# cannot read. Mirror the one setting the filter cares about onto a plain
# country variable: set when on, absent when off.
# Scope: country
bag_rgo_sync_settings = {
	cmm_sync_bool_alias = {
		setting = bag_rgo__location_only
		alias = bag_rgo_location_only
	}
}
