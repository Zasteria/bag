# A filter is handed the object it tests and little else, so the two panels
# park what their filter is missing in global variables, which any scope can
# reach. Each pair below is one probe: a query that tells the probe whether the
# stored value is already current, and the write that updates it.
#
# The buildings panel of a location parks the location it is showing; the build
# panel parks the building type being placed.

# Scope: country
bag_rgo_view_location_is_current = {
	scope = country

	is_shown = {
		has_global_variable = bag_rgo_view_location
		global_var:bag_rgo_view_location = scope:bag_rgo_loc
	}
	effect = { }
}

# Scope: country
bag_rgo_store_view_location = {
	scope = country

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		set_global_variable = { name = bag_rgo_view_location value = scope:bag_rgo_loc }
	}
}
