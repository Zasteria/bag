# The filter runs on a building type and is handed the country, never the
# location being looked at. These two bridge that gap: a probe widget watches
# the buildings panel and parks the viewed location on the player country,
# where the filter trigger can reach it through scope:target.

# True while the stored location already matches the one passed in, which is
# what keeps the probe from firing on every frame.
# Scope: country
bag_rgo_view_location_is_current = {
	scope = country

	is_shown = {
		has_variable = bag_rgo_view_location
		var:bag_rgo_view_location = scope:bag_rgo_loc
	}
	effect = { }
}

# Park the location the buildings panel is showing.
# Scope: country
bag_rgo_store_view_location = {
	scope = country

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		set_variable = { name = bag_rgo_view_location value = scope:bag_rgo_loc }
	}
}
