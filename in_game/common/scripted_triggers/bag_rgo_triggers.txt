# Entry point for the list filter. Root is the building type being tested, and
# that is all the filter gets — a building_type scoped filter is not handed
# scope:target, so everything else has to come from global variables. The probe
# widget parks the location currently on screen in one of them.
#
# The location itself always counts. The rest of its province is added too
# unless the Mod Menu narrowed the search, matching the shovel badge the game
# draws in the location panel — the bonus comes from raw materials anywhere in
# the province.
#
# Scope: building_type
bag_rgo_has_local_bonus = {
	has_global_variable = bag_rgo_view_location
	global_var:bag_rgo_view_location = {
		OR = {
			bag_rgo_location_feeds_root = yes
			AND = {
				NOT = { has_global_variable = bag_rgo_location_only }
				province = {
					any_location_in_province = {
						bag_rgo_location_feeds_root = yes
					}
				}
			}
		}
	}
}

# Diagnostic backing the one remaining "debug" filter. Passes every building
# type once the probe has stored a location, so an empty list means the probe
# never ran. Remove it, and its entry in gui/filters, once the real filter is
# confirmed working.
# Scope: building_type
bag_rgo_debug_location_known = {
	has_global_variable = bag_rgo_view_location
}
