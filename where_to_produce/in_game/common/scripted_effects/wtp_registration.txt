# The whole interface lives in the Community Mod Framework's Mod Menu: four
# goods pickers, the list of recipes that make the chosen good, and the
# shortlist of provinces for the recipe that is ticked. Nothing of our own is
# drawn, so it looks like the rest of the menu and no vanilla file is touched.
#
# Both lists are registered at their full height and never rebuilt; the rows are
# written in place and the unused ones hidden. See wtp_effects.txt for why.
#
# Scope: country
wtp_register = {
	cmf_register_mod = { mod_id = wtp }

	# One picker per category, the way the game already groups goods. Option
	# counts follow the generated catalogue.
	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = pick_rgo
		tab_id = plan
		group_id = goods
		default_index = 1
		option_count = 21
	}
	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = pick_basic
		tab_id = plan
		group_id = goods
		default_index = 1
		option_count = 11
	}
	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = pick_consumer
		tab_id = plan
		group_id = goods
		default_index = 1
		option_count = 14
	}
	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = pick_weapons
		tab_id = plan
		group_id = goods
		default_index = 1
		option_count = 5
	}

	# The recipes that make the chosen good. One tick decides which of them the
	# shortlist is built for; the column beside it is the most that recipe could
	# ever gain from local raw materials. Twenty rows is the widest any good
	# needs — liquor — and the generator says so if a patch changes that.
	cmm_register_global_settings_list = {
		mod_id = wtp
		setting_id = recipes
		tab_id = plan
		item_count = 20
		is_ordered = 0
	}
	cmm_register_list_bool_field = {
		mod_id = wtp
		setting_id = recipes
		field_id = pick
		default_value = 0
	}
	cmm_register_list_data_field = {
		mod_id = wtp
		setting_id = recipes
		field_id = ceiling
		default_value = 0
	}
	cmm_set_list_field_format = { mod_id = wtp setting_id = recipes field_id = ceiling }

	# The answer: provinces, best first, with the bonus the recipe gets in each.
	cmm_register_global_settings_list = {
		mod_id = wtp
		setting_id = provinces
		tab_id = plan
		item_count = 10
		is_ordered = 0
	}
	cmm_register_list_data_field = {
		mod_id = wtp
		setting_id = provinces
		field_id = bonus
		default_value = 0
	}
	cmm_set_list_field_format = { mod_id = wtp setting_id = provinces field_id = bonus }

	cmm_register_global_numeric_setting = {
		mod_id = wtp
		setting_id = shortlist_size
		tab_id = plan
		group_id = display
		default_value = 3
		min_value = 1
		max_value = 10
		step_value = 1
	}

	wtp_sync_settings = yes
	set_global_variable = { name = wtp_shown_rows value = global_var:wtp_rows }
	wtp_apply_pickers = yes

	# Both rebuilds clear their list first, so this is also what leaves the menu
	# empty and tidy when nothing has been chosen yet.
	if = {
		limit = { has_global_variable = wtp_good }
		set_global_variable = { name = wtp_shown_good value = global_var:wtp_good }
	}
	wtp_fill_recipes = yes
	wtp_sync_recipe_ticks = yes
	wtp_rebuild_shortlist = yes
}

# CMM keeps its values in a variable map that neither a script value nor an
# ordered iterator can read, so everything the ranking needs is mirrored onto
# plain globals.
# Scope: country
wtp_sync_settings = {
	cmm_sync_setting_alias = { setting = wtp__pick_rgo alias = wtp_pick_rgo }
	cmm_sync_setting_alias = { setting = wtp__pick_basic alias = wtp_pick_basic }
	cmm_sync_setting_alias = { setting = wtp__pick_consumer alias = wtp_pick_consumer }
	cmm_sync_setting_alias = { setting = wtp__pick_weapons alias = wtp_pick_weapons }
	cmm_sync_setting_alias = { setting = wtp__shortlist_size alias = wtp_rows }
}

# The pickers and the row count arrive here; ticking a recipe does not, because
# CMM has no auto-apply for list settings and routes those through the
# `wtp__recipes_on_changed` scripted GUI instead.
#
# Every mod's callback leaf runs on every callback, ours included, so the two
# things worth rebuilding are compared against what is currently on screen
# rather than rebuilt on the off chance.
# Scope: country
wtp_handle_callback = {
	wtp_sync_settings = yes
	wtp_apply_pickers = yes

	if = {
		limit = {
			has_global_variable = wtp_good
			OR = {
				NOT = { has_global_variable = wtp_shown_good }
				NOT = { global_var:wtp_shown_good = global_var:wtp_good }
			}
		}
		set_global_variable = { name = wtp_shown_good value = global_var:wtp_good }
		wtp_fill_recipes = yes
		wtp_sync_recipe_ticks = yes
		wtp_rebuild_shortlist = yes
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_global_variable = wtp_shown_rows }
				NOT = { global_var:wtp_shown_rows = global_var:wtp_rows }
			}
		}
		set_global_variable = { name = wtp_shown_rows value = global_var:wtp_rows }
		wtp_rebuild_shortlist = yes
	}
}

# Whichever picker is off "nothing chosen" names the good. Scope: country
wtp_apply_pickers = {
	if = { limit = { global_var:wtp_pick_rgo > 1 } wtp_apply_pick_rgo = yes }
	if = { limit = { global_var:wtp_pick_basic > 1 } wtp_apply_pick_basic = yes }
	if = { limit = { global_var:wtp_pick_consumer > 1 } wtp_apply_pick_consumer = yes }
	if = { limit = { global_var:wtp_pick_weapons > 1 } wtp_apply_pick_weapons = yes }
}
