# Building the shortlist.
#
# The player picks a good, ticks one of the recipes that make it, and gets the
# provinces where that recipe would gain most from local raw materials. With the
# recipe fixed its output is the same everywhere, so the bonus is the only thing
# left to rank on — which is also why the ceiling belongs beside the recipe
# rather than in the province rows, where it used to describe a different recipe
# on every line.

# A different good: refill the recipe list and drop the old answer.
# Scope: country
wtp_on_good_changed = {
	remove_global_variable = wtp_recipe

	cmm_begin_global_settings_list = {
		mod_id = wtp
		setting_id = recipes
		tab_id = plan
		is_ordered = 0
	}
	wtp_fill_recipes = yes
	cmm_finish_global_settings_list = {
		mod_id = wtp
		setting_id = recipes
	}

	wtp_label_recipe_ceilings = yes
	wtp_clear_shortlist = yes
}

# Which recipe is ticked. CMM can hand back the rows whose bool field is set as
# a plain variable list, which is how Construction Manager reads its own lists --
# no walking rows and poking at the settings map.
# Scope: country
wtp_read_picked_recipe = {
	remove_global_variable = wtp_recipe
	clear_variable_list = wtp_picked_recipes

	cmm_build_list_bool_list = {
		setting = wtp__recipes
		field_slot = 1
		list_name = wtp_picked_recipes
	}

	# One tick decides the answer; if several are set the first one wins.
	every_in_list = {
		variable = wtp_picked_recipes
		limit = { NOT = { exists = global_var:wtp_recipe } }
		root = {
			set_global_variable = { name = wtp_recipe value = prev }
		}
	}
}

# Collect every owned province where the ticked recipe would gain something,
# then keep the best of them.
#
# Provinces rather than locations: the bonus is province wide, so ten locations
# of one province would all score the same and crowd everything else out.
#
# Scope: country
wtp_rebuild_shortlist = {
	clear_global_variable_list = wtp_pool

	if = {
		limit = { has_global_variable = wtp_recipe }

		every_owned_location = {
			province = {
				if = {
					limit = {
						wtp_selected_bonus > 0
						NOT = {
							is_target_in_global_variable_list = {
								name = wtp_pool
								target = this
							}
						}
					}
					add_to_global_variable_list = { name = wtp_pool target = this }
				}
			}
		}

		cmm_begin_global_settings_list = {
			mod_id = wtp
			setting_id = provinces
			tab_id = plan
			is_ordered = 0
		}
		ordered_in_global_list = {
			variable = wtp_pool
			order_by = { value = wtp_selected_bonus }
			max = wtp_row_limit
			check_range_bounds = no
			root = {
				cmm_add_global_settings_list_item = {
					mod_id = wtp
					setting_id = provinces
					value = prev
					name = wtp_province_row
				}
			}
		}
		cmm_finish_global_settings_list = {
			mod_id = wtp
			setting_id = provinces
		}

		wtp_label_province_bonuses = yes
	}
}

# Fill the percentage column beside each province. cmm_set_list_data_value wants
# a literal row number, so the rows are walked by ordinal and each reads the
# province stored against it.
# Scope: country
wtp_label_province_bonuses = {
	cmm_build_list_ordered_values = {
		setting = wtp__provinces
		list_name = wtp_province_rows
	}
	set_variable = { name = wtp_row_cursor value = 0 }
	every_in_list = {
		variable = wtp_province_rows
		save_scope_as = wtp_row_province
		root = {
			change_variable = { name = wtp_row_cursor add = 1 }
			cmm_set_list_data_value = {
				mod_id = wtp
				setting_id = provinces
				field_id = bonus
				item = var:wtp_row_cursor
				value = scope:wtp_row_province.wtp_selected_bonus
			}
		}
	}
}

# Fill the ceiling column beside each recipe, same walk by ordinal.
# Scope: country
wtp_label_recipe_ceilings = {
	cmm_build_list_ordered_values = {
		setting = wtp__recipes
		list_name = wtp_recipe_rows
	}
	set_variable = { name = wtp_row_cursor value = 0 }
	every_in_list = {
		variable = wtp_recipe_rows
		save_scope_as = wtp_row_recipe
		root = {
			change_variable = { name = wtp_row_cursor add = 1 }
			cmm_set_list_data_value = {
				mod_id = wtp
				setting_id = recipes
				field_id = ceiling
				item = var:wtp_row_cursor
				value = wtp_row_ceiling
			}
		}
	}
}

# Forget the current answer. Scope: country
wtp_clear_shortlist = {
	clear_global_variable_list = wtp_pool
	cmm_begin_global_settings_list = {
		mod_id = wtp
		setting_id = provinces
		tab_id = plan
		is_ordered = 0
	}
	cmm_finish_global_settings_list = {
		mod_id = wtp
		setting_id = provinces
	}
}
