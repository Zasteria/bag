# What to build in one province.
#
# The player walks down region -> area -> province, and the last list answers:
# every building this mod can say something about, ranked by what it would get
# out of that province's raw materials.
#
# Two things a building can be ranked on, and the Display setting picks which:
#
#   local    the raw material bonus itself, as a percentage
#   volume   what the building turns out at that bonus, output x (1 + %)
#
# Volume is what compares two buildings. Efficiency multiplies output, so a mill
# at 3% still turns out more than a village workshop at its full 10%; the
# percentage alone would rank them the other way round.
#
# ---------------------------------------------------------------------------
# How the lists behave
# ---------------------------------------------------------------------------
#
# CMM draws list rows from `cmm_list_items_<setting>` and gates each one on
# `CMMListItemIsVisible`, but the vbox holding them has no `ignoreinvisible`, so
# a hidden row still takes its place in the layout. Every list is therefore
# *resized*: the item list is thrown away, `cmm_list_initialized_<setting>`
# removed, and the setting re-registered at exactly the height it needs. That is
# the same door the first registration goes through.
#
# A list can never be registered empty — `item_count` is clamped to at least one
# — so `is_shown` on its `_on_changed` scripted GUI is what takes an empty one
# off the screen. See wtp_scripted_gui.txt.
#
# Ordinals are literals throughout, because `item = var:x` is pasted into the CMM
# macro verbatim and dies at load with "More than one colon in event target
# link". wtp_generated_rows.txt holds the switches that turn a counter into one.

# Scope: country
wtp_fill_regions = {
	clear_global_variable_list = wtp_region_pool
	set_variable = { name = wtp_regions_count value = 0 }

	every_owned_location = {
		region = {
			if = {
				limit = {
					NOT = {
						is_target_in_global_variable_list = {
							name = wtp_region_pool
							target = this
						}
					}
				}
				add_to_global_variable_list = { name = wtp_region_pool target = this }
				root = { change_variable = { name = wtp_regions_count add = 1 } }
			}
		}
	}

	if = {
		limit = { var:wtp_regions_count > 20 }
		set_variable = { name = wtp_regions_count value = 20 }
	}
	wtp_resize_regions_list = yes

	set_variable = { name = wtp_row_cursor value = 0 }
	every_in_global_list = {
		variable = wtp_region_pool
		save_scope_as = wtp_row_target
		root = {
			change_variable = { name = wtp_row_cursor add = 1 }
			if = {
				limit = { var:wtp_row_cursor <= var:wtp_regions_count }
				wtp_place_regions_row = yes
			}
		}
	}
	wtp_sync_regions_ticks = yes
}

# Before the click exactly one row was ticked, so a tick that is not what is
# already stored is the one just made. That is what makes the column behave as a
# radio button rather than a set of independent boxes.
#
# Scope: country
# Args: s (key) -- setting id; g (key) -- the global holding the selection
wtp_read_pick = {
	clear_variable_list = wtp_ticked_rows
	cmm_build_list_bool_list = {
		setting = wtp__$s$
		field_slot = 1
		list_name = wtp_ticked_rows
	}

	if = { limit = { has_global_variable = wtp_new_pick } remove_global_variable = wtp_new_pick }
	set_variable = { name = wtp_tick_count value = 0 }

	every_in_list = {
		variable = wtp_ticked_rows
		save_scope_as = wtp_tick
		root = {
			change_variable = { name = wtp_tick_count add = 1 }
			if = {
				limit = {
					NOT = { has_global_variable = wtp_new_pick }
					OR = {
						NOT = { has_global_variable = $g$ }
						NOT = { global_var:$g$ = scope:wtp_tick }
					}
				}
				set_global_variable = { name = wtp_new_pick value = scope:wtp_tick }
			}
		}
	}

	if = {
		limit = { has_global_variable = wtp_new_pick }
		set_global_variable = { name = $g$ value = global_var:wtp_new_pick }
		remove_global_variable = wtp_new_pick
	}
	else_if = {
		limit = {
			var:wtp_tick_count = 0
			has_global_variable = $g$
		}
		remove_global_variable = $g$
	}
}

# Scope: country
# Args: s (key) -- setting id; g (key) -- selection global; i (number) -- ordinal
wtp_set_tick = {
	if = {
		limit = {
			has_global_variable = $g$
			exists = scope:cmm_list_current_item_value
			global_var:$g$ = scope:cmm_list_current_item_value
		}
		cmm_set_list_data_value = { mod_id = wtp setting_id = $s$ field_id = pick item = $i$ value = 1 }
	}
	else = {
		cmm_set_list_data_value = { mod_id = wtp setting_id = $s$ field_id = pick item = $i$ value = 0 }
	}
}

# Score every candidate building against the chosen province, then take the best
# few. Only one province is scored, so this is a couple of hundred script value
# reads rather than one per province in the realm.
# Scope: country
wtp_fill_top = {
	clear_variable_map = wtp_bonus_of
	clear_variable_map = wtp_volume_of
	clear_variable_map = wtp_desc_of
	clear_global_variable_list = wtp_top_pool
	set_variable = { name = wtp_top_count value = 0 }

	if = {
		limit = { has_global_variable = wtp_sel_province }
		global_var:wtp_sel_province = { save_scope_as = wtp_target }
		wtp_collect_buildings = yes
	}

	if = {
		limit = { var:wtp_top_count > wtp_row_limit }
		set_variable = { name = wtp_top_count value = wtp_row_limit }
	}
	if = {
		limit = { var:wtp_top_count > 12 }
		set_variable = { name = wtp_top_count value = 12 }
	}
	wtp_resize_top_list = yes
	wtp_write_top_rows = yes
}

# One candidate. Scope: country, with scope:wtp_target as the province.
# Args: b (key) -- building type
wtp_offer_building = {
	if = {
		limit = {
			wtp_building_in_scope_$b$ = yes
			OR = {
				NOT = { wtp_filtering_buildings = yes }
				wtp_building_available_$b$ = yes
			}
		}
		set_variable = { name = wtp_cand_bonus value = 0 }
		set_variable = { name = wtp_cand_volume value = 0 }
		if = { limit = { has_variable = wtp_cand_desc } remove_variable = wtp_cand_desc }

		wtp_methods_of_$b$ = yes

		if = {
			limit = { var:wtp_cand_bonus > 0 }
			add_to_global_variable_list = { name = wtp_top_pool target = flag:$b$ }
			add_to_variable_map = { name = wtp_bonus_of key = flag:$b$ value = var:wtp_cand_bonus }
			add_to_variable_map = { name = wtp_volume_of key = flag:$b$ value = var:wtp_cand_volume }
			add_to_variable_map = { name = wtp_desc_of key = flag:$b$ value = var:wtp_cand_desc }
			change_variable = { name = wtp_top_count add = 1 }
		}
	}
}

# One method of the candidate. The best by whatever is being ranked on wins, and
# its recipe is what the row's tooltip will show.
# Scope: country, with scope:wtp_target as the province.
# Args: m (key) -- the generated identifier of the method
wtp_try_method = {
	if = {
		limit = {
			scope:wtp_target.wtp_bonus_$m$ > 0
			OR = {
				AND = {
					wtp_ranking_by_volume = yes
					scope:wtp_target.wtp_volume_$m$ > var:wtp_cand_volume
				}
				AND = {
					NOT = { wtp_ranking_by_volume = yes }
					scope:wtp_target.wtp_bonus_$m$ > var:wtp_cand_bonus
				}
			}
		}
		set_variable = { name = wtp_cand_bonus value = scope:wtp_target.wtp_bonus_$m$ }
		set_variable = { name = wtp_cand_volume value = scope:wtp_target.wtp_volume_$m$ }
		set_variable = { name = wtp_cand_desc value = flag:wtp_recipe_$m$_desc }
	}
}

# Selection sort: twelve passes over at most a hundred and ten candidates, which
# is cheaper than it sounds and avoids needing an ordered iterator over a list
# whose ranking lives in a variable map.
# Scope: country
wtp_write_top_rows = {
	clear_variable_list = wtp_top_used
	wtp_take_top_row = { n = 1 }
	wtp_take_top_row = { n = 2 }
	wtp_take_top_row = { n = 3 }
	wtp_take_top_row = { n = 4 }
	wtp_take_top_row = { n = 5 }
	wtp_take_top_row = { n = 6 }
	wtp_take_top_row = { n = 7 }
	wtp_take_top_row = { n = 8 }
	wtp_take_top_row = { n = 9 }
	wtp_take_top_row = { n = 10 }
	wtp_take_top_row = { n = 11 }
	wtp_take_top_row = { n = 12 }
}

# Scope: country
# Args: n (number)
wtp_take_top_row = {
	if = {
		limit = { var:wtp_top_count >= $n$ }
		wtp_find_best_candidate = yes
		if = {
			limit = { has_variable = wtp_best_flag }
			wtp_write_top_row = { n = $n$ }
			add_to_variable_list = { name = wtp_top_used target = var:wtp_best_flag }
		}
	}
}

# Scope: country
wtp_find_best_candidate = {
	set_variable = { name = wtp_best_value value = 0 }
	if = { limit = { has_variable = wtp_best_flag } remove_variable = wtp_best_flag }
	every_in_global_list = {
		variable = wtp_top_pool
		save_scope_as = wtp_cand
		root = {
			if = {
				limit = {
					NOT = { is_target_in_variable_list = { name = wtp_top_used target = scope:wtp_cand } }
					wtp_candidate_rank > var:wtp_best_value
				}
				set_variable = { name = wtp_best_value value = wtp_candidate_rank }
				set_variable = { name = wtp_best_flag value = scope:wtp_cand }
			}
		}
	}
}

# The row label is the building's own key, which is also its localization key,
# so the flag standing for the candidate labels the row directly.
# Scope: country
# Args: n (number)
wtp_write_top_row = {
	set_local_variable = { name = wtp_key value = var:wtp_best_flag }
	set_variable = { name = wtp_row_local value = "variable_map(wtp_bonus_of|local_var:wtp_key)" }
	set_variable = { name = wtp_row_output value = "variable_map(wtp_volume_of|local_var:wtp_key)" }
	round_variable = { name = wtp_row_local nearest = 0.01 }
	round_variable = { name = wtp_row_output nearest = 0.01 }

	set_variable = { name = wtp__top_i$n$_name value = var:wtp_best_flag }
	set_variable = { name = wtp__top_i$n$_desc value = "variable_map(wtp_desc_of|local_var:wtp_key)" }
	cmm_set_list_item_value = { mod_id = wtp setting_id = top item = $n$ value = var:wtp_best_flag }
	cmm_set_list_data_value = { mod_id = wtp setting_id = top field_id = local item = $n$ value = var:wtp_row_local }
	cmm_set_list_data_value = { mod_id = wtp setting_id = top field_id = output item = $n$ value = var:wtp_row_output }

	if = { limit = { always = no } cmf_suppress = { v = wtp_key } }
}

# Picking a region drops the area and the province under it, and so on down.
# Scope: country
wtp_after_region_pick = {
	if = { limit = { has_global_variable = wtp_sel_area } remove_global_variable = wtp_sel_area }
	if = { limit = { has_global_variable = wtp_sel_province } remove_global_variable = wtp_sel_province }
	wtp_fill_areas = yes
	wtp_fill_provinces = yes
	wtp_fill_top = yes
}

	wtp_fill_provinces = yes
	wtp_fill_top = yes
}
