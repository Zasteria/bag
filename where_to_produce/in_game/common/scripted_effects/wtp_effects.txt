# Building the shortlist.
#
# The player picks a good, ticks the recipes worth considering, and gets the
# provinces where those recipes gain most from local raw materials. Each
# province is scored against every ticked recipe and keeps the best one, so the
# numbers on a row and the recipe named on it are always the same recipe.
#
# Two things a row can be ranked on, and the Display setting picks which:
#
#   local    the raw material bonus itself, as a percentage
#   volume   what the recipe actually turns out at that bonus, output x (1 + %)
#
# Volume is the one that compares recipes against each other. Efficiency
# multiplies output, so a jeweller's guild at 2% still beats a village carver at
# its full 10%; the percentage alone would rank them the other way round.
#
# Both lists are registered once, at their full height, and rows are written in
# place. CMM's other way of doing it -- `cmm_begin_settings_list`, a run of
# `cmm_add_settings_list_item`, `cmm_finish_settings_list` -- builds a list
# exactly once: every one of those effects is gated on
# `cmm_list_initialized_<setting>`, which registration itself sets. A second
# pass adds nothing and reports nothing.
#
# Writing in place means addressing a row by its ordinal, and an ordinal has to
# be a literal: `item = var:wtp_row_cursor` is pasted into the macro verbatim
# and lands in the log as
#
#     More than one colon in event target link 'flag:wtp__provinces_ivar:wtp_row_cursor_f1'
#
# For the recipes the generator already knows which row each one lands on, so it
# passes both as literals. For the provinces the order is only known at runtime,
# so a counter goes through a switch — which is how CMF turns one into a literal
# everywhere in its own source.

# Scope: country
# Args: n (number) -- row ordinal; m (key) -- the recipe
wtp_write_recipe_row = {
	set_variable = { name = wtp__recipes_i$n$_name value = flag:wtp_recipe_$m$ }
	set_variable = { name = wtp__recipes_i$n$_desc value = flag:wtp_recipe_$m$_desc }
	cmm_set_list_item_value = {
		mod_id = wtp
		setting_id = recipes
		item = $n$
		value = flag:wtp_recipe_$m$
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = pick
		item = $n$
		value = 1
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = ceiling
		item = $n$
		value = wtp_ceiling_$m$
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = peak
		item = $n$
		value = wtp_peak_$m$
	}
	cmm_show_list_item = { mod_id = wtp setting_id = recipes item = $n$ }
}

# Scope: country
wtp_reset_recipe_rows = {
	cmm_for_each_list_item = {
		setting = wtp__recipes
		effect = wtp_reset_recipe_row
	}
}

# Called per row by cmm_for_each_list_item, which substitutes the ordinal as a
# literal. Scope: country
# Args: setting (key), i (number)
wtp_reset_recipe_row = {
	cmm_hide_list_item = { mod_id = wtp setting_id = recipes item = $i$ }
	cmm_set_list_item_value = {
		mod_id = wtp
		setting_id = recipes
		item = $i$
		value = flag:wtp_empty_row
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = pick
		item = $i$
		value = 0
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = ceiling
		item = $i$
		value = 0
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = peak
		item = $i$
		value = 0
	}
	set_variable = { name = wtp__recipes_i$i$_name value = flag:wtp_empty_row }
	set_variable = { name = wtp__recipes_i$i$_desc value = flag:wtp_empty_row }

	# Accept the setting param cmm_for_each_list_item passes.
	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}

# The recipes currently ticked, as a global list the province scoring can walk
# from any scope.
#
# The rows a shorter good leaves over are hidden, not removed, and a right click
# on the column header ticks every row in the list — hidden ones included. Those
# all carry the placeholder value, which is why it is skipped here.
# Scope: country
wtp_read_active_recipes = {
	clear_variable_list = wtp_ticked_rows
	cmm_build_list_bool_list = {
		setting = wtp__recipes
		field_slot = 1
		list_name = wtp_ticked_rows
	}

	clear_global_variable_list = wtp_active_recipes
	every_in_list = {
		variable = wtp_ticked_rows
		save_scope_as = wtp_tick
		root = {
			if = {
				limit = { NOT = { scope:wtp_tick = flag:wtp_empty_row } }
				add_to_global_variable_list = { name = wtp_active_recipes target = scope:wtp_tick }
			}
		}
	}
}

# ---------------------------------------------------------------------------
# The shortlist
# ---------------------------------------------------------------------------

# Collect every owned province where some ticked recipe would gain, then keep
# the best of them.
#
# Provinces rather than locations: the bonus is province wide, so ten locations
# of one province would all score the same and crowd everything else out.
#
# Scope: country
wtp_rebuild_shortlist = {
	wtp_reset_province_rows = yes
	wtp_read_active_recipes = yes
	clear_global_variable_list = wtp_pool

	every_owned_location = {
		province = {
			if = {
				limit = {
					NOT = {
						is_target_in_global_variable_list = {
							name = wtp_pool
							target = this
						}
					}
				}
				wtp_score_province = yes
				if = {
					limit = { var:wtp_best_score > 0 }
					add_to_global_variable_list = { name = wtp_pool target = this }
				}
			}
		}
	}

	set_variable = { name = wtp_row_cursor value = 0 }
	ordered_in_global_list = {
		variable = wtp_pool
		order_by = { value = var:wtp_best_score }
		max = wtp_row_limit
		check_range_bounds = no
		save_scope_as = wtp_row_province
		root = {
			change_variable = { name = wtp_row_cursor add = 1 }
			wtp_place_province_row = yes
		}
	}
}

# The best ticked recipe for this province, by whichever measure is being ranked
# on, and what that one recipe scores. Recipes that gain nothing here are passed
# over: their volume is the plain output, which every other province matches, so
# they say nothing about this place.
# Scope: province
wtp_score_province = {
	save_scope_as = wtp_scored_province
	set_variable = { name = wtp_best_score value = 0 }
	set_variable = { name = wtp_best_bonus value = 0 }
	set_variable = { name = wtp_best_volume value = 0 }
	if = {
		limit = { has_variable = wtp_best_method }
		remove_variable = wtp_best_method
	}

	every_in_global_list = {
		variable = wtp_active_recipes
		save_scope_as = wtp_scored_recipe
		scope:wtp_scored_province = {
			if = {
				limit = {
					wtp_scored_bonus > 0
					wtp_scored_rank > var:wtp_best_score
				}
				set_variable = { name = wtp_best_score value = wtp_scored_rank }
				set_variable = { name = wtp_best_bonus value = wtp_scored_bonus }
				set_variable = { name = wtp_best_volume value = wtp_scored_volume }
				set_variable = { name = wtp_best_method value = scope:wtp_scored_recipe }
			}
		}
	}
}

# Scope: country
wtp_place_province_row = {
	switch = {
		trigger = var:wtp_row_cursor
		1 = { wtp_write_province_row = { n = 1 } }
		2 = { wtp_write_province_row = { n = 2 } }
		3 = { wtp_write_province_row = { n = 3 } }
		4 = { wtp_write_province_row = { n = 4 } }
		5 = { wtp_write_province_row = { n = 5 } }
		6 = { wtp_write_province_row = { n = 6 } }
		7 = { wtp_write_province_row = { n = 7 } }
		8 = { wtp_write_province_row = { n = 8 } }
		9 = { wtp_write_province_row = { n = 9 } }
		10 = { wtp_write_province_row = { n = 10 } }
	}
}

# One province row. Neither a province nor a stored recipe has a localization key
# the row label could be pointed at, so both are parked in globals and the label —
# a fixed key per ordinal — reads them back through data functions.
# Scope: country
# Args: n (number) -- row ordinal
wtp_write_province_row = {
	set_global_variable = { name = wtp_prov_row_$n$ value = scope:wtp_row_province }
	scope:wtp_row_province = {
		set_global_variable = { name = wtp_meth_row_$n$ value = var:wtp_best_method }
	}
	cmm_set_list_item_value = {
		mod_id = wtp
		setting_id = provinces
		item = $n$
		value = scope:wtp_row_province
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = provinces
		field_id = bonus
		item = $n$
		value = scope:wtp_row_province.wtp_stored_bonus
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = provinces
		field_id = volume
		item = $n$
		value = scope:wtp_row_province.wtp_stored_volume
	}
	cmm_show_list_item = { mod_id = wtp setting_id = provinces item = $n$ }
}

# Scope: country
wtp_reset_province_rows = {
	cmm_for_each_list_item = {
		setting = wtp__provinces

	}
}

wtp_reset_province_row = {
	cmm_hide_list_item = { mod_id = wtp setting_id = provinces item = $i$ }
	cmm_set_list_item_value = {
		mod_id = wtp
		setting_id = provinces
		item = $i$
		value = flag:wtp_empty_row
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = provinces
		field_id = bonus
		item = $i$
		value = 0
	}
	cmm_set_list_data_value = {
		mod_id = wtp

		item = $i$
		value = 0
	}
	if = {
		limit = { has_global_variable = wtp_prov_row_$i$ }
		remove_global_variable = wtp_prov_row_$i$

		limit = { has_global_variable = wtp_meth_row_$i$ }

	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}
