# Building the two lists.
#
# The player picks a good and gets every recipe that makes it, then the
# provinces where the ticked ones gain most. Each province is scored against
# every ticked recipe and keeps the best, so the recipe named on a row and the
# figures beside it are always the same recipe.
#
# Two things a row can be ranked on, and the Display setting picks which:
#
#   local    the raw material bonus itself, as a percentage
#   volume   what the recipe turns out at that bonus, output x (1 + %)
#
# Volume is what compares recipes against each other. Efficiency multiplies
# output, so a jeweller's guild at 2% still beats a village carver at its full
# 10%; the percentage alone would rank them the other way round.
#
# ---------------------------------------------------------------------------
# Why the lists are rebuilt rather than hidden
# ---------------------------------------------------------------------------
#
# CMM draws list rows from `cmm_list_items_<setting>` and gates each one on
# `CMMListItemIsVisible`, but the vbox holding them has no `ignoreinvisible`, so
# a hidden row still takes its place in the layout. Hiding the surplus therefore
# leaves the panel at its old height with the bottom half blank.
#
# So the item list is thrown away and the setting re-registered at exactly the
# height it needs. `cmm_register_*_settings_list` rebuilds the identities when
# `cmm_list_initialized_<setting>` is absent — it is the same door the first
# registration goes through — and `_cmm_reconcile_list_setting_item_growth`,
# which handles a re-registration, only ever adds rows, never removes them.
#
# What CMM cannot do is show a list with no rows at all: `item_count` is clamped
# to at least one. A list setting is only drawn while its `_on_changed` scripted
# GUI reports `is_shown`, so that is what hides an empty one — see
# wtp_scripted_gui.txt.
#
# `item_count = $n$` is fine because $n$ arrives as a literal from a switch. A
# variable is not: `item = var:wtp_row_cursor` is pasted into the macro verbatim
# and lands in the log at load as
#
#     More than one colon in event target link 'flag:wtp__provinces_ivar:wtp_row_cursor_f1'
#
# which is why every ordinal here comes through a switch, the way CMF turns a
# counter into a literal everywhere in its own source.

# One recipe row. The flag standing for the recipe is also its localization key,
# so the same value labels the row and identifies it to the scoring; the tooltip
# key cannot be built from it in script, so the index carries it.
#
# Rows arrive ticked. The shortlist is then an answer straight away, and
# unticking is how the player narrows it; wtp_fill_recipes only runs when the
# good changes, so a save reloads with the ticks it was left with.
# Scope: country
# Args: n (number) -- row ordinal
wtp_write_recipe_row = {
	set_variable = { name = wtp__recipes_i$n$_name value = scope:wtp_pending_recipe }
	set_variable = {
		name = wtp__recipes_i$n$_desc
		value = "global_variable_map(wtp_recipe_tooltip|scope:wtp_pending_recipe)"
	}
	cmm_set_list_item_value = {
		mod_id = wtp
		setting_id = recipes
		item = $n$
		value = scope:wtp_pending_recipe
	}
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = wtp_unticked_recipes
				target = scope:wtp_pending_recipe
			}
		}
		cmm_set_list_data_value = {
			mod_id = wtp
			setting_id = recipes
			field_id = pick
			item = $n$
			value = 0
		}
	}
	else = {
		cmm_set_list_data_value = {
			mod_id = wtp
			setting_id = recipes
			field_id = pick
			item = $n$
			value = 1
		}
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = ceiling
		item = $n$
		value = wtp_pending_ceiling
	}
	cmm_set_list_data_value = {
		mod_id = wtp
		setting_id = recipes
		field_id = peak
		item = $n$
		value = wtp_pending_peak
	}
}

# The recipes currently ticked, as a global list the province scoring can walk
# from any scope. A right click on the column header ticks every row at once,
# which is why nothing is assumed about how many come back.
# Scope: country
wtp_read_active_recipes = {
	clear_variable_list = wtp_ticked_rows
	cmm_build_list_bool_list = {
		setting = wtp__recipes
		field_slot = 1
		list_name = wtp_ticked_rows
	}

	clear_global_variable_list = wtp_active_recipes
	every_in_list = {
		variable = wtp_ticked_rows
		save_scope_as = wtp_tick
		root = {
			add_to_global_variable_list = { name = wtp_active_recipes target = scope:wtp_tick }
		}
	}
}

# What the player unticked, kept between refills.
#
# The rows are thrown away and rewritten whenever the list is resized — on every
# load, and whenever the good changes — so the ticks cannot live in the rows.
# They live here instead, as the recipes deliberately turned off; anything not
# in this list arrives ticked, which is what makes a newly chosen good an answer
# straight away.
# Scope: country
wtp_remember_ticks = {
	clear_global_variable_list = wtp_unticked_recipes
	every_in_global_list = {
		variable = wtp_shown_recipes
		save_scope_as = wtp_tick
		root = {
			if = {
				limit = {
					NOT = {
						is_target_in_global_variable_list = {
							name = wtp_active_recipes
							target = scope:wtp_tick
						}
					}
				}
				add_to_global_variable_list = { name = wtp_unticked_recipes target = scope:wtp_tick }
			}
		}
	}
}

# Collect every owned province where some ticked recipe would gain, then keep
# the best of them.
#
# Provinces rather than locations: the bonus is province wide, so ten locations
# of one province would all score the same and crowd everything else out.
#
# Scope: country
wtp_rebuild_shortlist = {
	wtp_read_active_recipes = yes
	clear_global_variable_list = wtp_pool
	set_variable = { name = wtp_shortlist_count value = 0 }

	every_owned_location = {
		province = {
			if = {
				limit = {
					NOT = {
						is_target_in_global_variable_list = {
							name = wtp_pool
							target = this
						}
					}
				}
				wtp_score_province = yes
				if = {
					limit = { var:wtp_best_score > 0 }
					add_to_global_variable_list = { name = wtp_pool target = this }
					root = { change_variable = { name = wtp_shortlist_count add = 1 } }
				}
			}
		}
	}

	if = {
		limit = { var:wtp_shortlist_count > wtp_row_limit }
		set_variable = { name = wtp_shortlist_count value = wtp_row_limit }
	}

	wtp_resize_province_list = yes

	set_variable = { name = wtp_row_cursor value = 0 }
	ordered_in_global_list = {
		variable = wtp_pool
		order_by = { value = var:wtp_best_score }
		max = wtp_row_limit
		check_range_bounds = no
		save_scope_as = wtp_row_province
		root = {
			change_variable = { name = wtp_row_cursor add = 1 }
			wtp_place_province_row = yes
		}
	}
}

# The best ticked recipe for this province, by whichever measure is being ranked
# on, and what that one recipe scores. Recipes that gain nothing here are passed
# over: their volume is the plain output, which every other province matches, so
# they say nothing about this place.
# Scope: province
wtp_score_province = {
	save_scope_as = wtp_scored_province
	set_variable = { name = wtp_best_score value = 0 }
	set_variable = { name = wtp_best_bonus value = 0 }
	set_variable = { name = wtp_best_volume value = 0 }
	if = {
		limit = { has_variable = wtp_best_method }
		remove_variable = wtp_best_method
	}

	every_in_global_list = {
		variable = wtp_active_recipes
		save_scope_as = wtp_scored_recipe
		scope:wtp_scored_province = {
			if = {
				limit = {
					wtp_scored_bonus > 0
					wtp_scored_rank > var:wtp_best_score
				}
				set_variable = { name = wtp_best_score value = wtp_scored_rank }
				set_variable = { name = wtp_best_bonus value = wtp_scored_bonus }
				set_variable = { name = wtp_best_volume value = wtp_scored_volume }
				set_variable = { name = wtp_best_method value = scope:wtp_scored_recipe }
			}
		}
	}

	# A data field prints what it is given, and 1.8766% reads as noise.
	round_variable = { name = wtp_best_bonus nearest = 0.01 }
	round_variable = { name = wtp_best_volume nearest = 0.01 }
}

# Four dropdowns, one good. Whichever one the player just moved decides, and the
# other three are put back to "nothing chosen" so the menu shows one answer
# rather than four half-remembered ones. Setting the only live picker back to
# "nothing chosen" clears the good, which is what empties both lists.
#
# CMM has no public setter for a plain setting, so the other three are written
# straight into the `cmm` map — the same shape `cmm_auto_apply_dropdown` writes,
# and registration only seeds a value when the key is absent, so it survives a
# reload.
# Scope: country
wtp_apply_pickers = {
	if = {
		limit = { wtp_picker_moved_rgo = yes }
		wtp_choose_rgo = yes
	}
	else_if = {
		limit = { wtp_picker_moved_basic = yes }
		wtp_choose_basic = yes
	}
	else_if = {
		limit = { wtp_picker_moved_consumer = yes }
		wtp_choose_consumer = yes
	}
	else_if = {
		limit = { wtp_picker_moved_weapons = yes }
		wtp_choose_weapons = yes
	}

	wtp_remember_pickers = yes
}

# Scope: country
wtp_choose_rgo = {
	wtp_clear_picker = { p = basic }
	wtp_clear_picker = { p = consumer }
	wtp_clear_picker = { p = weapons }
	if = { limit = { global_var:wtp_pick_rgo > 1 } wtp_apply_pick_rgo = yes }
	else = { wtp_forget_good = yes }
}
