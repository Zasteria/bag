# Whether the answer is ranked on the volume a building turns out rather than on
# the raw material percentage itself. Scope: any
wtp_ranking_by_volume = {
	has_global_variable = wtp_sort
	global_var:wtp_sort = 2
}

# The Display setting that hides buildings this country has no use for.
# Scope: any
wtp_filtering_buildings = {
	OR = {
		NOT = { has_global_variable = wtp_filter }
		global_var:wtp_filter = 1
	}
}

# The Display setting that hides buildings this country has not reached yet.
#
# Kept apart from the one above because the two answer different questions —
# "never" against "not yet" — and because this one leans on the engine's
# `can_build_building`, which is the one thing here nobody has watched work. If
# it turns out stricter than it looks and empties the list, this is the toggle to
# turn off, and the rest of the filtering carries on.
# Scope: any
wtp_only_unlocked_buildings = {
	OR = {
		NOT = { has_global_variable = wtp_unlocked }
		global_var:wtp_unlocked = 1
	}
}
