# True when the recipe list on screen was built for the good now chosen.
#
# Both halves have to be checked: no good chosen and no list built is a match,
# and either one without the other is not. A save reloads with its rows and its
# ticks already in the variables that hold them, so this is what keeps a reload
# from re-ticking everything the player had unticked.
# Scope: any
wtp_recipes_match_good = {
	OR = {
		AND = {
			NOT = { has_global_variable = wtp_good }
			NOT = { has_global_variable = wtp_shown_good }
		}
		AND = {
			has_global_variable = wtp_good
			has_global_variable = wtp_shown_good
			global_var:wtp_shown_good = global_var:wtp_good
		}
	}
}

# The Mod Menu setting that hides recipes this country has no use for.
# Scope: any
wtp_filtering_recipes = {
	OR = {
		NOT = { has_global_variable = wtp_filter }
		global_var:wtp_filter = 1
	}
}

# Whether a picker has moved since the last pass. On the first pass nothing has,
# because wtp_register remembers all four before anything reads them.
# Scope: any
wtp_picker_moved_rgo = {
	NOT = { global_var:wtp_seen_rgo = global_var:wtp_pick_rgo }
}

wtp_picker_moved_basic = {
	NOT = { global_var:wtp_seen_basic = global_var:wtp_pick_basic }
}

wtp_picker_moved_consumer = {
	NOT = { global_var:wtp_seen_consumer = global_var:wtp_pick_consumer }
}

wtp_picker_moved_weapons = {
	NOT = { global_var:wtp_seen_weapons = global_var:wtp_pick_weapons }
}
