# CMF extension point. Fires on new games, on save load and after a country
# transfer, once per human country.
# Scope: country
cmf_on_mod_registration = {
	on_actions = {
		wtp_on_registration
	}
}

# Fires when an action bar button is clicked or a Mod Menu setting changes.
# var:cmf_callback holds the key of whatever was acted on.
# Scope: country
cmf_on_callback = {
	on_actions = {
		wtp_on_callback
	}
}

# Scope: country
wtp_on_registration = {
	effect = {
		wtp_register = yes
	}
}

# Scope: country
wtp_on_callback = {
	effect = {
		wtp_handle_callback = yes
	}
}
