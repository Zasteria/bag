# Grants bureaucracy authority to IRQ once the sultanate has been formed and
# the player has the bureaucracies game rule enabled. Gated on tag rather than
# has_or_had_tag, so the slot grant does not stack when the player goes on to
# form Khilafah: the successor's own grant replaces this one.

nd_irq_bureaucracy_authority = {
	potential_trigger = {
		tag = IRQ
		has_variable = nd_irq_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
