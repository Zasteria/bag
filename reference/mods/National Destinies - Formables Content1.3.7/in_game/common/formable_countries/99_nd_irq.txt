# Iraq. The Mesopotamian state that consolidates instead of being absorbed.
# Identity split against the neighbouring mod formables in this space:
#   BNH / NIN are the Syriac Christian chain in the upper Jazira (Mosul, Nusaybin, Urfa).
#   KHL is the universal Sunni caliphate. IRQ at level 3 chains upward into it.
# IRQ is Muslim, Arabic speaking where it can be, and anchored on lower Mesopotamia:
# Baghdad, the Sawad, Kufa and the head of the Gulf at Basra.
IRQ_f = {

	level = 3
	required_locations_fraction = 0.5
	rule = plausible

	# Iraq is founded from Iraq, never annexed as a distant province.
	# Same idiom vanilla uses for USA_f and CAN_f.
	capital_required = yes

	potential = {
		religion.group = religion_group:muslim
		OR = {
			# The Arab claim: Iraqi, Bedouin, Hijazi, Levantine and the rest of arabic_group.
			culture = { has_culture_group = culture_group:arabic_group }
			# The Baghdad claim: whoever actually rules Mesopotamia from inside it.
			# This is the branch that keeps the Jalayirids (mongolian_culture, capital
			# at Baghdad in 1337) eligible, and it is also open to the Qara Qoyunlu and
			# Aq Qoyunlu once they move their seat to the Tigris, as Jahan Shah did.
			capital = { area = area:iraq_arabi_area }
		}
	}

	allow = {
		owns = location:baghdad
		owns = location:basra
		owns = location:kufa
		in_civil_war = no
	}

	name = IRQ
	flag = IRQ
	adjective = IRQ_ADJ
	tag = IRQ
	color = map_IRQ

	areas = {
		iraq_arabi_area
		jazira_area
	}

	form_effect = {
		set_capital = location:baghdad
		# The state adopts the identity of the land it rules. The proposal's
		# "Iraqi culture" is an outcome of formation, not a gate on it, because
		# no country on the 1337 map starts with iraqi_culture as its primary.
		# Pops are deliberately left alone: Mesopotamia stayed plural.
		if = {
			limit = { NOT = { culture = culture:iraqi_culture } }
			change_culture = culture:iraqi_culture
		}
		# Sultanate of Iraq, the Jalayirid rank restored.
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		add_stability = stability_mild_bonus
		# The formed markers are stamped here rather than only in the
		# formation event's options, so the content layer never depends on a
		# popup having been delivered. Hidden because the form_effect tooltip
		# should read as history, not as bookkeeping. The options set these
		# again, which is harmless: set_variable is idempotent. The path
		# variable stays in the event, because it is the player's answer.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_irq_formed
		}
		trigger_event_non_silently = nd_irq.1
	}
}
