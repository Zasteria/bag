# Pagarruyung: promoted from an event-payload record to a real formable.
#
# Vanilla PGR_f carries `potential = { always = no  #Formed by event }` and no
# territory clause at all: it exists only as the data payload for
# `form_country = formable_country:PGR_f` in flavor_maj.27, where Majapahit's
# Adityawarman expedition rebrands Siguntur. That call requires
# `c:SGT = { is_ai = yes }` (flavor_maj.txt:1787), so the one country the tag
# was written for can never be the player. PGR was the only formable in the
# game a human could never reach, and our whole content suite for it (37
# advances, four bureaucracies, five units, two levy tiers) was unreachable.
# Reported on the Workshop 2026-08-08.
#
# The vanilla route is untouched: scripted `form_country` runs form_effect
# regardless of potential, which is how vanilla forms JAP from the Sengoku
# situation while JAP_f is also `always = no`.
#
# Geography is the Minangkabau world as the sources describe it: the darek
# (the highland heartland of Tanah Datar) plus the four rantau river
# provinces its people settled. 19 of the map's 22 minangkabau_culture
# locations sit in these five provinces. Siguntur starts holding 8 of the 36,
# so this is a real expansion goal rather than a day-one button.
#
# fraction 0.6 sits inside the vanilla level-3 band for a formable of this
# shape (BAN_f 4 provinces 0.5, IRO_f 5 provinces 0.8, PON_f 6 provinces
# 0.75) and stays reachable whether or not the engine counts the five
# impassable and lake locations in these provinces toward the total.

REPLACE:PGR_f = {
	level = 3
	required_locations_fraction = 0.6
	rule = historical

	potential = {
		culture = culture:minangkabau_culture
	}
	allow = {
		# Pagaruyung in Tanah Datar is the seat itself. Adityawarman's move
		# from the Batang Hari lowlands up into the highland valleys is what
		# made the kingdom, so holding the highland seat is the one
		# non-negotiable condition.
		owns = location:pagaruyung
	}

	name = PGR
	flag = PGR
	adjective = PGR_ADJ
	tag = PGR
	color = map_PGR

	provinces = {
		pagaruyung_province
		upper_jambi_province
		riau_kampar_province
		riau_indragiri_province
		riau_rokan_province
	}

	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms PGR from
		# flavor_maj.27), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pgr_formed
		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_pgr.1
		add_stability = stability_mild_bonus
	}
}
