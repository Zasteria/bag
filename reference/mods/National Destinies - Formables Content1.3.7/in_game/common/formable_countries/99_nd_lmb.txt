# Kingdom of Lombardy (Regno di Lombardia)
# Plausible: the medieval Kingdom of Italy centred on Pavia, whose kings were crowned
# with the Iron Crown at Monza and Pavia. Gian Galeazzo Visconti had united the Po
# valley and stood one campaign from a royal title when plague took him in 1402; this
# formable lets a Lombard state finish that coronation.
#
# Footprint is the regnum, not the modern region. The six areas below are exactly
# the ones vanilla's own Milanese cabinet action uses to define the kingdom
# (in_game/common/cabinet_actions/mlo_restore_kingdom_lombardy.txt:16-25): the Po
# valley plus Liguria and Tuscany, which is the ground the Iron Crown actually
# covered. It shipped as `lombardy_area` alone, which asked for modern Lombardy and
# read as a Duchy of Milan rename; a Workshop report (2026-08-18) called that out.
#
# The fraction drops to 0.6 to keep the ask proportionate: 97 of 161 locations,
# against the 244 that vanilla ITA_f demands. Level stays at 2, below ITA_f's 3, so
# taking the Iron Crown still never locks a player out of forming Italy.
LMB_f = {

	level = 2
	required_locations_fraction = 0.6
	rule = plausible

	potential = {
		culture = culture:lombard
		religion = religion:catholic
	}

	allow = {
		owns = location:milano
		owns = location:pavia
	}

	name = LMB
	flag = LMB
	adjective = LMB_ADJ
	tag = LMB
	color = map_LMB

	areas = {
		lombardy_area
		piemonte_area
		emilia_area
		veneto_area
		liguria_area
		tuscany_area
	}

	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_lmb_formed
		}
		trigger_event_non_silently = nd_lmb.1
		add_stability = stability_mild_bonus
		if = {
			limit = {
				country_rank_level < 3
				nd_can_take_kingdom_rank = yes
			}
			set_country_rank_effect = {
				rank = country_rank:rank_kingdom
			}
		}
	}
}
