# Coptic Egypt.
#
# Not an imperial restoration. AEG (Aegyptus, level 4) is the Pharaonic and
# Mamluk claim over Egypt, Nubia, the Levant and the Hedjaz. CPT is the
# narrower thing that could plausibly have happened in the 1340s: the Coptic
# fiscal and scribal class that ran the Cairo revenue offices taking the state
# during the collapse that followed al-Nasir Muhammad's death in 1341, with the
# monasteries of Wadi el-Natrun behind it and Christian Nubia upriver.
#
# The window is narrow on purpose. Coptic decline was steep in exactly this
# period: the riot and church demolition waves of 1301 and 1321 were recent
# memory in 1337, and conversion accelerated after them. That narrowness is why
# the rule is plausible rather than historical.
#
# ENGINE FACT: there is no `coptic` religion in EU5. The game models Copts as
# religion:miaphysite on locations whose culture is lower_egyptian_culture or
# upper_egyptian_culture (5 of 30 populated locations in Lower Egypt, 8 of 30
# in Upper Egypt, close to the roughly 20 percent the sources give for the
# early Mamluk period). `coptic_culture` is defined in
# common/cultures/egypt.txt but is assigned to zero locations and zero
# countries in 1.3, so no country can ever hold it and gating on it alone is
# dead script. It is listed below only so this formable picks it up if a future
# patch ever populates it.
#
# Route: an Egyptian-culture country whose state religion becomes Miaphysite.
# The playable path is a dhimmi or clergy estate carrying
# pa_convert_to_estate_religion in parliament, which is the proposed scenario
# almost exactly: Coptic financiers converting fiscal weight into the state's
# confession.
#
# Level 3 on purpose, below AEG at 4, so a Coptic Egypt can go on to form
# Aegyptus.

CPT_f = {

	level = 3
	required_locations_fraction = 0.75
	rule = plausible

	potential = {
		religion = religion:miaphysite
		OR = {
			culture = culture:coptic_culture # dormant in 1.3, see header
			culture = culture:lower_egyptian_culture
			culture = culture:upper_egyptian_culture
		}
	}

	# The three anchors of a Coptic state rather than a share of desert edge:
	# Cairo (the revenue offices and the patriarch's residence since 1047),
	# Wadi el-Natrun (the monastic heart, where the popes are chosen), and
	# Asyut (the Coptic heartland of Upper Egypt). All three sit inside the two
	# listed areas, so holding them follows from the fraction rather than
	# pushing past it.
	allow = {
		owns = location:cairo
		owns = location:wadi_el_natrun
		owns = location:asyut
	}

	name = CPT
	flag = CPT
	adjective = CPT_ADJ
	tag = CPT
	color = map_CPT

	# The Nile and the Delta only. Cyrenaica and Sinai are neither Coptic nor
	# Egyptian in population, and Nubia is the ally upriver, not the territory:
	# NUB covers that ground and the alliance axis belongs in content.
	areas = {
		lower_egypt_area
		upper_egypt_area
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_cpt_formed

		}
		if = {
			limit = { NOT = { capital = location:cairo } }
			set_capital = location:cairo
		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_cpt.1
		add_stability = stability_mild_bonus
	}
}
