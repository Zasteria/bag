# Regnum Gothorum: coat of arms
#
# No authentic Visigothic banner survives, so nothing here is invented as one.
# The composition is built from two things that do survive and that the
# neo-Gothic claim actually rested on.
#
# The emblem is ce_cross_asturias, the Cruz de la Victoria: the jewelled cross
# with pendant alpha and omega that the Asturian kings raised as the sign of the
# restored Gothic monarchy, and the direct ancestor of every later
# Astur-Leonese-Castilian royal cross. Vanilla uses this texture in exactly one
# place, the arms of the Pelagia dynasty (the kings of Asturias), which is the
# same claim of succession this country makes. It clashes with no country flag.
#
# Gold body with garnet cells and pearls is the material vocabulary of the
# Treasure of Guarrazar, the Visigothic votive crowns found near Toledo: gold
# cloisonne set with garnets, sapphires and pearls, hung above altars with
# pendant letters spelling the king's name. Tyrian ground is the late antique
# royal purple the Toledan kings inherited from Roman ceremony, and it survives
# in Iberian heraldry as the purpure lion of Leon. Purple is used by no other
# Iberian flag, so the arms read as neither Castile, Spain, Aragon nor Occitania.
GOT = {
	pattern = "pattern_solid.dds"
	color1 = "tyrian"
	color2 = "yellow"
	color3 = "pearl"
	color4 = "red"

	colored_emblem = {
		texture = "ce_cross_asturias.dds"
		color1 = color2
		color2 = color3
		color3 = color4
		instance = { position = { 0.5 0.5 } scale = { 0.85 0.85 } }
	}
}
