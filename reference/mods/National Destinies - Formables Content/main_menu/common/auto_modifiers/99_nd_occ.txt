# Grants bureaucracy authority to Occitania once the country has been formed
# and the player has the bureaucracies game-rule enabled.
# Uses the strict tag, not has_or_had_tag, so slot capacity does not stack when
# the player goes on to form a successor. The bureaucracies themselves gate on
# has_or_had_tag and stay selectable.

nd_occ_bureaucracy_authority = {
	potential_trigger = {
		tag = OCC
		has_variable = nd_occ_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
