namespace = nd_pgr

# Formation event : fired from form_effect
#
# The fork is the one Adityawarman actually faced. His 1347 inscription on
# the Amoghapasa statue at Padang Roco, on the Batang Hari in the lowlands,
# styles him Maharajadhiraja of Malayapura, heir to the Melayu line. The
# kingdom he left behind sat somewhere else entirely: up in the highland
# valleys of Tanah Datar, where the crown presided over the nagari and their
# councils rather than ruling them. Whether Pagaruyung faces downriver as an
# imperial successor or inward as the apex of the nagari world is the choice
# offered here.
nd_pgr.1 = {
	type = country_event
	title = nd_pgr.1.t
	desc = nd_pgr.1.d
	outcome = neutral
	# Per-country guard, not fire_only_once: that is per CAMPAIGN, so a
	# tag Majapahit's Adityawarman chain already produced would burn the
	# event before any later former could ever see it (readme.txt:62).
	# The guard is its own variable, not nd_pgr_formed, which form_effect
	# sets directly so content visibility never depends on a delivered popup.
	trigger = {
		NOT = { has_variable = nd_pgr_e1_done }
	}
	immediate = {
		hidden_effect = { set_variable = nd_pgr_e1_done }
	}

	option = {
		name = nd_pgr.1.a
		add_country_modifier = {
			modifier = nd_pgr_malayapura_restored
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pgr_formed
			set_variable = nd_pgr_path_a
		}
	}
	option = {
		name = nd_pgr.1.b
		add_country_modifier = {
			modifier = nd_pgr_alam_minangkabau
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pgr_formed
			set_variable = nd_pgr_path_b
		}
	}
}

# DHE Event 1: Adat basandi Syarak: the formal synthesis of matrilineal adat with Islamic law
nd_pgr.2 = {
	type = country_event
	fire_only_once = yes

	dynamic_historical_event = {
		tag = PGR
		from = 1337.1.1
		to = 1836.1.1
		monthly_chance = 100
	}

	trigger = {
		has_advance = nd_pgr_adat_basandi_syarak
	}

	title = nd_pgr.2.t
	desc = nd_pgr.2.d
	outcome = neutral

	option = {
		name = nd_pgr.2.a
		add_country_modifier = {
			modifier = nd_pgr_syarak_harmony
			years = 15
			mode = add_and_extend
		}
		# The markers are re-set here only as a repair for a PGR that
		# vanilla's flavor_maj.27 produced before this file existed. The
		# PATH variable is deliberately NOT set: the destiny path belongs
		# to the formation choice in nd_pgr.1, and setting it again from a
		# later DHE would leave a country holding both paths at once.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pgr_formed
		}
	}
	option = {
		name = nd_pgr.2.b
		add_country_modifier = {
			modifier = nd_pgr_adat_primacy
			years = 15
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pgr_formed
		}
	}
}
