# ==========================================================================

# ==========================================================================
# THE SUCCESSION CHECKPOINTS: one static event per weak crown, fired at
# each rival claimant holding any pact standing on the pair (no scope
# plumbing has to survive event delivery). Option A is the Krewo lever:
# rulers marry, the hardcoded machinery unions the crowns. Option B is
# Wiener Neustadt: the pact becomes a claim war. The invoke bar is pact
# tier 3, or tier 2 with a Double Betrothal seed, plus standing 60.
# ==========================================================================

# The Austrian succession stumbles (to HUN or BOH).
nd_danube.12 = {
	type = country_event
	title = nd_danube.12.t
	desc = nd_danube.12.d
	outcome = neutral

	trigger = {
		country_exists = c:HAB
	}

	option = {
		name = nd_danube.12.a		# marry the thrones
		trigger = {
			# An AI claimant may only auto-marry an AI crown: a player's
			# realm is never unioned away without the player's own hand.
			OR = {
				is_ai = no
				c:HAB ?= { is_ai = yes }
			}
			has_ruler = yes
			ruler ?= {
				is_adult = yes
				is_married = no
			}
			c:HAB ?= {
				has_ruler = yes
				ruler ?= {
					is_adult = yes
					is_married = no
				}
			}
			OR = {
				ruler ?= { is_female = yes }
				c:HAB.ruler ?= { is_female = yes }
			}
		}
		ai_chance = {
			factor = 30
		}
		ruler = {
			marry_character = c:HAB.ruler
		}
		custom_tooltip = nd_dnb_wedding_tier_tt
		hidden_effect = {
			if = {
				limit = { tag = HUN }
				situation:nd_danube_succession ?= {
					set_variable = { name = nd_dnb_tier_hab_hun value = 3 }
				}
			}
			else_if = {
				limit = { tag = BOH }
				situation:nd_danube_succession ?= {
					set_variable = { name = nd_dnb_tier_hab_boh value = 3 }
				}
			}
		}
		add_prestige = prestige_mild_bonus
	}
	option = {
		name = nd_danube.12.b		# invoke the pact
		trigger = {
			# The weak crown can die between fire and click (annexation or
			# a tag switch, the Tabor campaign class); a dead target makes
			# the CB grant below a dead-scope no-op, so gate the option.
			country_exists = c:HAB
			OR = {
				AND = {
					tag = HUN
					situation:nd_danube_succession ?= {
						OR = {
							var:nd_dnb_tier_hab_hun >= 3
							AND = {
								var:nd_dnb_tier_hab_hun >= 2
								has_variable = nd_dnb_betrothal_hab_hun
							}
						}
						var:nd_dnb_standing_hun >= 60
					}
				}
				AND = {
					tag = BOH
					situation:nd_danube_succession ?= {
						OR = {
							var:nd_dnb_tier_hab_boh >= 3
							AND = {
								var:nd_dnb_tier_hab_boh >= 2
								has_variable = nd_dnb_betrothal_hab_boh
							}
						}
						var:nd_dnb_standing_boh >= 60
					}
				}
			}
		}
		ai_chance = {
			factor = 15
		}
		add_casus_belli = {
			type = casus_belli:cb_claim_throne
			target = c:HAB
		}
		c:HAB ?= { trigger_event_non_silently = nd_danube.15 }
	}
	option = {
		name = nd_danube.12.c		# the moment passes
		ai_chance = {
			factor = 20
		}
	}
}

}
