# Repair for tags vanilla can create WITHOUT the formable button.
#
# `form_country = formable_country:X_f` appears at 25 sites in vanilla:
# situations (sengoku forms JAP, rise_of_the_ottomans forms RUM), DHE flavor
# chains (MOL, AYU, TUS), peace treaties (TIM) and generic actions (IRE). A
# country arriving that way never had a player click the formation popup.
#
# The formation markers are set in every form_effect, which covers this if
# form_country runs form_effect. That is unverified against a live game, and
# the failure mode is silent and total: every advance gated on the marker
# stays unresearchable for the rest of the campaign. Reported on the Workshop
# 2026-08-08 for JAP formed at the end of the Sengoku Jidai.
#
# So the state is repaired independently on the four-yearly check. Every
# branch tests for the marker's ABSENCE, so this is free once the markers are
# present, and it repairs a save that is already broken.

nd_repair_script_formed_effect = {
	if = {
		limit = {
			exists = c:$TAG$
			NOT = { c:$TAG$ = { has_variable = nd_$LTAG$_formed } }
		}
		c:$TAG$ = {
			hidden_effect = {
				set_variable = nd_formed
				set_variable = nd_$LTAG$_formed
			}
			trigger_event_non_silently = { id = nd_$LTAG$.1 days = 3 }
		}
	}
}

# Same repair, without the day-3 event call.
#
# For the event-formed tags (FORMABLES.md section "Event-Formed Countries")
# nd_$LTAG$.1 is not a formation event at all: it is an advance-gated DHE, so
# calling it here would only fail its own trigger, exactly as the DAH and YUA
# note in 99_nd_on_actions.txt records. These tags have no formation popup to
# deliver; they need the markers and nothing else.
#
# MLC is the reason this cannot be left to form_effect alone. Nothing in the
# game calls `form_country = formable_country:MLC_f`; flavor_maj.txt:1493
# builds Malacca with create_country_from_location and
# `define_unique_country_tag = MLC`, which never runs a form_effect. This
# sweep is the only thing that ever sets nd_mlc_formed.
nd_repair_script_formed_marker_only = {
	if = {
		limit = {
			exists = c:$TAG$
			NOT = { c:$TAG$ = { has_variable = nd_$LTAG$_formed } }
		}
		c:$TAG$ = {
			hidden_effect = {
				set_variable = nd_formed
				set_variable = nd_$LTAG$_formed
			}
		}
	}
}
