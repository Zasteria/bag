# ==========================================================================
# THE CROWN CONFIDENCE COUNCIL (Item 2, dnm_proposal): Systems Library
# System 5, first disaster-scoped instantiation.
#
# Replaces the Pragmatic Sanction disaster's RNG-gated agency events
# (nd_dnm.40/.41/.43, retired 2026-07-31 from the disaster's on_monthly
# random_list) and the RNG-offered fight-or-cede decision (nd_dnm.23) with
# five on-demand actions the player invokes on their own schedule. The
# disaster's EXTERNAL shocks (nd_dnm.42/.44/.45) deliberately stay random:
# they happen TO the crown, not FOR it.
#
# Modeled field-by-field on the two vanilla disaster-scoped precedents:
# generic_actions/decline_of_majapahit_actions.txt and
# generic_actions/crisis_of_the_chinese_dynasty_actions.txt. The readme's
# own `type =` enum omits `disaster`; the two shipped vanilla files above
# are trusted over the stale enum (gameplay-literacy rule).
#
# SHARED COOLDOWN: the three resource-spending initiatives share one
# 3-year recess via the timed variable nd_dnm_council_recess. A native
# `cooldown = { type = X }` with one tag on several actions has NO vanilla
# precedent (every vanilla cooldown tag is used by exactly one action), so
# whether the engine would share it is unverified; the timed variable is
# the same guarantee through a mechanism this mod already relies on
# everywhere. The two stance actions (press the claim / sue for terms)
# carry no cooldown: a stance is not a resource tap.
#
# Costs are the exact tiers of the retired events (income-scaled add_gold
# with a floor via max = -N), spent in the effect rather than a price:
# definition, matching the disaster's own established cost idiom.
# ==========================================================================

# Petition the estates for loyalty: the retired nd_dnm.40's top tier.
# A magnate's faction bought into line: 12 months income (floor 100),
# +15 Crown Confidence.
nd_dnm_petition_estate_loyalty = {
	# Icon picked by eye from the generic-actions contact sheet
	# (gold placed in an open hand: loyalty, bought); without an icon field every action renders the
	# shared _default.dds, below vanilla's own presentation bar.
	icon = union_bribe
	show_message = no
	type = disaster

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = { tag = DNM }
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnm_council_recess_ready_tt
				NOT = { has_variable = nd_dnm_council_recess }
			}
		}
	}

	select_trigger = {
		looking_for_a = disaster
		target_flag = recipient
		name = "choose_disaster"
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:nd_dnm_pragmatic_sanction_crisis
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			add_gold = {
				value = monthly_income_trade_and_tax
				multiply = -12
				max = -100
			}
			custom_tooltip = {
				text = nd_dnm_ga_confidence_plus15_tt
				change_variable = { name = nd_dnm_crown_confidence add = 15 }
			}
			custom_tooltip = {
				text = nd_dnm_council_recess_applied_tt
				hidden_effect = {
					set_variable = { name = nd_dnm_council_recess value = yes years = 3 }
				}
			}
		}
	}

	ai_will_do = {
		value = -50
		add = {
			if = {
				limit = {
					scope:actor = {
						gold >= {
							value = monthly_income_trade_and_tax
							multiply = 18
						}
					}
				}
				value = 150
			}
		}
	}
}

# Sue for succession terms: the honest exit from the retired nd_dnm.23

	select_trigger = {

		target_flag = recipient

					}
				}
				value = 200
			}
		}
	}

# ==========================================================================
# THE NATIONALITIES QUESTION: on-demand exits (2026-07-31, the
# crisis-lifecycle audit). The NQ's three exits were previously reachable
# only when the disaster's monthly random_list deigned to OFFER the
# resolution event (2-3% a month): a player who had raised control above
# the reassertion target still waited years for nd_dnm.16 to roll (the
# red-turban class, flagged by eu5-validate --checks crisis_lifecycle).
# These three actions are the same three exits as levers the player pulls
# the moment they are earned. The monthly events remain as flavor and as
# an alternative delivery; both roads set the same resolution variables.
# ==========================================================================

# Break the Leagues: the reassertion exit (mirror of nd_dnm.16.a). The
# deed is the control target; the action is available the moment it is met.
nd_dnm_break_the_leagues = {
	icon = pw_crush_rebellion
	show_message = no
	type = disaster

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			tag = DNM
			NOT = { has_variable = nd_dnm_nq_resolved }
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnm_nq_control_met_tt
				has_variable = nd_dnm_nq_control_target
				average_control_in_home_region >= var:nd_dnm_nq_control_target
			}
		}
	}

	select_trigger = {
		looking_for_a = disaster
		target_flag = recipient
		name = "choose_disaster"
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:nd_dnm_nationalities_question
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnm_ga_reassert_tt
				hidden_effect = {
					set_variable = nd_dnm_nq_reasserted
					set_variable = nd_dnm_nq_resolved
				}
			}
		}
	}

	ai_will_do = {
		value = -50
		add = {
			if = {
				limit = {
					scope:actor = {
						has_variable = nd_dnm_nq_control_target
						average_control_in_home_region >= var:nd_dnm_nq_control_target
					}
				}
				value = 200
			}
		}
	}
}
