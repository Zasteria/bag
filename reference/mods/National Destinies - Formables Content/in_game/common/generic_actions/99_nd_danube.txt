# ==========================================================================
# THE SUCCESSION OF THE DANUBE: situation-scoped actions. Modeled on the
# vanilla situation-action idiom (guelphs_and_ghibellines.txt) and the
# DNM Council actions' cost and cooldown conventions.
# ==========================================================================

# Convene the Diet of the Three Crowns early instead of waiting out the
# ten-year cadence: seize the agenda while your standing is high.
nd_dnb_convene_the_diet = {
	# Icon picked by eye from the generic-actions pool (a summoned
	# assembly); without an icon field the action renders _default.dds.
	icon = call_parliament
	show_message = no
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_convene_ready_tt
				NOT = { has_variable = nd_dnb_convene_cd }
			}
		}
		# GLOBAL floor: one diet per 5 years no matter who convenes. The
		# per-actor cooldown alone let three courts take turns summoning
		# diets months apart (the 1444 diet-spam cascade, 2026-08-02).
		custom_tooltip = {
			text = nd_dnb_convene_global_tt
			situation:nd_danube_succession ?= {
				NOT = { has_variable = nd_dnb_convened_recently }
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:nd_danube_succession = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_convene_applied_tt
				hidden_effect = {
					set_variable = { name = nd_dnb_convene_cd value = yes years = 5 }
				}
			}
		}
		situation:nd_danube_succession ?= {
			nd_dnb_hold_diet = yes
		}
	}

	# Negative base for the same reason as Renounce: a 0 base means the
	# AI convenes on cooldown forever.
	ai_will_do = {
		value = -50
		add = {
			if = {
				limit = {
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 2 }
				}
				value = 90
			}
		}
	}
}

# Renounce the Question: the user's early-out. Available when your own
# candidacy dominates and you would rather close the book, or when the
# whole dynastic web is already dead. Records the Separate Ways marker
# the situation's can_end reads: a named on-demand exit, per the Crisis
# Lifecycle Contract.
nd_dnb_renounce_the_question = {
	# Icon picked by eye from the generic-actions pool (gates shut: each
	# crown keeps its own house).
	icon = close_the_borders
	show_message = no
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
		}
	}

	# Dominance only (2026-08-02 causes rework): the collective walk-away
	# now lives in the Old Liberties cause, pushed over decades by its
	# backers; it is not a clickable exit. Renounce remains solely the
	# dominant player's deliberate closure of the book.
	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_renounce_req_tt
				OR = {
					AND = {
						tag = HAB
						situation:nd_danube_succession ?= {
							var:nd_dnb_standing_hab >= 60
							var:nd_dnb_standing_hun <= 40
							var:nd_dnb_standing_boh <= 40
						}
					}
					AND = {
						tag = HUN
						situation:nd_danube_succession ?= {
							var:nd_dnb_standing_hun >= 60
							var:nd_dnb_standing_hab <= 40
							var:nd_dnb_standing_boh <= 40
						}
					}
					AND = {
						tag = BOH
						situation:nd_danube_succession ?= {
							var:nd_dnb_standing_boh >= 60
							var:nd_dnb_standing_hab <= 40
							var:nd_dnb_standing_hun <= 40
						}
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:nd_danube_succession = this
			situation_is_active = yes
		}
	}

	effect = {
		situation:nd_danube_succession ?= {
			custom_tooltip = {
				text = nd_dnb_renounce_applied_tt
				set_variable = nd_dnb_separate_ways
			}
		}
	}

	# The AI NEVER renounces (2026-08-02 causes rework): its walk-away is
	# expressed through backing the Old Liberties cause. Renounce is the
	# player's deliberate closure only.
	ai_will_do = {
		value = -1000
	}
}

# ==========================================================================
# THE STANCES (2026-08-03 rework, user ruling): allegiance is an
# on-demand declaration, never a popup re-roll. G&G join/leave model:
# switch at will behind a 3-year court cooldown; the AI declares from
# circumstances (menace, weakness, who leads) and holds its line.
# ==========================================================================

nd_dnb_back_the_union = {
	# Icon picked by eye (banners joining a league).
	icon = join_religious_league
	show_message = no
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_stance_ready_tt
				NOT = { has_variable = nd_dnb_stance_cd }
			}
			custom_tooltip = {
				text = nd_dnb_not_already_union_tt
				OR = {
					AND = {
						tag = HAB
						situation:nd_danube_succession ?= {
							NOT = { var:nd_dnb_posture_hab = 1 }
						}
					}
					AND = {
						tag = HUN
						situation:nd_danube_succession ?= {
							NOT = { var:nd_dnb_posture_hun = 1 }
						}
					}
					AND = {
						tag = BOH
						situation:nd_danube_succession ?= {
							NOT = { var:nd_dnb_posture_boh = 1 }
						}
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:nd_danube_succession = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_stance_union_applied_tt
				hidden_effect = {
					nd_dnb_set_own_posture = { VALUE = 1 }
					set_variable = { name = nd_dnb_stance_cd value = yes years = 3 }
					nd_dnb_announce_stance = yes
					# v2: declaring is investing. Postures no longer feed
					# the causes monthly, so the declaration itself pays
					# its cause once (PLAN.md 13/v2).
					situation:nd_danube_succession ?= {
						nd_dnb_bump_cause = { CAUSE = union AMOUNT = 8 }
					}
				}
			}
		}
	}

	ai_will_do = {
		value = -20
		add = {
			if = {
				limit = {
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 2 }
				}
				value = 40
			}
		}
		add = {
			if = {
				limit = {
					scope:actor = { nd_dnb_weak_succession = yes }
				}
				value = 30
			}
		}
		add = {
			# The strongest crown wants to lead the union, not fear it.
			if = {
				limit = {
					OR = {
						AND = {
							scope:actor = { tag = HAB }
							situation:nd_danube_succession ?= {
								var:nd_dnb_standing_hab >= var:nd_dnb_standing_hun
								var:nd_dnb_standing_hab >= var:nd_dnb_standing_boh
							}
						}
						AND = {
							scope:actor = { tag = HUN }
							situation:nd_danube_succession ?= {
								var:nd_dnb_standing_hun >= var:nd_dnb_standing_hab
								var:nd_dnb_standing_hun >= var:nd_dnb_standing_boh
							}
						}
						AND = {
							scope:actor = { tag = BOH }
							situation:nd_danube_succession ?= {
								var:nd_dnb_standing_boh >= var:nd_dnb_standing_hab
								var:nd_dnb_standing_boh >= var:nd_dnb_standing_hun
							}
						}
					}
				}
				value = 30
			}
		}
		add = {
			# Parchment argues for its own conclusion (see the liberties
			# stance for the 1444 finding).
			if = {
				limit = {
					OR = {
						AND = {
							scope:actor = { tag = HAB }
							situation:nd_danube_succession ?= {
								OR = {
									var:nd_dnb_tier_hab_hun >= 2
									var:nd_dnb_tier_hab_boh >= 2
								}
							}
						}
						AND = {
							scope:actor = { tag = HUN }
							situation:nd_danube_succession ?= {
								OR = {
									var:nd_dnb_tier_hab_hun >= 2
									var:nd_dnb_tier_hun_boh >= 2
								}
							}
						}
						AND = {
							scope:actor = { tag = BOH }
							situation:nd_danube_succession ?= {
								OR = {
									var:nd_dnb_tier_hab_boh >= 2
									var:nd_dnb_tier_hun_boh >= 2
								}
							}
						}
					}
				}
				value = 20
			}
		}
	}
}

# ==========================================================================
# HONOR THE LEAGUE (2026-08-03, user design): the pact's defensive half.
# Any pact (tier 1+) lets a crown join a partner's DEFENSIVE war against
# the Ottomans, on demand. Ottoman-led wars ONLY (user ruling 2026-08-03):
# a generic outsider filter dragged crowns into unrelated neighbors' wars
# and wrecked their gameplay. Modeled on iw_intervene_in_war
# (italian_wars.txt:1395): war select_trigger + join_war_with.
# ==========================================================================
nd_dnb_honor_the_league = {
	# Icon picked by eye (armed support under a sanctioned banner).
	icon = guelphs_sanction_emperor
	show_message = no
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			nd_dnb_crown_independent = yes
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = nd_dnb_league_peace_tt
				at_war = no
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:nd_danube_succession = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = war
		interaction_source_list = {
			c:HAB ?= {
				save_temporary_scope_as = nd_dnb_partner
				every_current_war = {
					limit = {
						defender_leader = scope:nd_dnb_partner
						NOT = { is_in_war = scope:actor }
						attacker_leader = c:TUR
						NOT = { scope:actor = { tag = HAB } }
						scope:nd_dnb_partner = { nd_dnb_crown_independent = yes }
						nd_dnb_pact_tier_ge_p = {
							X = scope:actor
							Y = scope:nd_dnb_partner
							N = 1
						}
					}
					add_to_list = source
				}
			}
			c:HUN ?= {
				save_temporary_scope_as = nd_dnb_partner
				every_current_war = {
					limit = {
						defender_leader = scope:nd_dnb_partner
						NOT = { is_in_war = scope:actor }
						attacker_leader = c:TUR
						NOT = { scope:actor = { tag = HUN } }
						scope:nd_dnb_partner = { nd_dnb_crown_independent = yes }
						nd_dnb_pact_tier_ge_p = {
							X = scope:actor
							Y = scope:nd_dnb_partner
							N = 1
						}
					}
					add_to_list = source
				}
			}
			c:BOH ?= {
				save_temporary_scope_as = nd_dnb_partner
				every_current_war = {
					limit = {
						defender_leader = scope:nd_dnb_partner
						NOT = { is_in_war = scope:actor }
						attacker_leader = c:TUR
						NOT = { scope:actor = { tag = BOH } }
						scope:nd_dnb_partner = { nd_dnb_crown_independent = yes }
						nd_dnb_pact_tier_ge_p = {
							X = scope:actor
							Y = scope:nd_dnb_partner
							N = 1
						}
					}
					add_to_list = source
				}
			}
		}
		target_flag = target_war
		name = "choose_war_to_join"
		none_available_msg_key = "nd_dnb_league_no_wars"
		column = {
			data = name
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target_war }
			scope:target_war = {
				defender_leader = { save_scope_as = nd_dnb_defended }
			}
			scope:actor = {
				join_war_with = {
					target = scope:nd_dnb_defended
					war = scope:target_war
					call_in_subjects = yes
				}
			}
		}
	}

	# IW's intervention weighting shape: solvency and manpower embolden,
	# exhaustion restrains; the gates and a deep pact push a crown to
	# actually ride out.
	ai_will_do = {
		add = {
			value = -40
			desc = "BASE_VALUE"
		}
		add = {
			if = {
				limit = {
					situation:nd_danube_succession ?= { var:nd_dnb_menace >= 2 }
				}
				value = 40
			}
		}
		if = {
			limit = { scope:actor.war_exhaustion > 0 }
			subtract = {
				value = scope:actor.war_exhaustion
				multiply = 2
			}
		}
		if = {
			limit = { scope:actor.manpower_percentage > 0.5 }
			add = {
				value = scope:actor.manpower_percentage
				multiply = 40
			}
		}
	}
}
