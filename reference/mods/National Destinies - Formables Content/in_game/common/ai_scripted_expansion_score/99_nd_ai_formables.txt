# ==========================================================================
# AI FORMABLE AMBITION (Layer 2 of the nd_ai_formables game rule,
# "ambitious" setting only; 2026-07-31).
#
# Boosts the AI's war-target scoring against countries that hold land the
# attacker still needs for a major formable it is already meaningfully on
# the way to. Uses the vanilla ai_scripted_expansion_score system (readme
# in this directory; live example guelphs_and_ghibellines.txt) and the
# vanilla-used location_progress_for_formable trigger
# (scripted_triggers/situation_triggers.txt:542).
#
# Semantics per entry:
#   attacker_potential (root = attacker): AI, rule active, and the
#     attacker already owns >= 30% of the formable's required locations.
#   target_trigger (root = TARGET): the target itself owns >= 5% of that
#     formable's required locations, i.e. it holds land the attacker
#     needs.
#   score: +15 against EXPANSION_TARGET_SCORE_NEEDED_TO_PICK = 20 /
#     NEEDED_TO_KEEP = 15 (00_defines.txt:1187-1188), a strong, visible
#     thumb on the scale, deliberately so for the pilot.
#
# Pilot five chosen away from Central Europe (the DNM playtest theater):
# Spain, Russia, Great Britain, Scandinavia, the Mughals. Germany/Italy
# are Layer-1 only for now, to keep the user's Danube campaign readable.
# ==========================================================================

nd_ai_form_spain = {
	attacker_potential = {
		is_ai = yes
		has_game_rule = nd_ai_formables_ambitious
		location_progress_for_formable = {
			formable_country = formable_country:SPA_f
			value >= 0.3
		}
	}
	target_trigger = {
		location_progress_for_formable = {
			formable_country = formable_country:SPA_f
			value >= 0.05
		}
	}
	score = {
		value = 15
	}
}
