# Josephine Triumph's unlock: Joseph II's own actual programme, a single
# administrative code and bureaucratic language across the crown lands
# (dnm_facts #29), completed here where the historical version was fought
# to a draw. It also finishes what the heritage advance
# nd_dnm_united_chancellery only attempted. The orthogonal cost is the
# estates' privilege, which the uniform patent overrides.
nd_dnm_uniform_patent_reform = {
	age = age_6_revolutions
	# Bare key, not government_type: the scoped form fails at load
	# ("Failed to read key reference", error.log 2026-07-31); vanilla
	# government_reforms/monarchy.txt:2 uses the bare key.
	government = monarchy
	major = yes

	potential = {
		has_or_had_tag = DNM
	}

	allow = {
		has_advance = nd_dnm_dest_josephine_triumph
	}

	country_modifier = {
		legislative_efficiency = 0.1
		global_pop_assimilation_speed_modifier = 0.15
		nobles_estate_target_satisfaction = -0.05
	}

	years = 2
}

# because the diets must actually be consulted.
