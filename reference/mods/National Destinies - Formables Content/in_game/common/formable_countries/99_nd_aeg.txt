AEG_f = {

	level = 4
	required_locations_fraction = 0.7
	rule = plausible

	# Aegyptus is the *place*, not a confession: the province of Egypt held by
	# whoever rules the Nile from Cairo. Two claimants qualify.
	#
	# The Coptic branch was already listed in the culture OR but was dead
	# script: `religion = religion:sunni` sat above it as a hard requirement,
	# so the only country it ever admitted was a Sunni-converted Copt. Coptic
	# Egypt is Miaphysite (there is no `coptic` religion in EU5; Copts are
	# religion:miaphysite + culture:coptic_culture), which meant the one
	# campaign the branch was written for (a Coptic restoration of Roman
	# Egypt) was the one campaign locked out. Reported on the Workshop
	# 2026-07-05. Each claimant now carries its own religion.
	potential = {
		OR = {
			AND = {
				religion = religion:sunni
				culture = { has_culture_group = culture_group:arabic_group }
			}
			AND = {
				religion = religion:miaphysite
				# `coptic_culture` is defined in common/cultures/egypt.txt but is
				# assigned to zero locations and zero countries in 1.3, so no
				# country can ever hold it. On its own this branch was therefore
				# still dead script after the 2026-07-05 religion fix. EU5 models
				# Copts as miaphysite pops on Egyptian-culture locations, so the
				# Egyptian cultures are what actually admits a Coptic claimant,
				# and what lets CPT (Coptic Egypt, level 3) chain up into
				# Aegyptus. The dormant culture stays listed in case a future
				# patch populates it.
				OR = {
					culture = culture:coptic_culture
					culture = culture:lower_egyptian_culture
					culture = culture:upper_egyptian_culture
				}
			}
		}
	}

	allow = {
		owns = location:cairo
		owns = location:damascus
		owns = location:mecca
		owns = location:dongola
		owns = location:benghazi
		in_civil_war = no
	}

	name = AEG
	flag = AEG
	adjective = AEG_ADJ
	tag = AEG
	color = map_AEG

	regions = {
		egypt_region
		nubia_region
	}
	areas = {
		levant_area
		hedjaz_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_aeg_formed
		}
		if = {
			limit = { NOT = { capital = location:cairo } }
			set_capital = location:cairo
		}
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		trigger_event_non_silently = nd_aeg.1
		add_stability = stability_severe_bonus
	}
}
