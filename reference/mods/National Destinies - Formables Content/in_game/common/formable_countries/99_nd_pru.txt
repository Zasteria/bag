# Intentional divergence from vanilla 1.2: omits `content_priority` (UI
# ordering hint, no functional effect). Mod's REPLACE preserves vanilla's
# allow/potential surface; the missing field is by design.
REPLACE:PRU_f = {
	level = 2
	required_locations_fraction = 0.75
	rule = historical

	capital_required = no

	potential = {
		culture = { has_culture_group = culture_group:german_group }
	}
	allow = {
		NOT = {
			has_reform = government_reform:military_order_reform 
			has_estate_privilege = estate_privilege:clergy_powerful_dioceses
		}
		has_estate_privilege = estate_privilege:nobles_land_rights

		"estate_power(estate_type:clergy_estate)" < 0.20
		"estate_power(estate_type:nobles_estate)" > 0.20
		at_war = no
	}
	name = PRU
	flag = PRU
	adjective = PRU_ADJ
	tag = PRU
	color = map_PRU
	
	
	areas = {
		prussia_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pru_formed
		}
		trigger_event_non_silently = nd_pru.1
		add_stability = stability_mild_bonus
		if = {
			limit = {
				government_type != government_type:republic
				government_type != government_type:monarchy
			}
			change_government_type = government_type:monarchy
			add_legitimacy = legitimacy_radical_bonus
		}
		if = {
			limit = {
				nd_can_take_kingdom_rank = yes
			}
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}
