# Outremer, the capstone of the mod's crusader chain.
#
# Level 4 is load-bearing, not cosmetic. The engine gates formation on the
# formable level of the FORMING country's own tag and requires strictly
# higher, so 4 is the only tier that keeps every crusader formable as a road
# in: Saint John (1), Crusader Africa (2), Jerusalem (3), the Latin Empire
# (3) and the Temple (3). (Re-tiered 2026-08-10: HOS to 1 and KOJ back to
# vanilla's 3, giving the chain HOS, then KOJ or LAT as mutually exclusive
# second acts, then OUT.) At level 3 Jerusalem and the Temple would be
# locked out, which is the bug this project already shipped twice (Tabor,
# Hen Ogledd). Countries with no formable definition of their own (Cyprus,
# Naples, Venice, France) carry no tier and qualify regardless.
#
# rule = plausible, defended rather than assumed. Outremer existed under this
# exact name and ended 46 years before the 1337 start, which is already the
# mod's own definition of plausible ("existed outside the timeframe"). Every
# city in `allow` was a real Frankish war aim: Cairo (Damietta 1219 and 1249,
# the Franks a march from Cairo in 1167), Baghdad and Mosul (taken in 1258 by
# a Mongol army with Bohemond VI of Antioch and Hethum I of Armenia in it),
# Mecca and Medina (Raynald of Chatillon's Red Sea flotilla of 1182-1183 put a
# landing party ashore between them), Aden (Albuquerque assaulted it in 1513 as
# Portuguese state policy). The extent is a hard stretch; the requirements are
# priced accordingly. It is also calibrated against the mod's own neighbours:
# NIN (a Syriac empire over the same ground, with no state at all in 1337) and
# AEG are both level 4 plausible, so rating Outremer fantasy would make it
# sillier than a restored Nineveh, which is indefensible.
OUT_f = {

	level = 4
	required_locations_fraction = 0.6
	rule = plausible

	potential = {
		religion = religion:catholic

		# The Latin East, not any Catholic realm that reaches the Jordan. The
		# five crusader formables are named rather than left to the rank
		# clause because Crusader Africa is the one that does not raise rank
		# on formation and would otherwise fall through.
		OR = {
			has_or_had_tag = KOJ
			has_or_had_tag = LAT
			has_or_had_tag = KOA
			has_or_had_tag = TKT
			has_or_had_tag = HOS
			has_reform = government_reform:military_order_reform
			country_rank = country_rank:rank_kingdom
			country_rank = country_rank:rank_empire
		}
	}

	allow = {
		# The vow and the seat. Jerusalem is definitional and the Workshop
		# proposal omitted it; Acre was the kingdom's actual capital from 1191
		# and its last mainland holding in 1291.
		owns = location:jerusalem
		owns = location:acre

		# The Egyptian project, the Mesopotamian dividend of the Mongol
		# alliance, the Hejaz that Raynald sailed for, and the Red Sea mouth.
		owns = location:cairo
		owns = location:baghdad
		owns = location:mosul
		owns = location:mecca
		owns = location:medina
		owns = location:adan

		in_civil_war = no

		# The governance clause, and the point of the whole country. Outremer
		# was a Frankish ruling caste over an overwhelmingly non-Latin
		# population, so a realm that has converted or emptied the East into a
		# Catholic majority is not Outremer, it is Jerusalem written large.
		# This also excludes a European superpower that keeps the Middle East
		# as an appendage: France with a Levantine province is France. The
		# country is named for being overseas, and this is the line that
		# enforces it.
		religion_percentage_in_country = {
			religion = religion:catholic
			value <= 0.5
		}
	}

	name = OUT
	flag = OUT
	adjective = OUT_ADJ
	tag = OUT
	color = map_OUT

	regions = {
		crescent_region
		egypt_region
	}
	areas = {
		hedjaz_area
		south_yemen_area
	}
	provinces = {
		# The seat of the exiled kingdom after 1291 and the staging base for
		# Peter I's Alexandria crusade of 1365.
		cyprus_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_out_formed

		}
		if = {
			limit = { NOT = { capital = location:jerusalem } }
			set_capital = location:jerusalem
		}
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		add_stability = stability_severe_bonus
		trigger_event_non_silently = nd_out.1
	}
}
