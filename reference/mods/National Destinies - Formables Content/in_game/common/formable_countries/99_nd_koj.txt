# Level 3, back to the vanilla tier (2026-08-10). The mod ran this at 2
# for the earlier-restoration philosophy, but that made KOJ a sibling of
# HOS instead of its successor: the strictly-higher gate reads the
# forming tag's own formable level, so at 2 the Hospitallers could never
# form Jerusalem. The topology is now HOS (1), then KOJ (3) or LAT (3)
# as mutually exclusive second acts, then OUT (4). Being level with LAT
# also kills the backwards Jerusalem-then-Constantinople road.
REPLACE:KOJ_f = {
	level = 3
	required_locations_fraction = 0.75
	rule = plausible

	capital_required = no

	potential = {
		religion = religion:catholic
		OR = {
			tag = CYP
			tag = NAP
			has_reform = government_reform:military_order_reform
		}
	}

	allow = {
		owns = location:jerusalem
	}
	
	name = KOJ
	flag = KOJ
	adjective = KOJ_ADJ
	tag = KOJ
	color = map_KOJ

	provinces = {
		filastin_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_koj_formed

		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		if = {
			limit = {
				has_variable = form_koj_variable
			}
			add_stability = stability_extreme_bonus
			add_government_power = government_power_extreme_bonus
			add_country_modifier = {
				modifier = jerusalem_resurgent_modifier
				years = 20
				mode = add_and_extend
			}
			remove_variable = form_koj_variable
		}
		trigger_event_non_silently = nd_koj.1
	}
}
