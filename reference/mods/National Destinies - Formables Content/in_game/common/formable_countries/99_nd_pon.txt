# Pontus (PON): REPLACE the vanilla [p] form_effect.
# Vanilla promotes to kingdom rank only. We preserve that and add the
# formation event trigger so the player chooses between the Komnenian
# Restoration and the Pontic Republic at formation.
REPLACE:PON_f = {

	level = 3
	required_locations_fraction = 0.75
	rule = fantasy

	potential = {
		OR = {
			culture = culture:pontic_greek_culture
			owns = location:trebizond
		}
	}

	allow = {
		AND = {
			culture = culture:pontic_greek_culture
			owns = location:trebizond
		}
	}

	name = PON
	flag = PON
	adjective = PON_ADJ
	tag = PON
	color = map_pontus

	areas = {
		pontus_area
		crimea_area
	}

	provinces = {
		taman_province
		samegrelo_province
		abkhazia_province

		amasya_province
		tokat_province
		erzurum_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_pon_formed

		}
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
		trigger_event_non_silently = nd_pon.1
	}
}
