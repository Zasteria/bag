TKT_f = {

	level = 3
	required_locations_fraction = 0.5
	rule = plausible

	capital_required = no

	# You must actually BE a military order. The old clause let the culture
	# groups satisfy the OR on their own, so every Catholic Iberian, French
	# and British country (Castile, Portugal, Aragon, France, England and
	# every minor among them) saw the Templars in its formable list, and the
	# AI took it often enough that players reported seeing a Templar state
	# "every other game" and asked for a toggle to switch it off (Workshop,
	# 2026-06-10 and 2026-07-21).
	#
	# The Templars were a sovereign military order, not a kingdom that renamed
	# itself, so the order government IS the requirement rather than one way
	# of meeting it. It stays reachable on purpose: military_order_reform
	# needs the clergy_military_orders privilege plus the monastic_order
	# policy (government_reforms/theocracy.txt), which is a deliberate run
	# nobody adopts by accident. The culture groups remain as a geographic
	# filter on top: this is the Iberian/Frankish Templar successor, not a
	# Livonian or Levantine one.
	potential = {
		religion = religion:catholic
		has_reform = government_reform:military_order_reform
		OR = {
			culture = { has_culture_group = culture_group:iberian_group }
			culture = { has_culture_group = culture_group:french_group }
			culture = { has_culture_group = culture_group:british_group }
		}
	}

	allow = {
		owns = location:tomar
		OR = {
			owns = location:peniscola
			owns = location:monzon
		}
	}

	name = TKT
	flag = TKT
	adjective = TKT_ADJ
	tag = TKT
	color = map_TKT

	areas = {
		south_portugal_area
		valencia_area
		aragon_area
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_tkt_formed

		}
		trigger_event_non_silently = nd_tkt.1
		add_stability = stability_mild_bonus
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		# TKT is the Templar successor as a sovereign military-religious
		# order, and `potential` now requires military_order_reform, whose
		# own prerequisite is this privilege, so a former almost always
		# holds it already. Kept (guarded) for the case where the privilege
		# was revoked after the reform was adopted; the militant_brotherhood
		# modifier then applies the institutional benefits immediately
		# without forcing a reform-slot swap.
		if = {
			limit = { NOT = { has_estate_privilege = estate_privilege:clergy_military_orders } }
			grant_estate_privilege = estate_privilege:clergy_military_orders
		}
		add_country_modifier = {
			modifier = nd_tkt_militant_brotherhood
			years = -1
		}
	}
}
