# Regnum Gothorum: National Destinies new formable
#
# The Visigothic kingdom of Toledo (507-711) restored as a living state rather
# than remembered as a lost one. This is not nostalgia bolted onto Castile: the
# claim to continue the regnum Gothorum was the load bearing legitimising
# ideology of the Asturian, Leonese and Castilian monarchy, the reason the
# Reconquista was framed in law as a restoration of a kingdom that still legally
# existed rather than as a conquest of a foreign one. Alfonso II rebuilt
# Visigothic court ritual at Oviedo, Alfonso III commissioned the chronicles
# that welded the Asturian kings to the Toledan ones, the Liber Iudiciorum
# stayed in force as the Fuero Juzgo, Toledo kept its ecclesiastical primacy
# over the peninsula, and the Leonese imperial title (Imperator totius
# Hispaniae) rested on the doctrine that whoever held Leon was the most direct
# representative of the Gothic kings.
#
# LEVEL 3 IS DELIBERATE. Spain (SPA_f) is also level 3 and the engine only lets
# a country form a formable of strictly higher level, so a country that has
# formed Spain can never form this, and a country that forms this can never
# form Spain. That mutual exclusion is the design: GOT is an alternative
# Iberian destiny for Castile, not a larger Spain stacked on top of it. Castile
# itself is a starting country with no formable level, so it can reach both.
#
# TERRITORY. Iberia plus Septimania. The three Languedoc provinces listed here
# are Septimania proper, the Gothic province in Gaul that survived the loss of
# Toulouse at Vouille in 507 and stayed Gothic until 720: the seven sees of
# Narbonne, Beziers, Agde, Lodeve, Maguelonne (melgueil), Nimes and Carcassonne
# all fall inside narbonnais, nimois and razes. Elne, the eighth candidate, sits
# in rosello_province and is already inside iberia_region. Toulouse and the
# Occitan interior are deliberately excluded: they were lost a full two
# centuries before the kingdom this formable restores, and they belong to
# Occitania (OCC_f), which this overlaps with only on the coastal corridor.
GOT_f = {

	level = 3
	required_locations_fraction = 0.75
	rule = plausible

	potential = {
		# Neo-Gothicism was a specific dynastic doctrine, not a general
		# Iberian sentiment. It ran Asturias to Leon to Castile, which is
		# why the gate is these two cultures and not the Iberian culture
		# group: Aragon and Catalonia built their legitimacy on the
		# Carolingian March and the Mediterranean crown instead (CDA_f),
		# and Portugal's national myth is Afonsine rather than Gothic.
		OR = {
			culture = culture:castilian
			culture = culture:leonese
		}
		# The Third Council of Toledo (589) fused crown and church, and
		# every later Gothic-succession argument was made inside a
		# Christian legal tradition. Kept at the group rather than at
		# Catholic so a reformed crown can still argue the succession
		# through the councils and the code.
		religion.group = religion_group:christian
		# Restates the level rule so Spain never sees a greyed button for
		# the destiny it has already foreclosed.
		NOT = { tag = SPA }
	}

	allow = {
		# Toledo: the royal city, the seat of the councils, and the see
		# that held primacy over all Iberia. Restoring the regnum without
		# it is meaningless.
		owns = location:toledo
		# Granada: the last Muslim polity in the peninsula. Until it falls
		# the restoration is a claim, not a fact.
		owns = location:granada
		# Narbonne: capital of Septimania and the last Gothic seat in Gaul.
		# This is the requirement that makes the formable a different shape
		# from Spain rather than a bigger one, because it forces a war
		# across the Pyrenees that no Iberian unification needs.
		owns = location:narbonne
		# A contested crown cannot proclaim a restored one. The Visigothic
		# succession was elective and murderous enough that Frankish
		# writers called regicide the morbus Gothorum, so the kingdom is
		# not refounded during a civil war. This also protects the
		# Castilian Civil War disaster chain, which is tag-scoped to CAS
		# and would be stranded by a tag switch mid-arc.
		in_civil_war = no
		has_any_active_disaster = no
	}

	name = GOT
	flag = GOT
	adjective = GOT_ADJ
	tag = GOT
	color = map_GOT

	regions = {
		iberia_region
	}
	provinces = {
		narbonnais_province
		nimois_province
		razes_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_got_formed

		}
		# Imperial rank follows the Leonese imperial title rather than the
		# Gothic royal one: Sancho the Great minted Imperator totius
		# Hispaniae, Alfonso VI and Alfonso the Battler claimed it, and
		# Alfonso VII was crowned to it at Leon in 1135 on the doctrine
		# that the holder of Leon was the direct heir of the Gothic kings.
		# The imperial title IS the Gothic-succession claim in its
		# institutional form.
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		# The kingdom being restored is the kingdom of Toledo, so the court
		# returns to Toledo. Guaranteed available: allow requires it.
		set_capital = location:toledo
		add_stability = stability_mild_bonus
		trigger_event_non_silently = nd_got.1
	}
}
