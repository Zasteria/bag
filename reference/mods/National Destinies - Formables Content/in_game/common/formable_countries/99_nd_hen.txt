# Level 3, not vanilla's 2, same reachability reasoning as TAB (Tabor):
# `level` gates on strictly-higher rank ("Players can only form countries
# of higher level", formable_countries readme), and the plausible Brythonic
# formers, Wales and Brittany, sit at duchy rank (level 2), so at level 2
# none of them could ever press the button (Workshop report, 2026-05-26:
# player had to edit the mod to form it as Wales). County-rank minors keep
# access; the pan-Brythonic Old North revival is a kingdom-scale project
# anyway. REPLACE because INJECT cannot change vanilla fields.
REPLACE:HEN_f = {
	level = 3
	required_locations_fraction = 0.5
	rule = fantasy

	potential = {
		culture.language = language:brythonic_language
	}

	allow = {
	}

	name = HEN
	flag = HEN
	adjective = HEN_ADJ
	tag = HEN
	color = map_HEN

	areas = {
		northumbria_area
		scottish_lowlands_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_hen_formed
		}
		trigger_event_non_silently = nd_hen.1
		add_stability = stability_mild_bonus
		if = {
			limit = {
				country_rank_level < 3
			}
			set_country_rank_effect = {
				rank = country_rank:rank_kingdom
			}
		}
	}
}
