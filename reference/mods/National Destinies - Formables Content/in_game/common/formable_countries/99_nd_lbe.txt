LBE_f = {

	level = 2
	required_locations_fraction = 0.65
	rule = historical

	potential = {
		culture.language = language:luba_language
	}

	allow = {
		# baluba, not lake_upemba. The Upemba Depression is the Luba
		# cradle, but both of its lake tiles are water, so `owns =
		# location:lake_upemba` was false forever and no country could
		# form the Luba Empire at all. Baluba is the one land location
		# on the map named for the Luba themselves, it sits inside
		# east_lunda_area (already required geography), and the tag's
		# own bureaucracy file uses it as the heartland seat, so the
		# formable and its content now agree on one capital.
		owns = location:baluba
	}

	name = LBE
	flag = LBE
	adjective = LBE_ADJ
	tag = LBE
	color = map_LBE

	areas = {
		east_lunda_area
		kasongo_area
		moshiko_area
	}
	form_effect = {
		# Formation markers live HERE, not only in the event's options.
		# A country can be formed by script (vanilla forms JAP from the
		# sengoku situation), and then the popup may never be answered,
		# leaving every advance gated on this variable unreachable.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_lbe_formed
		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		trigger_event_non_silently = nd_lbe.1
		add_stability = stability_mild_bonus
	}
}
