# Level 1 by chain design (2026-08-10, workshop feedback): the engine
# gates formation on the forming tag's own formable level, strictly
# higher required, and at level 2 the Hospitallers could never form
# Jerusalem (also 2 at the time). At 1 they choose a second act, KOJ (3)
# or LAT (3), with OUT (4) above both. A two-island order state is
# honestly tier 1 anyway.
HOS_f = {

	level = 1
	required_locations_fraction = 0.9
	rule = plausible

	capital_required = no

	potential = {
		religion = religion:catholic
		OR = {
			tag = KNI
			has_reform = government_reform:military_order_reform
			tag = CYP
			tag = NAP
			tag = SIC
			tag = ARG
		}
	}

	allow = {
		OR = {
			owns = location:rodos
			owns = location:malta
		}
	}

	name = HOS
	flag = HOS
	adjective = HOS_ADJ
	tag = HOS
	color = map_HOS

	provinces = {
		rodos_province
		cyprus_province
	}
	locations = {
		malta
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_hos_formed

		}
		trigger_event_non_silently = nd_hos.1
		add_stability = stability_mild_bonus
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		# HOS is by definition a sovereign military-religious order. Grant
		# the clergy_military_orders privilege (unlocks military_order_reform
		# in the Government UI for formal activation) and apply the
		# permanent militant_brotherhood modifier so the institutional
		# benefits apply immediately without forcing a reform-slot swap.
		grant_estate_privilege = estate_privilege:clergy_military_orders
		add_country_modifier = {
			modifier = nd_hos_militant_brotherhood
			years = -1
		}
	}
}
