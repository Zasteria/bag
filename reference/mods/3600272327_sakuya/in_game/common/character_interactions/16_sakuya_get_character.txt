sakuya_select_chosen_one = {
	message = yes
	is_consort_action = no
	on_own_nation = no
	
	potential = {
	}
	
	ai_tick = never
	
	allow = {
	}
	
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "sakuya_choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
			NOT = { has_variable = sakuya_the_chosen_one }
		}
		enabled = {
		}
	}
	
	effect = {
		scope:actor = {
			remove_variable = rch_marked_character
			rch_clear_dynasty_to_be_renamed = yes
		}
		every_country = {
			every_character = {
				limit = { has_variable = sakuya_the_chosen_one }
				remove_variable = sakuya_the_chosen_one
			}
		}
		scope:recipient = {
			set_variable = {
				name = sakuya_the_chosen_one
				value = yes
			}
		}
		scope:actor = {
			set_variable = {
				name = rch_marked_character
				value = scope:recipient
			}
			set_variable = {
				name = sakuya_character_mark_cache
				value = yes
			}
			rch_set_dynasty_to_be_renamed = yes
		}
	}
	
	ai_will_do = {
		add = {
			value = 0
		}
	}
}
