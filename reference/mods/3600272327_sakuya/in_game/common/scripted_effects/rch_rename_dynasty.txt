rch_test = {
	c:SCA = {
		ruler = {
			dynasty = {
				prev = {
					found_dynasty = sakuya_renameable_dynasty_1
					change_dynasty = dynasty:sakuya_renameable_dynasty_1
				}
				every_character_in_dynasty = {
					change_dynasty = dynasty:sakuya_renameable_dynasty_1
				}
			}
		}
	}
}

rch_rename_dynasty = {
	rch_prepare_dynasty_rename = yes
	if = {
		limit = {
			has_variable = rch_marked_character
			has_variable = rch_dynasty_to_be_renamed_selected
		}
		save_scope_as = rch_player_scope
		var:rch_marked_character = {
			dynasty = {
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 1 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 2 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 3 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 4 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 5 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 6 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 7 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 8 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 9 }
				rch_try_rename_dynasty = { player = scope:rch_player_scope num = 10 }
			}
		}
		remove_variable = rch_dynasty_renamed
		remove_variable = rch_marked_character
		rch_clear_dynasty_to_be_renamed = yes
	}
}

rch_try_rename_dynasty = {
	if = {
		limit = {
			$player$ = {
				NOT = { has_variable = rch_dynasty_renamed }
			}
			OR = {
				NOT = { has_global_variable = rch_using_renamable_dynasty_$num$ }
				this = dynasty:rch_renameable_dynasty_$num$
				this = dynasty:sakuya_renameable_dynasty_$num$
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = rch_using_renamable_dynasty_$num$ }
			}
			prev = { found_dynasty = sakuya_renameable_dynasty_$num$ }
			set_global_variable = rch_using_renamable_dynasty_$num$
		}
		every_character_in_dynasty = {
			change_dynasty = dynasty:sakuya_renameable_dynasty_$num$
		}
		$player$ = {
			set_variable = rch_dynasty_renamed
		}
	}
}

rch_clear_dynasty_to_be_renamed = {
	remove_variable = rch_renaming_dynasty_1
	remove_variable = rch_renaming_dynasty_2
	remove_variable = rch_renaming_dynasty_3
	remove_variable = rch_renaming_dynasty_4
	remove_variable = rch_renaming_dynasty_5
	remove_variable = rch_renaming_dynasty_6
	remove_variable = rch_renaming_dynasty_7
	remove_variable = rch_renaming_dynasty_8
	remove_variable = rch_renaming_dynasty_9
	remove_variable = rch_renaming_dynasty_10
}

rch_prepare_dynasty_rename = {
	remove_variable = rch_dynasty_to_be_renamed_selected
	rch_clear_dynasty_to_be_renamed = yes
	save_scope_as = rch_player_scope
	if = {
		limit = {
			has_variable = rch_marked_character
			var:rch_marked_character = {
				has_dynasty = yes
			}
		}
		var:rch_marked_character = {
			dynasty = {
				rch_set_dynasty_to_be_renamed_for_player = { player = scope:rch_player_scope }
			}
		}
	}
	else_if = {
		limit = {
			any_country = {
				any_character = {
					has_variable = sakuya_the_chosen_one
					has_dynasty = yes
				}
			}
		}
		every_country = {
			every_character = {
				limit = {
					has_variable = sakuya_the_chosen_one
					has_dynasty = yes
				}
				save_scope_as = rch_marked_dynasty_character
				scope:rch_player_scope = {
					set_variable = {
						name = rch_marked_character
						value = scope:rch_marked_dynasty_character
					}
				}
				dynasty = {
					rch_set_dynasty_to_be_renamed_for_player = { player = scope:rch_player_scope }
				}
			}
		}
	}
}

rch_set_dynasty_to_be_renamed_for_player = {
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  1 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  2 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  3 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  4 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  5 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  6 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  7 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  8 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num =  9 }
	rch_try_set_dynasty_to_be_renamed_for_player = { player = $player$ num = 10 }
}

rch_try_set_dynasty_to_be_renamed_for_player = {
	if = {
		limit = {
			$player$ = {
				NOT = { has_variable = rch_dynasty_to_be_renamed_selected }
			}
			OR = {
				NOT = { has_global_variable = rch_using_renamable_dynasty_$num$ }
				this = dynasty:rch_renameable_dynasty_$num$
				this = dynasty:sakuya_renameable_dynasty_$num$
			}
		}
		$player$ = {
			set_variable = rch_renaming_dynasty_$num$
			set_variable = rch_dynasty_to_be_renamed_selected
		}
	}
}

	scope:recipient = {
		if = {
			limit = {
				scope:actor = {
					NOT = { has_variable = rch_dynasty_to_be_renamed_selected}
				}
				OR = {
					NOT = { has_global_variable = rch_using_renamable_dynasty_$num$ }
					dynasty = dynasty:rch_renameable_dynasty_$num$
					dynasty = dynasty:sakuya_renameable_dynasty_$num$
				}
			}
			scope:actor = {
				set_variable = rch_renaming_dynasty_$num$
				set_variable = rch_dynasty_to_be_renamed_selected
			}
		}
	}
}
