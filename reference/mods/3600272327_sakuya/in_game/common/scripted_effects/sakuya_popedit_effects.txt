sakuya_loc_popedit_assimilate_marked_effect = {
	save_scope_as = player_country
	if = {
		limit = { has_variable = sakuya_loc_popedit_assim_source_target }
		random_country = {
			limit = { has_variable = sakuya_get_TAG_choosen }
			save_scope_as = sakuya_loc_popedit_target_country
		}
	}
	if = {
		limit = { has_variable = sakuya_loc_popedit_assim_source_selector }
		every_culture = {
			limit = { has_variable = sakuya_selected_culture }
			save_scope_as = sakuya_loc_popedit_selected_culture
		}
	}
	every_owned_location = {
		limit = { has_variable = sakuya_marked }
		sakuya_loc_popedit_assimilate_location_effect = { fraction = $fraction$ }
	}
}

sakuya_loc_popedit_assimilate_location_effect = {
	every_pop = {
		limit = {
			OR = {
				scope:player_country = {
					NOT = { has_variable = sakuya_loc_popedit_class_laborers }
					NOT = { has_variable = sakuya_loc_popedit_class_burghers }
					NOT = { has_variable = sakuya_loc_popedit_class_nobles }
					NOT = { has_variable = sakuya_loc_popedit_class_clergy }
					NOT = { has_variable = sakuya_loc_popedit_class_soldiers }
					NOT = { has_variable = sakuya_loc_popedit_class_peasants }
					NOT = { has_variable = sakuya_loc_popedit_class_tribesmen }
					NOT = { has_variable = sakuya_loc_popedit_class_slaves }
				}
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_laborers } pop_type = pop_type:laborers }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_burghers } pop_type = pop_type:burghers }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_nobles } pop_type = pop_type:nobles }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_clergy } pop_type = pop_type:clergy }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_soldiers } pop_type = pop_type:soldiers }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_peasants } pop_type = pop_type:peasants }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_tribesmen } pop_type = pop_type:tribesmen }
				AND = { scope:player_country = { has_variable = sakuya_loc_popedit_class_slaves } pop_type = pop_type:slaves }
			}
		}
		if = {
			limit = {
				scope:player_country = {
					NOT = { has_variable = sakuya_loc_popedit_assim_source_target }
					NOT = { has_variable = sakuya_loc_popedit_assim_source_selector }
				}
			}
			split_pop = {
				fraction = $fraction$
				culture = scope:player_country.culture
			}
		}
		if = {
			limit = {
				scope:player_country = { has_variable = sakuya_loc_popedit_assim_source_target }
			}
			split_pop = {
				fraction = $fraction$
				culture = scope:sakuya_loc_popedit_target_country.culture
			}
		}
		if = {
			limit = {
				scope:player_country = { has_variable = sakuya_loc_popedit_assim_source_selector }
			}
			split_pop = {
				fraction = $fraction$
				culture = scope:sakuya_loc_popedit_selected_culture
			}
		}
	}
}
