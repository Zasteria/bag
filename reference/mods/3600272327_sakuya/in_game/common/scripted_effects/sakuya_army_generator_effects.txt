# Army generator effects. The counting loop and port fallback follow the original QA debug events.

sakuya_army_generator_spawn_unit_type = {
    set_variable = { name = sakuya_army_generator_counter value = 0 }
    change_variable = {
        name = sakuya_army_generator_counter
        add = var:sakuya_army_generator_amount
    }
    while = {
        limit = { var:sakuya_army_generator_counter > 0 }
        change_variable = {
            name = sakuya_army_generator_counter
            subtract = 1
        }
        $location$ = {
            create_sub_unit = $type$
        }
    }
    remove_variable = sakuya_army_generator_counter
}

sakuya_army_generator_spawn_owned_unit_type = {
    set_variable = { name = sakuya_army_generator_counter value = 0 }
    change_variable = {
        name = sakuya_army_generator_counter
        add = var:sakuya_army_generator_amount
    }
    while = {
        limit = { var:sakuya_army_generator_counter > 0 }
        change_variable = {
            name = sakuya_army_generator_counter
            subtract = 1
        }
        $location$ = {
            create_sub_unit_with_owner = {
                type = $type$
                owner = $owner$
                origin = $location$
            }
        }
    }
    remove_variable = sakuya_army_generator_counter
}

sakuya_army_generator_spawn_land_type = {
    if = {
        limit = {
            has_variable = sakuya_army_generator_spawn_target_mode
        }
        random_country = {
            limit = {
                has_variable = sakuya_get_TAG_choosen
            }
            save_scope_as = sakuya_army_generator_spawn_owner
            capital = {
                save_scope_as = sakuya_army_generator_spawn_location
            }
        }
        sakuya_army_generator_spawn_owned_unit_type = {
            type = $owned_type$
            owner = scope:sakuya_army_generator_spawn_owner
            location = scope:sakuya_army_generator_spawn_location
        }
    }
    else_if = {
        limit = {
            any_owned_location = {
                has_variable = sakuya_marked
            }
        }
        random_owned_location = {
            limit = {
                has_variable = sakuya_marked
            }
            save_scope_as = sakuya_army_generator_spawn_location
        }
    }
    else = {
        capital = {
            save_scope_as = sakuya_army_generator_spawn_location
        }
    }
    sakuya_army_generator_spawn_unit_type = {
        type = $type$
        location = scope:sakuya_army_generator_spawn_location
    }
}

sakuya_army_generator_spawn_navy_type = {
    if = {
        limit = {
            has_variable = sakuya_army_generator_spawn_target_mode
        }
        random_country = {
            limit = {
                has_variable = sakuya_get_TAG_choosen
            }
            save_scope_as = sakuya_army_generator_spawn_owner
            if = {
                limit = {
                    capital = {
                        is_port = yes
                    }
                }
                capital = {
                    save_scope_as = sakuya_army_generator_spawn_location
                }
            }
            else = {
                random_port_in_country = {
                    save_scope_as = sakuya_army_generator_spawn_location
                }
            }
        }
        sakuya_army_generator_spawn_owned_unit_type = {
            type = $owned_type$
            owner = scope:sakuya_army_generator_spawn_owner
            location = scope:sakuya_army_generator_spawn_location
        }
    }
    else_if = {
        limit = {
            any_owned_location = {
                has_variable = sakuya_marked
                is_port = yes
            }
        }
        random_owned_location = {
            limit = {
                has_variable = sakuya_marked
                is_port = yes
            }
            save_scope_as = sakuya_army_generator_spawn_location
        }
    }
    else_if = {
        limit = {
            capital = {
                is_port = yes
            }
        }
        capital = {
            save_scope_as = sakuya_army_generator_spawn_location
        }
    }
    else = {
        random_port_in_country = {
            save_scope_as = sakuya_army_generator_spawn_location
        }
    }
    sakuya_army_generator_spawn_unit_type = {
        type = $type$
        location = scope:sakuya_army_generator_spawn_location
    }
}
