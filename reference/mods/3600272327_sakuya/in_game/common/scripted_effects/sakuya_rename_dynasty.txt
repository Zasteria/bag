sakuya_reset_dynasty_rename_state = {
	remove_variable = sakuya_dynasty_renamed
	remove_variable = sakuya_dynasty_to_be_renamed_selected
	sakuya_clear_dynasty_rename_slots = yes
}

sakuya_rename_dynasty = {
	sakuya_prepare_dynasty_rename = yes
	if = {
		limit = {
			has_variable = rch_marked_character
			has_variable = sakuya_dynasty_to_be_renamed_selected
		}
		save_scope_as = sakuya_player_scope
		var:rch_marked_character = {
			dynasty = {
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 1 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 2 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 3 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 4 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 5 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 6 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 7 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 8 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 9 }
				sakuya_try_rename_dynasty = { player = scope:sakuya_player_scope num = 10 }
			}
		}
		sakuya_reset_dynasty_rename_state = yes
	}
}

sakuya_try_rename_dynasty = {
	if = {
		limit = {
			$player$ = {
				NOT = { has_variable = sakuya_dynasty_renamed }
			}
			OR = {
				NOT = { has_global_variable = sakuya_using_renameable_dynasty_$num$ }
				this = dynasty:sakuya_renameable_dynasty_$num$
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = sakuya_using_renameable_dynasty_$num$ }
			}
			prev = {
				found_dynasty = sakuya_renameable_dynasty_$num$
			}
			set_global_variable = sakuya_using_renameable_dynasty_$num$
		}
		every_character_in_dynasty = {
			change_dynasty = dynasty:sakuya_renameable_dynasty_$num$
		}
		$player$ = {
			set_variable = sakuya_dynasty_renamed
		}
	}
}

sakuya_prepare_dynasty_rename = {
	sakuya_reset_dynasty_rename_state = yes
	save_scope_as = sakuya_player_scope
	if = {
		limit = {
			has_variable = rch_marked_character
			var:rch_marked_character = {
				has_dynasty = yes
			}
		}
		var:rch_marked_character = {
			dynasty = {
				sakuya_set_dynasty_to_be_renamed = { player = scope:sakuya_player_scope }
			}
		}
	}
	else_if = {
		limit = {
			any_country = {
				any_character = {
					has_variable = sakuya_the_chosen_one
					has_dynasty = yes
				}
			}
		}
		every_country = {
			every_character = {
				limit = {
					has_variable = sakuya_the_chosen_one
					has_dynasty = yes
				}
				save_scope_as = sakuya_marked_dynasty_character
				scope:sakuya_player_scope = {
					set_variable = {
						name = rch_marked_character
						value = scope:sakuya_marked_dynasty_character
					}
				}
				dynasty = {
					sakuya_set_dynasty_to_be_renamed = { player = scope:sakuya_player_scope }
				}
			}
		}
	}
}

sakuya_clear_dynasty_rename_slots = {
	remove_variable = sakuya_renaming_dynasty_1
	remove_variable = sakuya_renaming_dynasty_2
	remove_variable = sakuya_renaming_dynasty_3
	remove_variable = sakuya_renaming_dynasty_4
	remove_variable = sakuya_renaming_dynasty_5
	remove_variable = sakuya_renaming_dynasty_6
	remove_variable = sakuya_renaming_dynasty_7
	remove_variable = sakuya_renaming_dynasty_8
	remove_variable = sakuya_renaming_dynasty_9
	remove_variable = sakuya_renaming_dynasty_10
}

sakuya_set_dynasty_to_be_renamed = {
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 1 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 2 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 3 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 4 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 5 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 6 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 7 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 8 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 9 }
	sakuya_try_set_dynasty_to_be_renamed = { player = $player$ num = 10 }
}

sakuya_try_set_dynasty_to_be_renamed = {
	if = {
		limit = {
			$player$ = {
				NOT = { has_variable = sakuya_dynasty_to_be_renamed_selected }
			}
			OR = {
				NOT = { has_global_variable = sakuya_using_renameable_dynasty_$num$ }
				this = dynasty:sakuya_renameable_dynasty_$num$
			}
		}
		$player$ = {
			set_variable = sakuya_renaming_dynasty_$num$
			set_variable = sakuya_dynasty_to_be_renamed_selected
		}
	}
}
