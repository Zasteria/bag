sakuya_mark_country = { 
	type = diplomacy
	relation_type = oneway
	category = {
		offer = CATEGORY_ACCESS_ACTIONS
		request = CATEGORY_ACCESS_ACTIONS
		cancellation = CATEGORY_ACCESS_ACTIONS
		break = CATEGORY_ACCESS_ACTIONS
	}

    visible = {
        always = yes
    }

	offer_enabled = {
		NOT = {
			any_country = {
				has_variable = sakuya_get_TAG_choosen
			}
		}
	}

	request_visible = { always = no }

	request_enabled = {
		always = no
	}

	offer_effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				set_variable = {
					name = sakuya_get_TAG_choosen
					value = yes
					days = 90
				}
			}
		}
	}

	cancel_effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				remove_variable = sakuya_get_TAG_choosen
			}
		}
	}

	break_effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				remove_variable = sakuya_get_TAG_choosen
			}
		}
	}

	wants_to_receive_diplo_chance = {
		base = 100
	}

	wants_to_give_diplo_chance = {
		base = 0
	}
}
