# 咲夜的标记国家状态检测器
# 用于GUI中显示当前标记了多少个国家
sakuya_mark_country_status = {
	type = country
	
	# 情况1：没有任何国家被标记
	text = {
		trigger = {
			NOT = {
				any_country = {
					has_variable = sakuya_get_TAG_choosen
				}
			}
		}
		localization_key = "sakuya_mark_none"
	}
	
	# 情况2：正好标记了一个国家（正常情况）
	text = {
		trigger = {
			any_country = {
				has_variable = sakuya_get_TAG_choosen
			}
			NOT = {
				any_country = {
					has_variable = sakuya_get_TAG_choosen
					any_country = {
						NOT = { this = prev }
						has_variable = sakuya_get_TAG_choosen
					}
				}
			}
		}
		localization_key = "sakuya_mark_one"
	}
	
	# 情况3：标记了多个国家（异常情况）
	text = {
		trigger = {
			any_country = {
				has_variable = sakuya_get_TAG_choosen
				any_country = {
					NOT = { this = prev }
					has_variable = sakuya_get_TAG_choosen
				}
			}
		}
		localization_key = "sakuya_mark_multiple"
	}
	
# 默认值（以防万一）
text = {
	localization_key = "sakuya_mark_none"
	fallback = yes
}
}
