namespace = sakuya_truce

sakuya_truce.1 = {
	type = country_event
	title = sakuya_truce.1.title
	desc = sakuya_truce.1.desc
	outcome = neutral

	trigger = {
		any_country = {
			has_variable = sakuya_get_TAG_choosen
		}
	}

	immediate = {
		random_country = {
			limit = {
				has_variable = sakuya_get_TAG_choosen
			}
			save_scope_as = marked_country
		}
	}

	option = {
		name = sakuya_truce.1.a
		if = {
			limit = {
				has_truce_with = scope:marked_country
			}
			remove_truce_with = scope:marked_country
		}
		scope:marked_country = {
			if = {
				limit = {
					has_truce_with = root
				}
				remove_truce_with = root
			}
		}
	}

	option = {
		name = sakuya_truce.1.b
		if = {
			limit = {
				has_truce_with = scope:marked_country
			}
			remove_truce_with = scope:marked_country
		}
		scope:marked_country = {
			if = {
				limit = {
					has_truce_with = root
				}
				remove_truce_with = root
			}
		}
		add_truce_with = {
			target = scope:marked_country
			mutual = yes
			months = 1
		}
	}

	option = {
		name = sakuya_truce.1.c
		if = {
			limit = {
				has_truce_with = scope:marked_country
			}
			remove_truce_with = scope:marked_country
		}
		scope:marked_country = {
			if = {
				limit = {
					has_truce_with = root
				}
				remove_truce_with = root
			}
		}
		add_truce_with = {
			target = scope:marked_country
			mutual = yes
			years = 5
		}
	}

	option = {
		name = sakuya_truce.1.d
		if = {
			limit = {
				has_truce_with = scope:marked_country
			}
			remove_truce_with = scope:marked_country
		}
		scope:marked_country = {
			if = {
				limit = {
					has_truce_with = root
				}
				remove_truce_with = root
			}
		}
		add_truce_with = {
			target = scope:marked_country
			mutual = yes
			years = 10
		}
	}

	option = {
		name = sakuya_truce.1.e
		if = {
			limit = {
				has_truce_with = scope:marked_country
			}
			remove_truce_with = scope:marked_country
		}
		scope:marked_country = {
			if = {
				limit = {
					has_truce_with = root
				}
				remove_truce_with = root
			}
		}
		add_truce_with = {
			target = scope:marked_country
			mutual = yes
			years = 100
		}
	}

	option = {
		name = sakuya_truce.1.f
	}
}
