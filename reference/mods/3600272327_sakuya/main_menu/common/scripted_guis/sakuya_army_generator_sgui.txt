sakuya_army_generator_spawn_selected = {
    scope = country
    is_shown = { always = yes }
    is_valid = {
        has_variable = sakuya_army_generator_amount
        var:sakuya_army_generator_amount > 0
        has_variable = sakuya_army_generator_selected_index
        has_variable = sakuya_army_generator_selected_domain
        OR = {
            var:sakuya_army_generator_selected_domain = 1
            AND = {
                var:sakuya_army_generator_selected_domain = 2
                num_of_ports > 0
            }
        }
    }
    effect = {
        remove_variable = sakuya_army_generator_spawn_target_mode
        sakuya_army_generator_spawn_selected_type = yes
    }
}

sakuya_army_generator_spawn_selected_target = {
    scope = country
    is_shown = { always = yes }
    is_valid = {
        has_variable = sakuya_army_generator_amount
        var:sakuya_army_generator_amount > 0
        has_variable = sakuya_army_generator_selected_index
        has_variable = sakuya_army_generator_selected_domain
        any_country = {
            has_variable = sakuya_get_TAG_choosen
            OR = {
                root.var:sakuya_army_generator_selected_domain = 1
                AND = {
                    root.var:sakuya_army_generator_selected_domain = 2
                    num_of_ports > 0
                }
            }
        }
    }
    effect = {
        set_variable = { name = sakuya_army_generator_spawn_target_mode value = yes }
        sakuya_army_generator_spawn_selected_type = yes
        remove_variable = sakuya_army_generator_spawn_target_mode
    }
}
