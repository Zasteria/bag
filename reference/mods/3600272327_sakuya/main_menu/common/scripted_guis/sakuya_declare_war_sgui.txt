# sakuya declare war window scripted GUI actions

sakuya_declare_war_player_to_target_cb_none = {
    scope = country
    is_shown = { always = yes }
    is_valid = {
        any_country = { has_variable = sakuya_get_TAG_choosen }
    }
    effect = {
        random_country = {
            limit = { has_variable = sakuya_get_TAG_choosen }
            save_scope_as = marked_country
        }
        declare_war_with_cb = {
            target = scope:marked_country
            type = casus_belli:cb_none
        }
    }
}

        scope:marked_country = {
            declare_war_with_cb = {
                target = root
                type = casus_belli:cb_following_through_on_threat
                target_province = scope:sakuya_declare_war_target_location.province
            }
        }
    }
}

    is_shown = { always = yes }
    is_valid = {
        any_country = { has_variable = sakuya_get_TAG_choosen }
    }
    effect = {
        random_country = {
            limit = { has_variable = sakuya_get_TAG_choosen }
            save_scope_as = marked_country
        }
        if = {
            limit = { any_owned_location = { has_variable = sakuya_marked } }
            random_owned_location = {
                limit = { has_variable = sakuya_marked }
                save_scope_as = sakuya_declare_war_target_location
            }

        any_country = { has_variable = sakuya_get_TAG_choosen }
    }
    effect = {
        random_country = {
            limit = { has_variable = sakuya_get_TAG_choosen }
            save_scope_as = marked_country
        }
        if = {
            limit = { any_owned_location = { has_variable = sakuya_marked region = region:italy_region } }
            random_owned_location = {
                limit = { has_variable = sakuya_marked region = region:italy_region }
                save_scope_as = sakuya_declare_war_target_location
            }
        }
        else = {
            if = {
                limit = { any_owned_location = { region = region:italy_region } }
                random_owned_location = {
                    limit = { region = region:italy_region }
                    save_scope_as = sakuya_declare_war_target_location
                }
            }
            else = {
                capital = { save_scope_as = sakuya_declare_war_target_location }
            }
        }
        scope:marked_country = {
            declare_war_with_cb = {
                target = root
                type = casus_belli:cb_secure_italy
                target_province = scope:sakuya_declare_war_target_location.province
            }
        }
    }
}
