# AI nation formation (2026-07-31). Community-verified vanilla gap: the AI
# essentially never forms nations (Russia never, Austria-Hungary never; no
# formable carries any AI steering, 0 occurrences of ai_will_do in vanilla
# formable_countries). Off by default: turning it on changes world
# difficulty, which is the subscriber's call, not ours.
#   opportunistic: AI countries already meeting a major formable's
#                    requirements take it (pulse in 99_nd_on_actions.txt).
#   ambitious    : additionally steers AI conquest toward the territory
#                    still needed (ai_scripted_expansion_score entries).
nd_ai_formables_rule = {
	default = nd_ai_formables_off

	nd_ai_formables_off = {
		flag = flavour_rule
	}

	nd_ai_formables_opportunistic = {
		flag = nd_ai_formables_opportunistic
		flag = flavour_rule
	}

	nd_ai_formables_ambitious = {
		flag = nd_ai_formables_ambitious
		flag = flavour_rule
	}
}

# Reaction of the Reich (2026-08-14). Enabled by default, because the
# Empire's answer is the point of breaking with the Reich: it is meant to
# cost something. Off = the break still sweeps the realm out of the Empire
# and still grants imperial rank, but no situation follows; both Danubian
# disasters keep running, because they gate on nd_dnm_formed rather than
# on the situation.
#
# THIS IS NOT THE AI SWITCH (2026-08-25 correction). It shipped in 1.3.6 as
# the answer to Sharkseant's Workshop request for a way to keep the AI from
# founding the Monarchy, on the claim that "formables carry no ai_will_do,
# so the AI cannot be made reluctant". The claim is beside the point: no AI
# presses a formable button by itself (vanilla AI forms no nations; only
# our nd_ai_formables_rule pulse does, and DNM was not in it until
# 2026-08-25). The road by which an AI actually became the Monarchy was our
# own Succession of the Danube situation, whose ending force-forms an AI
# holder (nd_dnb_proclaim_or_offer). That road is what
# nd_danube_succession_rule below switches off; this rule only decides
# whether the Empire answers a Monarchy that already exists. Kept because it
# is shipped and answers a real, different wish.
#
# Read as `has_game_rule = nd_reich_reaction_enabled` through the
# nd_reich_reaction_is_enabled scripted trigger, matching the shape of
# nd_bureaucracies_rule above.
nd_reich_reaction_rule = {
	default = nd_reich_reaction_enabled

	nd_reich_reaction_enabled = {
		flag = nd_reich_reaction_enabled
		flag = flavour_rule
	}

	nd_reich_reaction_disabled = {
		flag = flavour_rule
	}
}

# Succession of the Danube (2026-08-25, the actual answer to Sharkseant's
# 2026-08-14 Workshop request; the 2026-08-19 report that AI Bohemia still
# founds the Monarchy under 1.3.6 is this road running as written). Enabled
# by default: the situation is the mod's primary road into the Danubian
# Monarchy, and the subscriber who wants a Danube the AI never unites opts
# out.
#
# Disabled = the situation never starts: the world-spark watcher
# (nd_dnb_spark_watch) and the pact interaction (nd_dnb_propose_the_pact,
# the player's own match) both read the rule, and those are the only two
# activation sites. Everything downstream is then unreachable by
# construction: no Congress, no election into fiefdoms, no claim-throne
# grants, no forced proclamation. The Monarchy stays formable at the
# button (marriage unions and conquest both satisfy its allow). An AI
# reaches that button only through nd_ai_formables_rule above, whose
# pulse lists DNM since 2026-08-25 (user ruling: a crown that unites the
# three by marriage forms the Monarchy whenever AI Forms Nations is on);
# with that rule off, no AI Danubian Monarchy can arise under this
# setting.
#
# Read through the nd_danube_succession_is_enabled scripted trigger.
nd_danube_succession_rule = {
	default = nd_danube_succession_enabled

	nd_danube_succession_enabled = {
		flag = nd_danube_succession_enabled
		flag = flavour_rule
	}

	nd_danube_succession_disabled = {
		flag = flavour_rule
	}
}
