# ==========================================================================
# SUE FOR IMPERIAL RECONCILIATION (Item 4, dnm_proposal): the mod's
# first country_interaction for any tag. The Reaction of the Reich's
# fourth exit: a diplomatic overture to the sitting Emperor, available at
# will while the pariah status is live, instead of a war or a 50-year
# wait. Modeled on country_interactions/hre.txt (request_elector_status's
# accept-script shape and its Emperor-targeting select_trigger idiom).
#
# The engine's block_when_at_war default (yes, per readme) supplies the
# "not while at war with the Emperor" condition natively.
#
# Success sets nd_dnm_reich_reconciled, which the situation's can_end
# reads as its fourth exit and nd_dnm.29 reads for its differentiated
# reward (and, for a Bohemia-origin former-elector union, the vacant-seat
# bestowal). Rejection is not free: the court's public mockery costs
# prestige, and the 10-year cooldown runs from the attempt either way.
# ==========================================================================
nd_dnm_sue_for_imperial_reconciliation = {
	type = diplomacy

	# News-feed message types (WE_PERFORM_X_ACTION etc.) live in the
	# non-additive gui/messagetypes.txt; suppressing messages instead of
	# overriding that whole vanilla file (the mod's own generic actions
	# already use the same suppression). The interaction's own UI, the
	# PROPOSE_/INCOMING_OFFER_ loc, and the effect text all still render.
	show_message = no
	show_message_to_target = no

	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = {
			tag = DNM
			has_country_modifier = nd_dnm_reich_pariah
			# Idempotency: once reconciled, the overture is moot.
			NOT = { has_variable = nd_dnm_reich_reconciled }
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		# There is exactly one Emperor at a time, so the selection list is
		# trivially one entry (the request_elector_status idiom).
		visible = {
			is_emperor = yes
		}
	}

	cooldown = {
		type = nd_dnm_sue_for_imperial_reconciliation
		years = 10
	}

	# The tribute: two years of trade-and-tax income, floored at 200 gold,
	# paid to the Emperor, the same income-scaled add_gold shape the
	# DNM crisis content already uses (no price: definition needed).
	effect = {
		scope:actor = {
			add_gold = {
				value = monthly_income_trade_and_tax
				multiply = -24
				max = -200
			}
		}
		scope:recipient = {
			add_gold = {
				value = scope:actor.monthly_income_trade_and_tax
				multiply = 24
				min = 200
			}
		}
		scope:actor = {
			custom_tooltip = {
				text = nd_dnm_reconciliation_success_tt
				hidden_effect = { set_variable = nd_dnm_reich_reconciled }
			}
		}
	}

	reject_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
			custom_tooltip = nd_dnm_reconciliation_rejected_tt
		}
	}

	# The Emperor's answer. Reads, in order: the Reich's own live cohesion
	# (a weak Emperor needs friends, not more enemies), his grudge, his
	# distraction, the upstart's respectability, and, from Item 1, the
	# origin-dependent grievance: a founding vote's defection is not
	# easily forgiven, and Bohemia's Golden Bull seat (fact #4) is the
	# single hardest grievance to paper over.
	accept = {
		add = {
			desc = "ND_DNM_ACCEPT_BASE"
			value = -40
		}
		subtract = {
			desc = "ND_DNM_ACCEPT_IMPERIAL_AUTHORITY"
			value = international_organization:hre.var:imperial_authority
			multiply = 0.5
		}
		if = {
			limit = { "scope:recipient.antagonism(scope:actor)" > 0 }
			subtract = {
				desc = "DIPLOREASON_ANTAGONISM"
				value = "scope:recipient.antagonism(scope:actor)"
			}
		}
		if = {
			# `at_war`, not `is_at_war`: the latter is not a trigger
			# (error.log 2026-07-31; trigger loc country_triggers.txt:788).
			limit = { scope:recipient = { at_war = yes } }
			add = {
				desc = "ND_DNM_ACCEPT_EMPEROR_DISTRACTED"
				value = 30
			}
		}
		add = {
			desc = "ND_DNM_ACCEPT_LEGITIMACY"
			value = scope:actor.legitimacy
			multiply = 0.5
		}
		add = {
			desc = "ND_DNM_ACCEPT_PRESTIGE"
			value = scope:actor.prestige
			multiply = 0.25
		}
		if = {
			limit = { scope:actor = { has_variable = nd_dnm_origin_was_elector } }
			subtract = {
				desc = "ND_DNM_ACCEPT_DEFECTED_ELECTOR"
				value = 30
			}
		}
		if = {
			limit = { scope:actor = { has_variable = nd_dnm_origin_boh } }
			subtract = {
				desc = "ND_DNM_ACCEPT_BOHEMIAN_SEAT"
				value = 30
			}
		}
	}

	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:actor = {
						gold >= {
							value = monthly_income_trade_and_tax
							multiply = 36
						}
					}
				}
				value = 50
			}
		}
	}
}

# ==========================================================================
# BREAK WITH THE REICH (2026-08-24, the Reich question). A Monarchy that
# chose to remain within the Empire at formation may leave later, at the
# same price the break option charges on day one: the imperial land goes
# out with the Emperor's claim on it, empire rank, the Reaction of the
# Reich. Declared to the Emperor; if we wear the crown ourselves it is
# declared to an elector, because we lay the crown down first (Francis II,
# 1806, in reverse: the dynasty keeps the Danube and gives up the Reich).
# Unilateral, no accept block (the rescind_elector_status shape).
#
# The vanilla leave button (hre.txt:363 can_leave_trigger) exists as well,
# gated on the Golden Bull and the year; the monthly pulse catches a
# departure by that road and applies the same consequences (nd_dnm.54).
# ==========================================================================
nd_dnm_break_with_reich = {
	type = diplomacy
	show_message = no
	show_message_to_target = no
	ai_tick = monthly
	ai_tick_frequency = 12
	potential = {
		scope:actor = {
			tag = DNM
			nd_dnm_is_in_reich = yes
		}
	}
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			OR = {
				AND = {
					scope:actor = { is_emperor = no }
					is_emperor = yes
				}
				AND = {
					scope:actor = { is_emperor = yes }
					international_organization:hre ?= {
						country_has_special_status = {
							type = special_status:elector
							country = root
						}
					}
				}
			}
		}
	}
	effect = {
		scope:actor = {
			nd_dnm_break_with_reich_effect = yes
		}
	}
	# The AI leaves only when the Reich has become a cage: it cannot
	# wear the crown (wrong faith for the electors) and does not hold it.
	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:actor = {
						is_emperor = no
						NOT = { hre_country_fulfills_imperial_religion_requirement = { type = voting } }
					}
				}
				value = 25
			}
		}
	}
}

# ==========================================================================
# LAY THE SANCTION BEFORE THE DIET (2026-08-24, the Reich question). The
# in-Reich road to the welding. The Pragmatic Sanction upheld inside the
# Empire (nd_dnm.21, option c) leaves the constituent crowns standing until
# the Diet ratifies it, as the Diet did in 1732 and at a price. A member
# petitions the Emperor: the Emperor weighs his own grip on the Reich (a
# strong Emperor will not let one member swallow three crowns), his
# opinion of us, the faith, and our standing. If we wear the crown
# ourselves the petition goes to an elector and the price is imperial
# authority instead. Success fires nd_dnm.53, the weld.
# ==========================================================================
nd_dnm_seek_diet_ratification = {
	type = diplomacy
	show_message = no
	show_message_to_target = no
	ai_tick = monthly
	ai_tick_frequency = 12
	potential = {
		scope:actor = {
			tag = DNM
			nd_dnm_is_in_reich = yes
			has_variable = nd_dnm_sanction_awaits_diet
		}
	}
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			OR = {
				AND = {
					scope:actor = { is_emperor = no }
					is_emperor = yes
				}
				AND = {
					scope:actor = { is_emperor = yes }
					international_organization:hre ?= {
						country_has_special_status = {
							type = special_status:elector
							country = root
						}
					}
				}
			}
		}
	}
	cooldown = {
		type = nd_dnm_seek_diet_ratification
		years = 10
	}
	# The concessions: two years of trade-and-tax income, floored at 200
	# gold, paid to the court we petition (the reconciliation shape).
	effect = {
		scope:actor = {
			add_gold = {
				value = monthly_income_trade_and_tax
				multiply = -24
				max = -200
			}
		}
		scope:recipient = {
			add_gold = {
				value = scope:actor.monthly_income_trade_and_tax
				multiply = 24
				min = 200
			}
		}
		if = {
			limit = { scope:actor = { is_emperor = yes } }
			scope:actor = {
				change_imperial_authority = { value = imperial_authority_strong_penalty }
			}
		}
		else = {
			scope:recipient = {
				change_imperial_authority = { value = imperial_authority_weak_bonus }
			}
		}
		scope:actor = {
			custom_tooltip = {
				text = nd_dnm_ratification_success_tt
				trigger_event_non_silently = nd_dnm.53
			}
		}
	}
	reject_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
			custom_tooltip = nd_dnm_ratification_rejected_tt
		}
	}
	accept = {
		add = {
			desc = "ND_DNM_RATIFY_BASE"
			value = -20
		}
		add = {
			desc = "DIPLOREASON_OPINION"
			value = "scope:recipient.opinion(scope:actor)"
			multiply = 0.4
		}
		if = {
			limit = { scope:recipient = { is_emperor = yes } }
			subtract = {
				desc = "ND_DNM_RATIFY_IMPERIAL_AUTHORITY"
				value = international_organization:hre.var:imperial_authority
				multiply = 0.3
			}
		}
		if = {
			limit = { scope:actor = { religion = scope:recipient.religion } }
			add = {
				desc = "ND_DNM_RATIFY_SAME_FAITH"
				value = 20
			}
		}
		if = {
			limit = { scope:actor = { at_war = yes } }
			subtract = {
				desc = "ND_DNM_RATIFY_AT_WAR"
				value = 20
			}
		}
		add = {
			desc = "ND_DNM_ACCEPT_LEGITIMACY"
			value = scope:actor.legitimacy
			multiply = 0.25
		}
		add = {
			desc = "ND_DNM_ACCEPT_PRESTIGE"
			value = scope:actor.prestige
			multiply = 0.25
		}
	}
	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = {
					scope:actor = {
						gold >= {
							value = monthly_income_trade_and_tax
							multiply = 36
						}
					}
				}
				value = 50
			}
		}
	}
}
