# ==========================================================================
# DANUBIAN MONARCHY: the Reich question, as effects.
#
# THE PREMISE THAT WAS WRONG (2026-08-24). The formable used to eject the
# former from the Holy Roman Empire inside form_effect on the reading that
# "HRE members are engine-blocked from empire rank". That block
# (`block_from_change_to_empire_rank`, hre.txt:120) lives only in the rank's
# `allow` (country_ranks/00_default.txt:46-48), which governs the player's
# rank button; nothing in vanilla demotes a country that already holds the
# rank, and vanilla's own Germany formable grants empire rank by
# set_country_rank_effect to sitting members without leaving
# (00_formable_countries.txt:1564-1586). The exit was a design choice
# wearing the costume of an engine constraint, and it made the historical
# Habsburg road (stay, and wear the crown) impossible. Reported on the
# Workshop and Reddit as "forming the Monarchy leaves a void in the Reich".
#
# Now the Reich question is a CHOICE at formation (nd_dnm.52), and the two
# roads are these effects. Staying keeps kingdom rank: the one Empire
# inside the Reich is the Reich, and the Monarchy's dignity within it is
# the imperial crown itself, which it may win. Breaking takes empire rank,
# sweeps the imperial land out with the Emperor's cores on it, and wakes
# the Reaction of the Reich. A Monarchy that stayed can break later
# (country_interactions/99_nd_dnm.txt, nd_dnm_break_with_reich), and a
# departure by any vanilla road is caught by the monthly pulse; a break is
# one way.
# ==========================================================================

# ----- THE WELDING (shared by the Sanction, the Diet, and a late break) --
# A constituent crown that answers to us is absorbed outright: cores
# first, then annexation (vanilla idiom: flavor_wittelsbach_house_union).
# No Reich sweep in here: the break road sweeps afterwards, and inside the
# Reich the welded land stays imperial, which is the point of the Diet's
# ratification.
nd_dnm_weld_crowns_effect = {
	save_scope_as = nd_dnm_realm
	if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { nd_dnm_crown_answers_to = { REALM = scope:nd_dnm_realm } }
		}
		custom_tooltip = {
			text = nd_dnm_weld_hun_tt
			c:HUN = {
				every_core_location = { add_core = scope:nd_dnm_realm }
			}
			annex_country = { country = c:HUN }
		}
	}
	if = {
		limit = {
			country_exists = c:BOH
			c:BOH = { nd_dnm_crown_answers_to = { REALM = scope:nd_dnm_realm } }
		}
		custom_tooltip = {
			text = nd_dnm_weld_boh_tt
			c:BOH = {
				every_core_location = { add_core = scope:nd_dnm_realm }
			}
			annex_country = { country = c:BOH }
		}
	}
	if = {
		limit = {
			country_exists = c:HAB
			c:HAB = { nd_dnm_crown_answers_to = { REALM = scope:nd_dnm_realm } }
		}
		custom_tooltip = {
			text = nd_dnm_weld_hab_tt
			c:HAB = {
				every_core_location = { add_core = scope:nd_dnm_realm }
			}
			annex_country = { country = c:HAB }
		}
	}
}

# ----- THE SWEEP ----------------------------------------------------------
# Every imperial location we own leaves the Empire and the Emperor gets a
# core on it: vanilla's own secede path (events/hre.txt:81-89) does exactly
# this, and the core is what gives the imperial-ban casus belli something
# to collect. Location membership is per LOCATION, so this runs whatever
# our own membership was (a Hungarian former was never a member and still
# walks Austrian and Bohemian land out). Guarded on the Empire existing:
# it can be destroyed outright (events/hre.txt:166).
nd_dnm_reich_sweep_effect = {
	if = {
		limit = { exists = international_organization:hre }
		custom_tooltip = {
			text = nd_dnm_hre_sweep_tt
			every_owned_location = {
				limit = { is_owned_by_international_organization = international_organization:hre }
				if = {
					limit = {
						international_organization:hre = {
							international_organization_has_leader = yes
						}
					}
					add_core = international_organization:hre.leader_country
				}
				remove_from_international_organization = international_organization:hre
			}
		}
	}
}

# ----- THE BREAK ----------------------------------------------------------
# Runs from the formation event's break option, from the break
# interaction, and from the monthly pulse when a departure by a vanilla
# road is detected. Idempotent in effect: a second run finds nothing left
# to sweep, no membership to end, the rank already held, and a situation
# already active.
nd_dnm_break_with_reich_effect = {
	save_scope_as = nd_dnm_realm
	# An Emperor who breaks lays the crown down first (Francis II, 1806,
	# in reverse: the dynasty keeps the Danube and gives up the Reich).
	# The flag is captured BEFORE the membership ends, because whether
	# `is_emperor` survives the removal is not documented; the electors
	# are then told to choose again once we are gone.
	if = {
		limit = { is_emperor = yes }
		custom_tooltip = {
			text = nd_dnm_break_abdicate_tt
			add_prestige = prestige_severe_penalty
			ruler_or_heir_if_regent ?= {
				add_character_modifier = {
					modifier = abdicated_from_the_imperial_throne
					years = -1
					mode = add
				}
			}
			hidden_effect = { set_variable = nd_dnm_abdicating }
		}
	}
	nd_dnm_reich_sweep_effect = yes
	if = {
		limit = {
			exists = international_organization:hre
			is_member_of_international_organization = international_organization:hre
		}
		international_organization:hre = {
			remove_country_from_international_organization = scope:nd_dnm_realm
		}
	}
	if = {
		limit = { has_variable = nd_dnm_abdicating }
		hidden_effect = {
			remove_variable = nd_dnm_abdicating
			international_organization:hre ?= {
				international_organization_chooses_new_leader = this
			}
		}
	}
	# Empire rank. set_country_rank_effect is not bound by the rank's
	# allow (see the header), so this holds whatever the Catholic and
	# Reich blocks say.
	if = {
		limit = { country_rank_level < 4 }
		set_country_rank_effect = { rank = country_rank:rank_empire }
	}
	hidden_effect = {
		remove_variable = nd_dnm_in_reich
		set_variable = nd_dnm_reich_broken
		# The Endurance exit's clock starts at the break, not at
		# formation: a Monarchy that stayed forty years and then left
		# has not endured forty years of the Reich's hostility.
		set_variable = { name = nd_dnm_pariah_timer value = yes years = 50 }
		# CRISIS LIFECYCLE CONTRACT: every can_end clause of the Reaction
		# must be false at activation. A late break can arrive with the
		# exit markers of a previous era already stamped (a Sanction
		# resolved inside the Reich, a reconciliation from an earlier
		# incarnation). Clear them so the situation cannot end on the
		# day it starts. Nothing but the situation and its coda reads
		# these (checked 2026-08-24).
		remove_variable = nd_dnm_ps_resolved
		remove_variable = nd_dnm_reich_humbled
		remove_variable = nd_dnm_reich_reconciled
		remove_variable = nd_dnm_reich_endured
	}
	# The Crown of Saint Stephen's grievance was the Reich; it goes with it.
	if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { has_country_modifier = nd_dnm_crown_of_st_stephen }
		}
		c:HUN = { remove_country_modifier = nd_dnm_crown_of_st_stephen }
	}
	# A Sanction upheld inside the Reich and still awaiting the Diet
	# becomes our own law the moment we leave: there is no Diet left to
	# ask, and indivisibiliter ac inseparabiliter needs no ratification
	# but ours.
	if = {
		limit = { has_variable = nd_dnm_sanction_awaits_diet }
		custom_tooltip = {
			text = nd_dnm_sanction_own_law_tt
			nd_dnm_weld_crowns_effect = yes
			hidden_effect = {
				remove_variable = nd_dnm_sanction_awaits_diet
				remove_variable = nd_dnm_ps_upheld
			}
		}
	}
	# The Reich's answer, behind two gates: the lobby rule (a player who
	# does not want the Empire to hound the Danube turns it off and keeps
	# everything else), and a Reich that still exists to answer.
	if = {
		limit = {
			exists = international_organization:hre
			nd_reich_reaction_is_enabled = yes
			NOT = {
				situation:nd_dnm_reaction_of_the_reich ?= { situation_is_active = yes }
			}
		}
		activate_situation = situation:nd_dnm_reaction_of_the_reich
	}
}

# ----- THE STAY -----------------------------------------------------------
# Remain within the Reich (or, for a non-member origin, enter it). Kingdom
# rank only where the Empire's own rank rule allows it: Bohemia is the one
# kingdom the Golden Bull tolerates, so an Austrian former keeps the
# archducal (duchy-level) rank inside the Reich unless the lobby's rank
# rule lifts the block (nd_can_take_kingdom_rank reads what vanilla
# reads). That is historically exact: Austria never became a kingdom.
nd_dnm_stay_in_reich_effect = {
	save_scope_as = nd_dnm_realm
	if = {
		limit = {
			exists = international_organization:hre
			NOT = { is_member_of_international_organization = international_organization:hre }
		}
		international_organization:hre = {
			add_country_to_international_organization = scope:nd_dnm_realm
		}
	}
	if = {
		limit = {
			country_rank_level < 3
			nd_can_take_kingdom_rank = yes
		}
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
	}
	hidden_effect = {
		set_variable = nd_dnm_in_reich
	}
	nd_dnm_crown_of_st_stephen_maintain_effect = yes
}

# ----- THE CROWN OF SAINT STEPHEN ----------------------------------------
# The Hungarian Diet never accepted that the Kingdom of Hungary belonged to
# the Reich, and said so at every coronation. While the Monarchy is a
# member and Hungary still stands as a crown that answers to it (a subject
# or a union junior), Hungary carries the grievance: less loyal, its
# nobles less content. The weld ends it (Hungary is then our own land and
# the Nationalities Question owns the tension), so does the break. Run at
# the stay and from the monthly pulse, so it follows the live state
# (Hungary released, re-subjugated, annexed) without a second bookkeeping
# path. Root = the Monarchy.
nd_dnm_crown_of_st_stephen_maintain_effect = {
	save_scope_as = nd_dnm_realm
	if = {
		limit = {
			country_exists = c:HUN
			nd_dnm_is_in_reich = yes
			c:HUN = {
				nd_dnm_crown_answers_to = { REALM = scope:nd_dnm_realm }
				NOT = { has_country_modifier = nd_dnm_crown_of_st_stephen }
			}
		}
		c:HUN = {
			add_country_modifier = {
				modifier = nd_dnm_crown_of_st_stephen
				years = -1
				mode = add
			}
		}
	}
	else_if = {
		limit = {
			country_exists = c:HUN
			c:HUN = { has_country_modifier = nd_dnm_crown_of_st_stephen }
			NOR = {
				AND = {
					nd_dnm_is_in_reich = yes
					c:HUN = { nd_dnm_crown_answers_to = { REALM = scope:nd_dnm_realm } }
				}
			}
		}
		c:HUN = { remove_country_modifier = nd_dnm_crown_of_st_stephen }
	}
}
