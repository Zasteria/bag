# Angevin Empire: National Destinies new formable
# Plantagenet cross-Channel personal union (1154-1214 historical core).
# Players holding England plus the continental Plantagenet domains (Normandy,
# Anjou, Aquitaine) can re-form the Angevin Empire. Catholic, multi-cultural
# (English + french_group cultures of the Plantagenet domains).
ANV_f = {

	level = 3
	required_locations_fraction = 0.55
	rule = plausible

	potential = {
		religion = religion:catholic
		OR = {
			culture = culture:english
			culture = { has_culture_group = culture_group:french_group }
		}
		# No empire is proclaimed mid-civil-war (2026-08-03 blast-radius
		# audit, user ruling): England carries the War of the Roses and
		# English Civil War disasters, France the Wars of Religion; a
		# tag switch during one strands its event chains mid-arc.
		has_any_active_disaster = no
	}

	allow = {
		owns = location:london
		owns = location:rouen
		owns = location:angers
	}

	name = ANV
	flag = ANV
	adjective = ANV_ADJ
	tag = ANV
	color = map_ANV

	areas = {
		home_counties_area
		midlands_area
		northumbria_area
		east_anglia_area
		west_country_area
		wales_area
		leinster_area
		normandy_area
		anjou_area
		brittany_area
		poitou_area
		guyenne_area
		gascony_area
		limousin_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_anv_formed

		}
		# Empire rank, not kingdom. Level 3 formables in this mod split
		# roughly evenly between the two (10 empire / 15 kingdom), so the
		# deciding factor is scope, and this one requires England, Wales, a
		# foothold in Ireland and the whole Atlantic half of France across
		# 14 areas: a larger realm than several of the level-3 formables
		# that already grant empire (Latin Empire, Tibetan Empire). It is
		# also the only formable in the mod whose *name* is an empire while
		# its holder stayed a king, which players noticed and asked about
		# (Workshop, 2026-07-22).
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		trigger_event_non_silently = nd_anv.1
		add_stability = stability_mild_bonus
	}
}
