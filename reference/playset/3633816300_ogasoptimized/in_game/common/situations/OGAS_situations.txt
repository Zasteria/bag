OGAS = {
	monthly_spawn_chance = 1

	can_start = {
	}

	can_end = {
		always = no
	}

	visible = {
		is_ai = no
	}

	on_start = {
	}

	on_monthly = {
	}

	tooltip = {
		if = {
			limit = {
				has_owner = yes
				owner ?= {
					is_human = yes
					has_variable = OGAS_Mode
					var:OGAS_Mode = 1
				}
			}
			if = {
				limit = {
					has_variable = OGAS_Predicted
					owner ?= var:OGAS_Plan_Owner
				}
				if = {
					limit = { NOT = { has_variable = OGAS_Parent } }
					custom_tooltip = "OGAS_Complete_desc"
				}
				else_if = {
					limit = { has_latest_road_to = var:OGAS_Parent }
					custom_tooltip = "OGAS_Complete_desc"
				}
				else = {
					custom_tooltip = "OGAS_Ongoing_desc"
				}
			}
			else = {
				custom_tooltip = "OGAS_Cannot_Reach_desc"
			}
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= {
					is_human = yes
					has_variable = OGAS_Mode
					var:OGAS_Mode = 2
				}
			}
			if = {
				limit = {
					has_variable = OGAS_Predicted
					owner ?= var:OGAS_Plan_Owner
				}
				if = {
					limit = { NOT = { has_variable = OGAS_Parent } }
					custom_tooltip = "OGAS_Complete_Market_Access_desc"
				}
				else_if = {
					limit = { has_latest_road_to = var:OGAS_Parent }
					custom_tooltip = "OGAS_Complete_Market_Access_desc"
				}
				else = {
					custom_tooltip = "OGAS_Ongoing_Market_Access_desc"
				}
			}
			else = {
				custom_tooltip = "OGAS_Cannot_Reach_Market_Access_desc"
			}
		}
	}

	is_data_map = yes

	map_color = {
		if = {
			limit = {
				has_variable = OGAS_Predicted
				owner ?= { is_human = yes }
				owner ?= var:OGAS_Plan_Owner
				OR = {
					NOT = { has_variable = OGAS_Parent }
					has_latest_road_to = var:OGAS_Parent
				}
			}
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			limit = {
				has_variable = OGAS_Predicted
				owner ?= { is_human = yes }
				owner ?= var:OGAS_Plan_Owner
				has_variable = OGAS_Parent
				NOT = { has_latest_road_to = var:OGAS_Parent }
			}
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}
}
