OGAS_Auto_Road_Initialize_sgui = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		OGAS_Auto_Road_Refresh = yes
	}
}

OGAS_Auto_Road_Construct_sgui = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		OGAS_Auto_Road_Ensure_Settings = yes
		if = {
			limit = { var:OGAS_Auto_Build = 1 }
			set_variable = { name = OGAS_Auto_Build value = 0 }
		}
		else = {
			set_variable = { name = OGAS_Auto_Build value = 1 }
			if = {
				limit = {
					NOT = { any_owned_location = { has_variable = OGAS_Predicted } }
				}
				OGAS_Auto_Road_Refresh = yes
			}
		}
	}
}
