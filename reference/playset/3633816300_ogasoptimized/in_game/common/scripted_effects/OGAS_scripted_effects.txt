OGAS_Auto_Road_Ensure_Settings = {
	if = {
		limit = {
			NOT = { has_variable = OGAS_Auto_Build }
			has_variable = OGAS_Auto_Road_Construct
		}
		set_variable = {
			name = OGAS_Auto_Build
			value = var:OGAS_Auto_Road_Construct
		}
	}
	if = {
		limit = { NOT = { has_variable = OGAS_Auto_Build } }
		set_variable = { name = OGAS_Auto_Build value = 0 }
	}

	if = {
		limit = {
			NOT = { has_variable = OGAS_Mode }
			has_variable = OGAS_Auto_Road_Mode
		}
		set_variable = {
			name = OGAS_Mode
			value = var:OGAS_Auto_Road_Mode
		}
	}
	if = {
		limit = { NOT = { has_variable = OGAS_Mode } }
		set_variable = { name = OGAS_Mode value = 1 }
	}

	if = {
		limit = {
			NOT = { has_variable = OGAS_Gold_Reserve }
			has_variable = OGAS_Auto_Road_Min_Gold
		}
		set_variable = {
			name = OGAS_Gold_Reserve
			value = var:OGAS_Auto_Road_Min_Gold
		}
	}
	if = {
		limit = { NOT = { has_variable = OGAS_Gold_Reserve } }
		set_variable = { name = OGAS_Gold_Reserve value = 1000 }
	}

	if = {
		limit = { NOT = { has_variable = OGAS_Refresh_Interval } }
		set_variable = { name = OGAS_Refresh_Interval value = 3 }
	}
	if = {
		limit = { NOT = { has_variable = OGAS_Months_Since_Refresh } }
		set_variable = {
			name = OGAS_Months_Since_Refresh
			value = var:OGAS_Refresh_Interval
		}
	}
}

OGAS_Auto_Road_Clamp_Gold_Reserve = {
	if = {
		limit = { var:OGAS_Gold_Reserve < 0 }
		set_variable = { name = OGAS_Gold_Reserve value = 0 }
	}
	if = {
		limit = { var:OGAS_Gold_Reserve > 100000 }
		set_variable = { name = OGAS_Gold_Reserve value = 100000 }
	}
}

OGAS_Auto_Road_Clamp_Refresh_Interval = {
	if = {
		limit = { var:OGAS_Refresh_Interval < 0 }
		set_variable = { name = OGAS_Refresh_Interval value = 0 }
	}
	if = {
		limit = { var:OGAS_Refresh_Interval > 120 }
		set_variable = { name = OGAS_Refresh_Interval value = 120 }
	}
}

OGAS_Auto_Road_Refresh = {
	OGAS_Auto_Road_Ensure_Settings = yes

	remove_variable = OGAS_Debug_Queue_Exhausted
	remove_variable = OGAS_Queue_Head
	remove_variable = OGAS_Queue_Tail
	remove_variable = OGAS_Auto_Road_Queue_Head
	remove_variable = OGAS_Auto_Road_Queue_Tail
	set_variable = { name = OGAS_Months_Since_Refresh value = 0 }

	every_owned_location = {
		remove_variable = OGAS_Predicted
		remove_variable = OGAS_Parent
		remove_variable = OGAS_Value
		remove_variable = OGAS_Plan_Owner
		remove_variable = OGAS_Subtree_Demand
		remove_variable = OGAS_InQueue
		remove_variable = OGAS_Queue_Next

		remove_variable = OGAS_Cache_Auto_Road_Complete
		remove_variable = OGAS_Cache_Auto_Road_Ongoing
		remove_variable = OGAS_Cache_Auto_Road_Source
		remove_variable = OGAS_Cache_Auto_Road_Value
		remove_variable = OGAS_Cache_Auto_Road_InQueue
		remove_variable = OGAS_Cache_Auto_Road_Queue_Next
	}

	if = {
		limit = { var:OGAS_Mode = 1 }
		OGAS_Auto_Road_Pathfinding_Proximity = yes
		OGAS_Auto_Road_Calculate_Value_Proximity = yes
	}
	else_if = {
		limit = { var:OGAS_Mode = 2 }
		OGAS_Auto_Road_Pathfinding_Market_Access = yes
		OGAS_Auto_Road_Calculate_Value_Market_Access = yes
	}

	refresh_map_colors = yes
}

OGAS_Auto_Road_Queue_Enqueue = {
	save_temporary_scope_as = OGAS_Queue_New_Node
	if = {
		limit = { NOT = { has_variable = OGAS_InQueue } }
		set_variable = OGAS_InQueue
		remove_variable = OGAS_Queue_Next
		root = {
			if = {
				limit = { has_variable = OGAS_Queue_Tail }
				var:OGAS_Queue_Tail = {
					set_variable = {
						name = OGAS_Queue_Next
						value = scope:OGAS_Queue_New_Node
					}
				}
				set_variable = {
					name = OGAS_Queue_Tail
					value = scope:OGAS_Queue_New_Node
				}
			}
			else = {
				set_variable = {
					name = OGAS_Queue_Head
					value = scope:OGAS_Queue_New_Node
				}
				set_variable = {
					name = OGAS_Queue_Tail
					value = scope:OGAS_Queue_New_Node
				}
			}
		}
	}
}

OGAS_Auto_Construct_Road = {
	OGAS_Auto_Road_Ensure_Settings = yes
	if = {
		limit = {
			var:OGAS_Auto_Build = 1
			has_advance = road_building
		}
		set_local_variable = { name = OGAS_Build_Stop value = 0 }
		ordered_owned_location = {
			limit = {
				has_variable = OGAS_Parent
				has_variable = OGAS_Value
				owner ?= var:OGAS_Plan_Owner
				var:OGAS_Parent = { owner ?= root }
				NOT = { has_latest_road_to = var:OGAS_Parent }
				has_road_constructions = no
			}
			order_by = var:OGAS_Value
			max = 100

			if = {
				limit = { local_var:OGAS_Build_Stop = 0 }
				if = {
					limit = {
						root.gold >= {
							value = root.var:OGAS_Gold_Reserve
							add = OGAS_Auto_Road_Gold_Cost
						}
					}
					save_temporary_scope_as = OGAS_Auto_Construct_Road_Scope_From
					var:OGAS_Parent = { save_temporary_scope_as = OGAS_Auto_Construct_Road_Scope_To }
					if = {
						limit = { owner = { has_advance = railroad_advance } }
						owner = {
							construct_road = {
								road_type = road_type:railroad
								from = scope:OGAS_Auto_Construct_Road_Scope_From
								to = scope:OGAS_Auto_Construct_Road_Scope_To
							}
						}
					}
					else_if = {
						limit = { owner = { has_advance = modern_road_advance } }
						owner = {
							construct_road = {
								road_type = road_type:modern_road
								from = scope:OGAS_Auto_Construct_Road_Scope_From
								to = scope:OGAS_Auto_Construct_Road_Scope_To
							}
						}
					}
					else_if = {
						limit = { owner = { has_advance = paved_road_advance } }
						owner = {
							construct_road = {
								road_type = road_type:paved_road
								from = scope:OGAS_Auto_Construct_Road_Scope_From
								to = scope:OGAS_Auto_Construct_Road_Scope_To
							}
						}
					}
					else_if = {
						limit = { owner = { has_advance = road_building } }
						owner = {
							construct_road = {
								road_type = road_type:gravel_road
								from = scope:OGAS_Auto_Construct_Road_Scope_From
								to = scope:OGAS_Auto_Construct_Road_Scope_To
							}
						}
					}
				}
				else = {
					set_local_variable = { name = OGAS_Build_Stop value = 1 }
				}
			}
		}
	}
}
