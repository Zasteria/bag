# Player toggles that block or allow this mod auto-improving relations with a country.
# atd_improve_blocked is read by atd_is_valid_improve_target.

atd_block_autonomous_improving = {
	type = diplomacy
	category = CATEGORY_ATD_ACTIONS
	use_enroute = no
	ai_tick = never
	show_message = no
	show_message_to_target = no

	potential = {}
	allow = {}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			NOT = { has_variable = atd_improve_blocked }
		}
		enabled = {}
	}

	effect = {
		scope:recipient = {
			set_variable = atd_improve_blocked
		}
	}
}

atd_allow_autonomous_improving = {
	type = diplomacy
	category = CATEGORY_ATD_ACTIONS
	use_enroute = no
	ai_tick = never
	show_message = no
	show_message_to_target = no

	potential = {}
	allow = {}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			has_variable = atd_improve_blocked
		}
		enabled = {}
	}

	effect = {
		scope:recipient = {
			remove_variable = atd_improve_blocked
		}
	}
}

# Start / Stop Auto-Gifting queue a country for the send-gift automation in
# atd_auto_interactions_effects.txt (one queue slot per country).

atd_start_auto_gifting = {
	type = diplomacy
	category = CATEGORY_ATD_ACTIONS
	use_enroute = no
	ai_tick = never
	show_message = no
	show_message_to_target = no

	potential = {}
	allow = {}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			atd_is_queued_for_auto_interaction = no
			scope:actor = { var:atd_queue_count < 50 }
		}
		enabled = {}
	}

	effect = {
		scope:recipient = { save_scope_as = atd_queue_add_target }
		scope:actor = {
			atd_queue_find_free_slot = yes
			# Hovering the interaction evaluates these limits without running the finder above.
			if = {
				limit = { has_variable = atd_queue_free_slot }
				if = {
					limit = { var:atd_queue_free_slot > 0 }
					atd_queue_assign_slot_by_number = { slot_value = var:atd_queue_free_slot type = send_gifts }
				}
				remove_variable = atd_queue_free_slot
			}
		}
	}
}

atd_stop_auto_gifting = {
	type = diplomacy
	category = CATEGORY_ATD_ACTIONS
	use_enroute = no
	ai_tick = never
	show_message = no
	show_message_to_target = no

	potential = {}
	allow = {}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			atd_is_queued_for_auto_interaction = yes
		}
		enabled = {}
	}

	effect = {
		scope:recipient = { save_scope_as = atd_queue_remove_target }
		scope:actor = {
			atd_queue_clear_target = yes
		}
	}
}
