# Interactions (interactions) Tab
atd__auto_interactions_on_changed = {
	scope = country

	effect = {
		cmm_apply_list_change = {
			setting = atd__auto_interactions
		}
		# The cancel button (f2) is the list's only button field.
		if = {
			limit = { has_variable = cmm_list_button_item }
			atd_queue_clear_slot_by_number = { slot_value = var:cmm_list_button_item }
			remove_variable = cmm_list_button_item
			remove_variable = cmm_list_button_slot
		}
	}
}

# Diplomat reserve pull from the automation panel slider. Polled, so the
# guards live here: skip while a push is pending, while the slider reads the
# stale zero, and when the value already matches. Writes the CMM value map
# directly so the settings menu shows the new value without firing the
# callback, which would set atd_reserve_from_cmm and push right back.
# Root: player country. Saved scopes: atd_reserve (value).
atd_automation_reserve_changed = {
	scope = country
	saved_scopes = { atd_reserve }

	effect = {
		set_local_variable = { name = atd_tmp_reserve value = scope:atd_reserve }
		if = {
			limit = {
				NOT = { has_variable = atd_reserve_from_cmm }
				local_var:atd_tmp_reserve > 0
				NOT = { var:atd_min_diplomats = local_var:atd_tmp_reserve }
			}
			set_variable = { name = atd_min_diplomats value = scope:atd_reserve }
			remove_from_variable_map = { name = cmm key = flag:atd__min_diplomats }
			add_to_variable_map = { name = cmm key = flag:atd__min_diplomats value = scope:atd_reserve }
		}
	}
}

# Root: player country. Saved scopes: atd_max (value).

		atd_update_reserve_slider_bounds = yes
	}
}

# Root: player country.
atd_automation_reserve_pushed = {
	scope = country

	effect = {
		if = {
			limit = { has_variable = atd_reserve_from_cmm }
			remove_variable = atd_reserve_from_cmm
		}
	}
}

# Root: player country.
atd_automation_reserve_reconcile = {
	scope = country

	effect = {

	}
}

# Marks the engine's improve-opinion sub-flag as cleared for this save. Stored on
# the country rather than in GetVariableSystem, whose variables outlive the game
# they were set in and so would suppress the clear in every later game of the
# same session.
# Root: player country.
atd_automation_engine_improve_cleared = {
	scope = country

	effect = {
		set_variable = atd_engine_improve_opinion_cleared

		# Only the card's visibility gate reads it, and a GUI read doesn't count.
		if = {
			limit = { always = no }
			cmf_suppress = { v = atd_engine_improve_opinion_cleared }
		}
	}
}
