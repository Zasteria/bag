atd_enforce_on_subjects = {
	save_scope_as = atd_actor_scope
	set_variable = { name = atd_enforce_threshold value = { value = var:atd_min_diplomats add = 1 } }
	if = {
		limit = {
			num_of_diplomats >= var:atd_enforce_threshold
		}
		cmm_for_each_list_item = {
			setting = atd__enforce
			effect = atd_enforce_each_item
		}
	}
}

atd_enforce_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:religion = {
			if = {
				limit = { has_variable = atd_enforce_religion_enabled }
				atd_enforce_religion_on_subjects = yes
			}
		}
		flag:culture = {
			if = {
				limit = { has_variable = atd_enforce_culture_enabled }
				atd_enforce_culture_on_subjects = yes
			}
		}
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

atd_enforce_religion_on_subjects = {
	while = {
		limit = {
			num_of_diplomats >= var:atd_enforce_threshold
			NOT = { has_cooldown = enforce_religion }
			# Mirrors the vanilla subject-action gates MIN_SUBJECT_LOYALTY_FOR_ACTION
			# and MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION, both 50:
			# in_game/common/script_values/define_values.txt:33
			any_subject = {
				NOT = { religion = scope:atd_actor_scope.religion }
				subject_loyalty >= 50
				liberty_desire < 50
				trigger_if = {
					limit = {
						modifier:blocked_from_conversion = yes
						has_reform = government_reform:military_order_reform
					}
					scope:atd_actor_scope = { has_reform = government_reform:military_order_reform }
				}
				trigger_else = {
					modifier:blocked_from_conversion = no
				}
			}
		}
		count = 10
		random_subject = {
			limit = {
				NOT = { religion = scope:atd_actor_scope.religion }
				subject_loyalty >= 50
				liberty_desire < 50
				trigger_if = {
					limit = {
						modifier:blocked_from_conversion = yes
						has_reform = government_reform:military_order_reform
					}
					scope:atd_actor_scope = { has_reform = government_reform:military_order_reform }
				}
				trigger_else = {
					modifier:blocked_from_conversion = no
				}
			}
			save_scope_as = atd_target_subject
			enforce_religion_effect = {
				actor = scope:atd_actor_scope
				target = scope:atd_target_subject
			}
			add_liberty_desire = liberty_desire_ultimate_plus
		}
		add_diplomats = -1
		# Mirrors the vanilla enforce_religion interaction cooldown:
		# in_game/common/country_interactions/enforce_religion.txt:12
		add_cooldown = {
			type = enforce_religion
			years = 10
		}
	}
}
