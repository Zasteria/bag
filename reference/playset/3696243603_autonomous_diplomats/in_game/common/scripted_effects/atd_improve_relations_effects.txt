atd_improve_relations = {
	save_scope_as = atd_actor_scope
	set_variable = { name = atd_action_threshold value = { value = var:atd_min_diplomats add = 1 } }
	atd_check_for_cancel = yes
	if = {
		limit = {
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		atd_update_type_counters = yes
		cmm_for_each_list_item = {
			setting = atd__improve_relations
			effect = atd_improve_relations_each_item
		}
	}
}

# Sets atd_pending_cancel on each tracked improve target whose opinion of the
# player has reached var:atd_max_relations, and on each target claimed by the
# outraged category whose antagonism towards the player is below 5; clears it
# otherwise. Read by the auto-cancel widget in atd_outliner_widgets.gui.
# The requester also records the target in its own atd_cancel_requests map, which
# is what keeps one human player's cancel off another's improve on that target.
# Scope: player country (atd_improve_relations saves atd_actor_scope).
atd_check_for_cancel = {
	if = {
		limit = { has_variable_map = atd_improving_with }
		every_key_in_variable_map = {
			variable = atd_improving_with
			save_scope_as = atd_cancel_target
			scope:atd_cancel_target = {
				# A cap of 200, the slider max, means no limit.
				if = {
					limit = {
						scope:atd_actor_scope = { var:atd_max_relations < 200 }
						opinion = { target = scope:atd_actor_scope value >= scope:atd_actor_scope.var:atd_max_relations }
					}
					if = {
						limit = { NOT = { has_variable = atd_pending_cancel } }
						set_variable = atd_pending_cancel
						scope:atd_actor_scope = {
							add_to_variable_map = { name = atd_cancel_requests key = scope:atd_cancel_target value = 1 }
						}
						# Keeps the mod from re-improving this country until it expires.
						set_variable = { name = atd_cancel_cooldown value = yes years = 4 }
					}
				}
				# Cancel outraged targets below 5 antagonism.
				else_if = {
					limit = {
						scope:atd_actor_scope = {
							has_variable_map = atd_improving_type
							is_key_in_variable_map = { name = atd_improving_type target = scope:atd_cancel_target }
							"variable_map(atd_improving_type|scope:atd_cancel_target)" = flag:outraged
						}
						antagonism = { target = scope:atd_actor_scope value < 5 }
					}
					if = {
						limit = { NOT = { has_variable = atd_pending_cancel } }
						set_variable = atd_pending_cancel
						scope:atd_actor_scope = {
							add_to_variable_map = { name = atd_cancel_requests key = scope:atd_cancel_target value = 1 }
						}
					}
				}
				else = {
					if = {
						limit = {
							has_variable = atd_pending_cancel
							scope:atd_actor_scope = {
								has_variable_map = atd_cancel_requests
								is_key_in_variable_map = { name = atd_cancel_requests target = scope:atd_cancel_target }
							}
						}
						remove_variable = atd_pending_cancel
						scope:atd_actor_scope = {
							remove_from_variable_map = { name = atd_cancel_requests key = scope:atd_cancel_target }
						}
					}
				}
			}
		}
	}
}

# Recomputes per-type active counters from atd_improving_type. Runs once per
# monthly tick.
atd_update_type_counters = {
	set_variable = { name = atd_n_neighbors_improving_with value = 0 }
	set_variable = { name = atd_n_subjects_improving_with value = 0 }
	set_variable = { name = atd_n_outraged_improving_with value = 0 }
	set_variable = { name = atd_n_allies_improving_with value = 0 }
	set_variable = { name = atd_n_threatened_improving_with value = 0 }
	set_variable = { name = atd_n_hre_members_improving_with value = 0 }
	set_variable = { name = atd_n_hre_electors_improving_with value = 0 }
	set_variable = { name = atd_n_market_owners_improving_with value = 0 }
	set_variable = { name = atd_n_kingdoms_improving_with value = 0 }
	set_variable = { name = atd_n_imperial_circle_leaders_improving_with value = 0 }
	set_variable = { name = atd_n_largest_improving_with value = 0 }

	if = {
		limit = {
			has_variable_map = atd_improving_type
		}
		every_key_in_variable_map = {
			variable = atd_improving_type
			save_scope_as = atd_nation_improving
			scope:atd_actor_scope = {
				switch = {
					trigger = "variable_map(atd_improving_type|scope:atd_nation_improving)"
					flag:neighbor = {
						change_variable = { name = atd_n_neighbors_improving_with add = 1 }
					}
					flag:subject = {
						change_variable = { name = atd_n_subjects_improving_with add = 1 }
					}
					flag:outraged = {
						change_variable = { name = atd_n_outraged_improving_with add = 1 }
					}
					flag:ally = {
						change_variable = { name = atd_n_allies_improving_with add = 1 }
					}
					flag:threatened = {
						change_variable = { name = atd_n_threatened_improving_with add = 1 }
					}
					flag:hre_member = {
						change_variable = { name = atd_n_hre_members_improving_with add = 1 }
					}
					flag:hre_elector = {
						change_variable = { name = atd_n_hre_electors_improving_with add = 1 }
					}
					flag:market_owner = {
						change_variable = { name = atd_n_market_owners_improving_with add = 1 }
					}
					flag:kingdom = {
						change_variable = { name = atd_n_kingdoms_improving_with add = 1 }
					}
					flag:imperial_circle_leader = {
						change_variable = { name = atd_n_imperial_circle_leaders_improving_with add = 1 }
					}
					flag:largest = {
						change_variable = { name = atd_n_largest_improving_with add = 1 }
					}
				}
			}
		}
	}
}

# Clears the per-cycle "seen" marker before the outliner sync widgets fire
# their _show and populate it. Called by atd_on_monthly before flipping the
# hidden window's gate. Scope: player country.
atd_gui_sync_begin = {
	if = {
		limit = { has_variable_list = atd_improving_seen }
		clear_variable_list = atd_improving_seen
	}
}

# Reconciles atd_improving_with against atd_improving_seen: evict tracked
# entries not seen this pass, add newly-seen-but-untracked entries as
# player-tracked (no atd_improving_type, so they don't count toward mod
# budgets). Stale entries collected into atd_improving_to_remove first to
# avoid mutating atd_improving_with during its own iteration.
# Scope: player country.
atd_gui_sync_end = {
	save_scope_as = atd_sync_actor
	if = {
		limit = { has_variable_list = atd_improving_to_remove }
		clear_variable_list = atd_improving_to_remove
	}
	if = {
		limit = { has_variable_map = atd_improving_with }
		every_key_in_variable_map = {
			variable = atd_improving_with
			save_scope_as = atd_nation_to_check
			scope:atd_sync_actor = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_list = atd_improving_seen }
							NOT = {
								is_target_in_variable_list = {
									name = atd_improving_seen
									target = scope:atd_nation_to_check
								}
							}
						}
					}
					add_to_variable_list = {
						name = atd_improving_to_remove
						target = scope:atd_nation_to_check
					}
				}
			}
		}
	}
	if = {
		limit = { has_variable_list = atd_improving_to_remove }
		every_in_list = {
			variable = atd_improving_to_remove
			save_scope_as = atd_nation_to_remove
			scope:atd_sync_actor = {
				remove_from_variable_map = {
					name = atd_improving_with
					key = scope:atd_nation_to_remove
				}
				if = {
					limit = {
						has_variable_map = atd_improving_type
						is_key_in_variable_map = {
							name = atd_improving_type
							target = scope:atd_nation_to_remove
						}
					}
					remove_from_variable_map = {
						name = atd_improving_type
						key = scope:atd_nation_to_remove
					}
					# A mod improve gone below the +200 cap never persisted.
					scope:atd_nation_to_remove = {
						if = {
							limit = { NOT = { has_variable = atd_cancel_cooldown } }
							set_variable = { name = atd_retarget_cooldown value = yes years = 1 }
						}
					}
				}
			}
			# Clear the auto-cancel flag from the target country so a later
			# re-targeting doesn't get cancelled immediately by stale state.
			# atd_check_for_cancel can't clean these up after the
			# country leaves atd_improving_with because it only iterates
			# entries that ARE in atd_improving_with.
			scope:atd_sync_actor = {
				if = {
					limit = { has_variable_map = atd_cancel_requests }
					remove_from_variable_map = { name = atd_cancel_requests key = scope:atd_nation_to_remove }
				}
			}
			scope:atd_nation_to_remove = {
				if = {
					limit = { has_variable = atd_pending_cancel }
					remove_variable = atd_pending_cancel
				}
			}
		}
		clear_variable_list = atd_improving_to_remove
	}
	# Add newly discovered improvements (typically player-initiated) to the map
	# so subsequent mod ticks correctly see them as in-progress and skip them.
	if = {
		limit = { has_variable_list = atd_improving_seen }
		every_in_list = {
			variable = atd_improving_seen
			save_scope_as = atd_nation_seen
			scope:atd_sync_actor = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_map = atd_improving_with }
							NOT = {
								is_key_in_variable_map = {
									name = atd_improving_with
									target = scope:atd_nation_seen
								}
							}
						}
					}
					add_to_variable_map = {
						name = atd_improving_with
						key = scope:atd_nation_seen
						value = 1
					}
				}
			}
		}
		clear_variable_list = atd_improving_seen
	}
}

atd_improve_relations_each_item = {
	# $i$ is the resolved item number (1-11)
	# scope:cmm_list_current_item_value  (attached game object)
	# Per-item aliases for count:
	#   Imperial Circle Leaders: var:atd_improve_imperial_circle_leaders_count
	#   HRE Electors: var:atd_improve_hre_electors_count
	#   HRE Members: var:atd_improve_hre_members_count
	#   Neighboring Countries: var:atd_improve_neighbors_count
	#   Own Subject Countries: var:atd_improve_relations_count
	#   Outraged Countries: var:atd_improve_outraged_count
	#   Allies: var:atd_improve_allies_count
	#   Threatening Countries: var:atd_improve_threatened_count
	#   Market Owners: var:atd_improve_market_owners_count
	#   Kingdoms and Empires: var:atd_improve_kingdoms_count
	#   Largest Nations: var:atd_improve_largest_count

	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:neighbors = { atd_improve_neighbors = yes }
		flag:subjects = { atd_improve_subjects = yes }
		flag:outraged = { atd_improve_outraged = yes }
		flag:allies = { atd_improve_allies = yes }
		flag:threatening = { atd_improve_threatened = yes }
		flag:hre_members = { atd_improve_hre_members = yes }
		flag:hre_electors = { atd_improve_hre_electors = yes }
		flag:market_owners = { atd_improve_market_owners = yes }
		flag:kingdoms = { atd_improve_kingdoms = yes }
		flag:imperial_circle_leaders = { atd_improve_imperial_circle_leaders = yes }
		flag:largest = { atd_improve_largest = yes }
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Starts improving relations with scope:atd_target_country if not already improving.
# Must be called inside scope:atd_actor_scope.
# Params: type (type flag name), counter (counter variable name)
atd_start_improving_with = {
	if = {
		limit = {
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
			scope:atd_target_country = { atd_is_valid_improve_target = yes }
		}
		perform_diplomatic_action = {
			type = improve_relation
			actor = scope:atd_actor_scope
			recipient = scope:atd_target_country
		}
		add_to_variable_map = {
			name = atd_improving_with
			key = scope:atd_target_country
			value = 1
		}
		add_to_variable_map = {
			name = atd_improving_type
			key = scope:atd_target_country
			value = flag:$type$
		}
		change_variable = { name = $counter$ add = 1 }
	}

	# Suppress "used but never set" warnings for type flags passed via $type$
	if = {
		limit = { always = no }
		cmf_suppress = { v = neighbor }
		cmf_suppress = { v = subject }
		cmf_suppress = { v = outraged }
		cmf_suppress = { v = ally }
		cmf_suppress = { v = threatened }
		cmf_suppress = { v = hre_member }
		cmf_suppress = { v = hre_elector }
		cmf_suppress = { v = market_owner }
		cmf_suppress = { v = kingdom }
		cmf_suppress = { v = imperial_circle_leader }
		cmf_suppress = { v = largest }
	}
}

atd_improve_neighbors = {
	while = {
		limit = {
			var:atd_n_neighbors_improving_with < var:atd_improve_neighbors_count
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		count = 10
		random_neighbor_country = {
			limit = {
				atd_is_valid_improve_target = yes
			}
			save_scope_as = atd_target_country
			scope:atd_actor_scope = {
				atd_start_improving_with = { type = neighbor counter = atd_n_neighbors_improving_with }
			}
		}
	}
}

		limit = {
			var:atd_n_outraged_improving_with < var:atd_improve_outraged_count
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		count = 10
		ordered_country_in_diplomatic_range = {
			limit = {
				# Outraged at 25+ antagonism.
				antagonism = { target = scope:atd_actor_scope value >= 25 }
				atd_is_valid_improve_target = yes
			}
			order_by = this_antagonism_towards_prev
			max = 1
			save_scope_as = atd_target_country
			scope:atd_actor_scope = {
				atd_start_improving_with = { type = outraged counter = atd_n_outraged_improving_with }
			}

	}
}
