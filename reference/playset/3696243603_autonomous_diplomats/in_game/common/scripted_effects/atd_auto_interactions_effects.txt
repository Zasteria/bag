# $i$ is the stable slot number (1-50). Empty slots have no atd_queue_target_$i$
# and are skipped; dead targets free their slot.
atd_auto_interactions_each_item = {
	if = {
		limit = { has_variable = atd_queue_target_$i$ }
		if = {
			limit = { NOT = { exists = var:atd_queue_target_$i$ } }
			atd_queue_clear_slot = { slot = $i$ }
		}
		else = {
			set_local_variable = { name = atd_pause_key value = flag:atd__auto_interactions_i$i$_f1 }
			if = {
				limit = { "variable_map(cmm|local_var:atd_pause_key)" < 1 }
				switch = {
					trigger = var:atd_queue_type_$i$
					flag:send_gifts = {
						atd_send_gift = { slot = $i$ }
					}
				}
			}
			else = {
				cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $i$ value = flag:atd_cmm_status_paused }
			}
		}
	}

	cmf_suppress = { v = $setting$ }
}

# Sends the maximum-generosity gift to slot $slot$'s target when the target is
# valid, its gift cooldown is over, the diplomat reserve holds, and the treasury
# stays above the gold reserve after the cost. Writes the slot's status column
# with whichever gate blocked it.
atd_send_gift = {
	var:atd_queue_target_$slot$ = { save_scope_as = atd_gift_target }
	# A countdown without the engine cooldown is stale (the send never landed).
	if = {
		limit = {
			has_variable = atd_queue_cd_$slot$
			NOT = { has_gifted_gold_to = scope:atd_gift_target }
		}
		remove_variable = atd_queue_cd_$slot$
	}
	if = {
		limit = { has_gifted_gold_to = scope:atd_gift_target }
		if = {
			limit = { has_variable = atd_queue_cd_$slot$ }
			change_variable = { name = atd_queue_cd_$slot$ add = -1 }
			if = {
				limit = { var:atd_queue_cd_$slot$ < 1 }
				remove_variable = atd_queue_cd_$slot$
			}
		}
		if = {
			limit = { has_variable = atd_queue_cd_$slot$ }
			cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_cooldown_$slot$ }
		}
		else = {
			cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_cooldown }
		}
	}
	else_if = {
		limit = { NOT = { scope:atd_gift_target = { atd_is_valid_gift_target = yes } } }
		cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_blocked }
	}
	else_if = {
		limit = { NOT = { num_of_diplomats >= var:atd_gift_threshold } }
		cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_no_diplomats }
	}
	else = {
		# Mirrors the send_gift price, 2 months of recipient income plus 1 of the
		# sender's with a floor of 25 (in_game/common/prices/00_hardcoded.txt:155),
		# scaled by Send Gift Cost and doubled for the two gifts sent.
		set_variable = {
			name = atd_gift_cost
			value = {
				value = scope:atd_gift_target.monthly_income_total
				multiply = 2
				add = monthly_income_total
			}
		}
		if = {
			limit = { var:atd_gift_cost < 25 }
			set_variable = { name = atd_gift_cost value = 25 }
		}
		set_variable = {
			name = atd_gift_cost_scale
			value = {
				value = modifier:send_gift_cost_modifier
				add = 1
				multiply = 2
			}
		}
		change_variable = { name = atd_gift_cost multiply = var:atd_gift_cost_scale }
		remove_variable = atd_gift_cost_scale

		if = {
			limit = {
				gold >= {
					value = var:atd_min_gold
					add = var:atd_gift_cost
				}
			}
			# Script cannot set the gift's generosity slider, so two default-size
			# gifts stand in for one maxed gift: double gold, 10 favors, and the
			# full 40 opinion.
			perform_diplomatic_action = {
				type = giftaction
				actor = scope:atd_actor_scope
				recipient = scope:atd_gift_target
			}
			perform_diplomatic_action = {
				type = giftaction
				actor = scope:atd_actor_scope
				recipient = scope:atd_gift_target
			}
			# The pair costs two diplomats; refund one so a queue entry costs
			# the same single diplomat a manual gift does.
			add_diplomats = 1
			change_variable = { name = atd_gift_threshold add = 1 }
			# SEND_GIFT_COOLDOWN months (game loading_screen/common/defines/00_defines.txt:1994).
			set_variable = { name = atd_queue_cd_$slot$ value = 120 }
			cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_cooldown_$slot$ }
		}
		else = {
			cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_no_gold }
		}
		remove_variable = atd_gift_cost
	}
}

# Binds scope:atd_queue_add_target to slot $slot$ with interaction type flag
# $type$ and shows its row.
atd_queue_assign_slot = {
	set_variable = { name = atd_queue_target_$slot$ value = scope:atd_queue_add_target }
	set_variable = { name = atd_queue_type_$slot$ value = flag:$type$ }
	add_to_variable_list = { name = atd_queue_members target = scope:atd_queue_add_target }
	# A reused slot keeps the last occupant's pause state.
	remove_from_variable_map = { name = cmm key = flag:atd__auto_interactions_i$slot$_f1 }
	add_to_variable_map = { name = cmm key = flag:atd__auto_interactions_i$slot$_f1 value = 0 }
	change_variable = { name = atd_queue_count add = 1 }
	cmm_set_list_text_value = { mod_id = atd setting_id = auto_interactions field_id = status item = $slot$ value = flag:atd_cmm_status_pending }
	cmm_show_list_item = { mod_id = atd setting_id = auto_interactions item = $slot$ }
	# Keep active rows contiguous at the top so the reorder arrows always swap visible rows.
	set_variable = { name = atd_queue_move_pos value = { value = var:atd_queue_count subtract = 1 } }
	cmm_move_list_item = { setting = atd__auto_interactions item = $slot$ position = var:atd_queue_move_pos }
	remove_variable = atd_queue_move_pos
}

# Frees slot $slot$, hides its row, and parks it at the bottom of the display order.
atd_queue_clear_slot = {
	if = {
		limit = { has_variable = atd_queue_target_$slot$ }
		remove_list_variable = { name = atd_queue_members target = var:atd_queue_target_$slot$ }
	}
	remove_variable = atd_queue_target_$slot$
	remove_variable = atd_queue_type_$slot$
	remove_variable = atd_queue_cd_$slot$
	change_variable = { name = atd_queue_count add = -1 }
	cmm_hide_list_item = { mod_id = atd setting_id = auto_interactions item = $slot$ }
	cmm_move_list_item = { setting = atd__auto_interactions item = $slot$ position = 49 }
}
