# True when this country can receive an auto-sent gift from scope:atd_actor_scope.
# Opt-in queueing is explicit, so the improve-relations block flag and opinion
# caps deliberately do not apply here.
# Scope: candidate/target country. Reads scope:atd_actor_scope.
atd_is_valid_gift_target = {
	is_discovered_by = scope:atd_actor_scope
	NOT = { is_at_war_with = scope:atd_actor_scope }
}

# True when scope:actor already queued this country for auto-interactions.
# Scope: candidate/target country. Reads scope:actor (interaction picker).
atd_is_queued_for_auto_interaction = {
	scope:actor = {
		has_variable_list = atd_queue_members
		is_target_in_variable_list = {
			name = atd_queue_members
			target = PREV
		}
	}
}
