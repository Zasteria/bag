# True when this country is a valid new improve-relations target for scope:atd_actor_scope.
# Scope: candidate/target country. Reads scope:atd_actor_scope (saved by atd_improve_relations).
atd_is_valid_improve_target = {
	is_discovered_by = scope:atd_actor_scope
	NOT = { is_at_war_with = scope:atd_actor_scope }
	# A cap of 200, the slider max, means no limit.
	trigger_if = {
		limit = { scope:atd_actor_scope = { var:atd_max_relations < 200 } }
		NOT = { opinion = { target = scope:atd_actor_scope value >= scope:atd_actor_scope.var:atd_max_relations } }
	}
	trigger_if = {
		limit = { scope:atd_actor_scope = { var:atd_max_antagonism < 200 } }
		NOT = { antagonism = { target = scope:atd_actor_scope value > scope:atd_actor_scope.var:atd_max_antagonism } }
	}
	NOT = { has_variable = atd_improve_blocked }
	NOT = { has_variable = atd_cancel_cooldown }
	NOT = { has_variable = atd_retarget_cooldown }
	scope:atd_actor_scope = {
		OR = {
			NOT = { has_variable_map = atd_improving_with }
			NOT = {
				is_key_in_variable_map = {
					name = atd_improving_with
					target = PREV
				}
			}
		}
	}
}
