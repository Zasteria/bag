cmf_monthly_human_country_pulse = {
	on_actions = {
		atd_on_monthly
	}
}

# Clears atd_improving_seen before flipping the hidden window's gate, so the
# outliner sync widgets fire their _show into a fresh marker list when the
# gate flips visible.
atd_on_monthly = {
	effect = {
		atd_update_hre_visibility = yes
		if = {
			limit = { has_variable = atd_enabled }
			atd_gui_sync_begin = yes
			set_variable = atd_pending_action
		}
	}
}
