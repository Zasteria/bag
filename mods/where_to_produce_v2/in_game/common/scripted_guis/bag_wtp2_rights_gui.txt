# Кнопки окна прав -- по двери на эффект.

bag_wtp2_tr_press_now = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_tr_run_now = yes
	}
}

bag_wtp2_tr_press_end = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_tr_run_end = yes
	}
}

# Строка провинции. `saved_scopes` -- как scripted GUI получает `AddScope`,
# который передала кнопка строки.
bag_wtp2_tr_press_row = {
	scope = country
	saved_scopes = { wtp2_row }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp2_row = {
			bag_wtp2_tr_toggle_open = yes
		}
	}
}

# Считает ли окно «на конец». Читается только интерфейсом, чтобы показать,
# каким режимом посчитано то, что в нём лежит.
bag_wtp2_tr_is_end = {
	scope = country

	is_shown = {
		has_global_variable = bag_wtp2_tr_end
		global_var:bag_wtp2_tr_end = 1
	}

	effect = {
	}
}
