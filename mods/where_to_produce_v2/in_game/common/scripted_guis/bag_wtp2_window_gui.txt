# Мостики окна в скрипт. Кнопка в `.gui` умеет позвать только scripted GUI,
# и каждый из них -- дверь ровно в один эффект.
#
# `saved_scopes` -- это то, как scripted GUI получает `AddScope`, переданный
# кнопкой. Здесь он не нужен ни одному: земля выбирается действиями на карте,
# а они приходят своим путём, через `generic_actions`.

bag_wtp2_close_plan_a_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_close_plan_a = yes
	}
}

bag_wtp2_close_plan_b_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_close_plan_b = yes
	}
}

# «Моя земля» и «Очистить» -- обе кнопки стоят в обоих окнах и делают одно и то
# же с одним и тем же выбором: земля у мода одна.
bag_wtp2_press_my_land = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_pick_my_land = yes
	}
}

bag_wtp2_press_clear = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp2_clear_all = yes
	}
}
