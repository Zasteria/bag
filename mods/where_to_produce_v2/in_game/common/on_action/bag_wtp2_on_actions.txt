# CMF's two extension points, and nothing else. Both merge with whatever other
# mods declare, so nothing of another mod's wiring is displaced.
#
# **Only these two are CMF's own.** Vanilla fires `on_game_start` before the
# country is chosen, so `is_ai = no` matches nothing there; CMF's
# `on_game_start_after_lobby` chain exists for that and is fired from a scripted
# GUI of CMF's (`cmf_after_lobby_caller_scripted_gui.txt`). If this mod ever
# leaves CMF, that file is the thing to copy -- the mechanism is script, not
# engine.

# New game, save load and country transfer, once per human country.
# Scope: country
cmf_on_mod_registration = {
	on_actions = {
		bag_wtp2_on_registration
	}
}

# Scope: country
bag_wtp2_on_registration = {
	effect = {
		bag_wtp2_register = yes
	}
}

# A CMM setting changed, or a button on the mod page was clicked. The lists do
# not arrive here -- a list reaches nothing but its own `_on_changed` scripted
# GUI, which is why those are generated beside the lists themselves.
# Scope: country
cmf_on_callback = {
	on_actions = {
		bag_wtp2_on_callback
	}
}

# Scope: country
bag_wtp2_on_callback = {
	effect = {
		bag_wtp2_on_button = yes
	}
}
