# The mod page: three tabs, three buttons, and that is all of it.
#
# **The rule this mod starts from, and the first one broke twice before
# learning it**: у функции на странице мода одна кнопка «открыть», всё
# остальное живёт внутри её окна. Так что здесь нет ни одного лимита, ни одного
# тумблера и ни одного списка товаров -- только «Земля», две двери и
# «Техническая».
#
# | вкладка | что на ней |
# | --- | --- |
# | «Земля» | континенты и регионы -- рамка, внутри которой всё ищется |
# | «Расчёты» | «План 1» и «План 2»: каждая кнопка только открывает своё окно |
# | «Техническая» | «Диагностика» и штамп сборки |
#
# **A tab exists because a setting names it**, in the order the settings are
# registered (`_cmm_register_tab`, CMF 2.4.1): the zone lists come first, so
# «Земля» is the tab the page opens on.
#
# Scope: country
bag_wtp2_register = {
	cmf_register_mod = { mod_id = bag_wtp2 }

	# ---- «Земля» -----------------------------------------------------------
	#
	# Continents and regions, both ticking into the same frame: a continent is
	# the blunt one, a region the exact one. The fine choosing happens on the
	# map afterwards, from inside whichever window is open.
	bag_wtp2_register_zone_list = yes
	bag_wtp2_register_region_lists = yes

	# ---- «Расчёты» — two functions, two doors ------------------------------
	#
	# **Neither of them computes anything yet.** They open two windows that
	# carry the ground strip and an empty sheet, which is deliberate: the
	# interface is what is being agreed first, and a pass that does not exist
	# cannot be blamed for what the window shows.
	cmm_register_button_setting = {
		mod_id = bag_wtp2
		setting_id = plan_a
		tab_id = plan
		group_id = open
	}

	cmm_register_button_setting = {
		mod_id = bag_wtp2
		setting_id = plan_b
		tab_id = plan
		group_id = open
	}

	# ---- «Техническая» -----------------------------------------------------
	#
	# One press, everything the mod knows, into `debug.log`, with the headline
	# repeated into `error.log` so a clean log still says the press arrived.
	# **It reads and writes nothing but its own scratch**, so it is safe at any
	# moment. Today it has almost nothing to report -- the ground, and that the
	# mod is alive -- and that is itself the point: it is here from the first
	# build, so nothing is ever built without a place to put a probe.
	cmm_register_button_setting = {
		mod_id = bag_wtp2
		setting_id = diag
		tab_id = tech
		group_id = tech
	}

	# Every picker folded shut, once, the first time this player sees the page:
	# six lists open is a page to scroll rather than a page to read. After this
	# the player's own folding is his.
	bag_wtp2_fold_pickers = yes

	# Every counter the interface prints, created before anything reads it. A
	# script value reading a global that was never set is a blank on screen and
	# a line a frame in `error.log` -- the hardest kind of nothing to diagnose.
	bag_wtp2_init_counters = yes
}

# Scope: country
bag_wtp2_fold_pickers = {
	if = {
		limit = { NOT = { has_variable = bag_wtp2_folded_once } }
		set_variable = { name = bag_wtp2_folded_once value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__continent value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__region_europe value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__region_asia value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__region_africa value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__region_america value = 1 }
		add_to_variable_map = { name = cmm_group_collapsed key = flag:bag_wtp2__zone__region_oceania value = 1 }
	}
}

# **Which button was pressed arrives in `var:cmf_callback`**, as the setting's
# full key. A Mod Menu button carries nothing else -- no argument, no scope --
# which is the whole reason anything that needs one is a control inside a window
# instead.
# Scope: country
bag_wtp2_on_button = {
	if = {
		limit = { var:cmf_callback = flag:bag_wtp2__plan_a }
		bag_wtp2_open_plan_a = yes
	}

	if = {
		limit = { var:cmf_callback = flag:bag_wtp2__plan_b }
		bag_wtp2_open_plan_b = yes
	}

	if = {
		limit = { var:cmf_callback = flag:bag_wtp2__diag }
		bag_wtp2_diag = yes
	}
}
