# Проход по городским правам: одно нажатие, выбранная земля, ничего больше.
#
# **Считается только выбранная земля и только по нажатию.** Карта CM считает
# весь мир на загрузке; владелец отказался от CM ровно из-за цены. Здесь проход
# ходит по `bag_wtp2_picked_locations` и ни по чему другому, и мод не делает
# ничего периодически вовсе.
#
# **Определение провинции считается один раз.** Покрытие -- свойство всего
# определения, поэтому первая же его локация из выбора считает его целиком и
# метит `bag_wtp2_tr_done` на всех локациях определения; остальные выбранные
# локации того же определения проход пропускает.

# «Текущая эпоха»: методы, чьё здание держава уже строит и которое ей ещё не
# заменили.
# Scope: country
bag_wtp2_tr_run_now = {
	set_global_variable = { name = bag_wtp2_tr_end value = 0 }
	bag_wtp2_tr_run = yes
}

# «Последняя эпоха»: методы, чьё здание не заменяет ничто. Продвижения не
# спрашиваются вовсе -- план это цель, а не опись того, что уже открыто.
# Scope: country
bag_wtp2_tr_run_end = {
	set_global_variable = { name = bag_wtp2_tr_end value = 1 }
	bag_wtp2_tr_run = yes
}

# Scope: country
bag_wtp2_tr_run = {
	bag_wtp2_tr_clear = yes
	change_global_variable = { name = bag_wtp2_tr_runs add = 1 }
	bag_wtp2_tr_set_avail = yes

	every_in_global_list = {
		variable = bag_wtp2_picked_locations
		limit = { NOT = { has_variable = bag_wtp2_tr_done } }

		province_definition = {
			bag_wtp2_tr_definition = yes
			every_location_in_province_definition = {
				limit = { is_ownable = yes }
				set_variable = { name = bag_wtp2_tr_done value = 1 }
				add_to_global_variable_list = { name = bag_wtp2_tr_locs target = this }
				change_global_variable = { name = bag_wtp2_tr_loc_count add = 1 }
				bag_wtp2_tr_score_location = yes
			}
		}

		# Строка окна -- одна на определение провинции, и ей достаётся та
		# локация, с которой определение посчитали.
		add_to_global_variable_list = { name = bag_wtp2_tr_raw target = this }
	}

	bag_wtp2_tr_sort = yes
}

# **Метки снимаются с тех локаций, которые их получили**, а не со всей земли:
# список `_locs` для того и держится. Покрытия на провинциях не чистятся --
# следующий проход их перезапишет, а прочесть их может только локация, у которой
# есть `bag_wtp2_tr_best_idx`, и он снимается здесь.
# Scope: country
bag_wtp2_tr_clear = {
	every_in_global_list = {
		variable = bag_wtp2_tr_locs
		remove_variable = bag_wtp2_tr_done
		remove_variable = bag_wtp2_tr_best_idx
		remove_variable = bag_wtp2_tr_best_mil
		remove_variable = bag_wtp2_tr_open
	}
	clear_global_variable_list = bag_wtp2_tr_locs
	clear_global_variable_list = bag_wtp2_tr_raw
	clear_global_variable_list = bag_wtp2_tr_rows
	set_global_variable = { name = bag_wtp2_tr_loc_count value = 0 }
	set_global_variable = { name = bag_wtp2_tr_row_count value = 0 }
}

# Строки по убыванию счёта. `order_by` считается в скоупе самой строки, а
# `bag_wtp2_tr_order` -- это минус счёт, поэтому по возрастанию идёт лучшее
# первым.
#
# **Потолок в 200 строк -- это окно, а не расчёт.** Посчитано будет всё, что
# выбрано; в список попадут двести лучших определений провинций.
# Scope: country
bag_wtp2_tr_sort = {
	ordered_in_global_list = {
		variable = bag_wtp2_tr_raw
		order_by = bag_wtp2_tr_order
		max = 200
		check_range_bounds = no
		add_to_global_variable_list = { name = bag_wtp2_tr_rows target = this }
		change_global_variable = { name = bag_wtp2_tr_row_count add = 1 }
	}
}

# Строка провинции разворачивается в свои локации и сворачивается обратно.
# Scope: location
bag_wtp2_tr_toggle_open = {
	if = {
		limit = { has_variable = bag_wtp2_tr_open }
		remove_variable = bag_wtp2_tr_open
	}
	else = {
		set_variable = { name = bag_wtp2_tr_open value = 1 }
	}
}
