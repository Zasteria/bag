# Открыть и закрыть окно -- и почему это эффект, а не свойство окна.
#
# **Скрипт не умеет трогать систему переменных интерфейса**, а кнопка на
# странице мода -- это скриптовый эффект. Поэтому окно смотрит на переменную
# игрока (`bag_wtp2_plan_a_open`), а кнопка эту переменную ставит.
#
# **Два окна исключают друг друга.** Не из вкусовщины: окно -- это scripted
# widget, он никогда не разбирается, и спрятанное окно продолжает каждый кадр
# считать свои `visible`, `enabled` и `datacontext`. Пока внутри пусто, это
# ничего не стоит; правило записано сейчас, чтобы к нему не возвращаться, когда
# внутри появятся ряды.

# Scope: country
bag_wtp2_open_plan_a = {
	remove_variable = bag_wtp2_plan_b_open
	set_variable = { name = bag_wtp2_plan_a_open value = 1 }
}

# Scope: country
bag_wtp2_open_plan_b = {
	remove_variable = bag_wtp2_plan_a_open
	set_variable = { name = bag_wtp2_plan_b_open value = 1 }
}

# Scope: country
bag_wtp2_close_plan_a = {
	remove_variable = bag_wtp2_plan_a_open
}

# Scope: country
bag_wtp2_close_plan_b = {
	remove_variable = bag_wtp2_plan_b_open
}
