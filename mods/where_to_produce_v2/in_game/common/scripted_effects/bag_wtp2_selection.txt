# Выбор земли. Первая функция мода и единственная, которая здесь уже работает.
#
# **Выбор записан дважды, и писать его имеют право только `bag_wtp2_pick` и
# `bag_wtp2_drop`:**
#
#   `bag_wtp2_picked`            переменная на локации -- то, что красит карта
#   `bag_wtp2_picked_locations`  глобальный список -- то, что обходит расчёт
#
# Переменная нужна потому, что `map_color` режима карты -- это триггерный блок в
# скоупе локации: переменная на локации становится цветом на карте без ничего
# между. Список нужен потому, что дешёвого «каждая локация с переменной X» в
# скрипте нет, а обойти мир, чтобы их найти, дороже, чем держать список.
# **Две записи одного факта расходятся в ту же секунду, как их напишет кто-то
# третий.**

# Scope: location
bag_wtp2_pick = {
	set_variable = { name = bag_wtp2_picked value = 1 }
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = bag_wtp2_picked_locations
					target = this
				}
			}
		}
		add_to_global_variable_list = { name = bag_wtp2_picked_locations target = this }
		change_global_variable = { name = bag_wtp2_picked_count add = 1 }
	}
}

# Scope: location
bag_wtp2_drop = {
	remove_variable = bag_wtp2_picked
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = bag_wtp2_picked_locations
				target = this
			}
		}
		remove_list_global_variable = { name = bag_wtp2_picked_locations target = this }
		change_global_variable = { name = bag_wtp2_picked_count subtract = 1 }
	}
}

# **Каждый пикер переключает, а не добавляет.** Второе нажатие на ту же землю
# отменяет первое, поэтому «готово» -- это просто закрыть панель, и сохранять
# нечего.

# Scope: location
bag_wtp2_toggle_one = {
	if = {
		limit = { has_variable = bag_wtp2_picked }
		bag_wtp2_drop = yes
	}
	else = {
		bag_wtp2_pick = yes
	}
}

# Scope: province
bag_wtp2_toggle_province_scope = {
	if = {
		limit = {
			any_location_in_province = {
				is_ownable = yes
				NOT = { has_variable = bag_wtp2_picked }
			}
		}
		every_location_in_province = {
			limit = { is_ownable = yes }
			bag_wtp2_pick = yes
		}
	}
	else = {
		every_location_in_province = {
			limit = { has_variable = bag_wtp2_picked }
			bag_wtp2_drop = yes
		}
	}
}

# Scope: market
#
# **Рынок -- не земля, поэтому `bag_wtp2_in_zone` у него не спрашивают:**
# отмеченные континенты -- это рамка, в которой *ищут*, а рынок, названный по
# имени, сам себе рамка. Его локации фильтруются `is_ownable`, как у всех
# остальных, что и убирает морские клетки, которые рынок наверняка содержит.
bag_wtp2_toggle_market_scope = {
	if = {
		limit = {
			any_location_in_market = {
				is_ownable = yes
				NOT = { has_variable = bag_wtp2_picked }
			}
		}
		every_location_in_market = {
			limit = { is_ownable = yes }
			bag_wtp2_pick = yes
		}
	}
	else = {
		every_location_in_market = {
			limit = { has_variable = bag_wtp2_picked }
			bag_wtp2_drop = yes
		}
	}
}

# «Моя земля»: всё, что держава владеет, разом.
#
# Обычная кнопка, а не действие на карте: щёлкать не по чему, ответ уже известен.
# Рамка отмеченных континентов и регионов действует и здесь -- иначе кнопка
# набрала бы колонии на другом конце света.
# Scope: country
bag_wtp2_pick_my_land = {
	every_owned_location = {
		limit = { bag_wtp2_is_candidate = yes }
		bag_wtp2_pick = yes
	}
}

# «Очистить»: везде, а не только внутри того, что сейчас на экране.
#
# **Это правило первого мода, и оно стоило ему прогона.** Выбор, сделанный под
# одной рамкой, переживает смену рамки и молча расширяет следующий расчёт.
# Scope: country
bag_wtp2_clear_all = {
	every_in_global_list = {
		variable = bag_wtp2_picked_locations
		remove_variable = bag_wtp2_picked
	}
	clear_global_variable_list = bag_wtp2_picked_locations
	set_global_variable = { name = bag_wtp2_picked_count value = 0 }
}

# Галочка в списке континентов или регионов. Ничего не обходится: меняется
# только то, что пикеры предлагают и что красит карта.
# Scope: country
bag_wtp2_zone_changed = {
	bag_wtp2_rebuild_zone = yes
	bag_wtp2_rebuild_regions = yes
}

# Каждое число, которое печатает интерфейс, существует до того, как его прочтут.
# Scope: country
bag_wtp2_init_counters = {
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_picked_count } }
		set_global_variable = { name = bag_wtp2_picked_count value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_zone_count } }
		set_global_variable = { name = bag_wtp2_zone_count value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_region_count } }
		set_global_variable = { name = bag_wtp2_region_count value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_diag_runs } }
		set_global_variable = { name = bag_wtp2_diag_runs value = 0 }
	}
	# Счётчики прохода по городским правам. Окно печатает все три с момента,
	# когда его можно открыть, то есть до первого нажатия.
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_tr_runs } }
		set_global_variable = { name = bag_wtp2_tr_runs value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_tr_row_count } }
		set_global_variable = { name = bag_wtp2_tr_row_count value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_tr_loc_count } }
		set_global_variable = { name = bag_wtp2_tr_loc_count value = 0 }
	}
	# Какой эпохой считает проход. Существует заранее, потому что ворота эпохи
	# спрашивают его сравнением, а сравнение с несуществующей глобалкой -- это
	# тихое «нет» для всех двухсот методов разом.
	if = {
		limit = { NOT = { has_global_variable = bag_wtp2_tr_end } }
		set_global_variable = { name = bag_wtp2_tr_end value = 0 }
	}
}
