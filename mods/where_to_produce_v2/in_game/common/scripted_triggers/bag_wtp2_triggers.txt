#
# Scope: location
bag_wtp2_is_candidate = {
	# **Морские клетки не имеют провинции**, а всякий расчёт выгоды ходит по
	# провинции. Это защита, а не опрятность -- и `is_ownable` -- вся она
	# целиком: «не море, не озеро, не непроходимое», то есть ровно та земля, на
	# которой может стоять здание.
	is_ownable = yes

	bag_wtp2_in_zone = yes

	# **И больше ничего.** «Только там, где здание вообще может стоять» стояло

	# владеет, а `can_build_building` отвечает про локацию как она есть, и
	# невладение он проваливает. Фильтр опустошал таблицу ровно для того случая,
	# ради которого мод и существует.
