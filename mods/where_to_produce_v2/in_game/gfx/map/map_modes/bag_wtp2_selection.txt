# Выбор, нарисованный на карте.
#
# `map_color` -- триггерный блок в скоупе локации, и это делает режим дешёвым:
# выбор и так переменная на локации, так что покраска -- один `has_variable` и
# больше ничего.
#
# **`category = economy`, индекс 3, и это не свободный выбор.** Категория
# geography перечислена движком как 0-3 явно, и её третий индекс занимает
# `where_to_produce` первой версии (а также Advanced Auto Build). В economy
# заняты 0 и 1 -- Construction Manager, -- и 2 -- план первой версии. Два мода
# на одном индексе сталкиваются. **Держать этот список в комментарии дешевле,
# чем ловить столкновение в игре.**
bag_wtp2_selection = {
	map_color = {
		if = {
			limit = { has_variable = bag_wtp2_picked }
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			# Внутри рамки, но не выбрано -- земля, из которой выбирают, чтобы
			# выбранное было видно на фоне.
			limit = { bag_wtp2_in_zone = yes }
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else = { value = define:NMapColors|DEFAULT_COLOR }
	}

	tooltip_key = {
		if = {
			limit = { has_variable = bag_wtp2_picked }
			value = bag_wtp2_map_picked_tt
		}
		else_if = {
			limit = { bag_wtp2_in_zone = yes }
			value = bag_wtp2_map_browsable_tt
		}
		else = { value = bag_wtp2_map_outside_tt }
	}

	legend_key = {
		desc = bag_wtp2_map_legend_picked
		color = define:NMapColors|MAP_COLOR_HIGH
	}
	legend_key = {
		desc = bag_wtp2_map_legend_browsable
		color = define:NMapColors|MAP_COLOR_LOW
	}

	small_map_names = location
	medium_map_names = province
	large_map_names = area
	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location
	category = economy
	index = 3
	allow_allocate_hotkey = yes
	flatmap_behaviour = Always
	fill_in_impassable = no
	use_fow = no
	enable_rivers = yes
	map_markers = { all = no }

	gradient_parameters = {
		zoom_step = 2
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	flatmap_gradient_parameters = {
		zoom_step = 12
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	color_refresh_counters = { Day }
	color_and_names_refresh_counters = { LocationOwnerChanged }
}
