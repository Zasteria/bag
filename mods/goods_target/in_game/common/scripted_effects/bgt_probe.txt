# What this mod knows, written where it can be checked.
#
# Still nothing is built and nothing is subsidised.
#
# Everything numeric goes into country variables that script values read back
# for the interface, and the log gets plain lines with no arguments at all. That
# split is deliberate: the previous round logged a count through
# `cmf_log_value = { value = var:... }`, and a CMF macro pastes its arguments
# verbatim -- the same shape that kills a CMM list on `item = var:x`. Whether
# that was the fault or the setting read was, neither can swallow this version.
#
# Both lists are counted, because the two are kept apart: CMM clears the list it
# builds into, so a merged one would hold only whichever was built last.
#
# Scope: country
bgt_probe_readings = {
	if = {
		limit = { NOT = { has_variable = bgt_pulse_count } }
		set_variable = { name = bgt_pulse_count value = 0 }
	}
	change_variable = { name = bgt_pulse_count add = 1 }

	set_variable = { name = bgt_watched_count value = 0 }
	set_variable = { name = bgt_subsidised_count value = 0 }
	bgt_count_ticks = { list = construction }
	bgt_count_ticks = { list = other }

	if = {
		limit = { bgt_logging_on = yes }
		cmf_log = { action = bgt_log_pulse }
		bgt_log_ticks = { list = construction }
		bgt_log_ticks = { list = other }
	}
}

# Scope: country. $list$ is one of the two goods lists.
bgt_count_ticks = {
	every_in_list = {
		variable = bgt_$list$_build
		change_variable = { name = bgt_watched_count add = 1 }
	}
	every_in_list = {
		variable = bgt_$list$_subsidise
		change_variable = { name = bgt_subsidised_count add = 1 }
	}
}

# Scope: country. $list$ is one of the two goods lists.
bgt_log_ticks = {
	every_in_list = {
		variable = bgt_$list$_build
		save_scope_as = cmf_log_arg2
		cmf_log_with_scope_arg = { action = bgt_log_good }
	}
	every_in_list = {
		variable = bgt_$list$_subsidise
		save_scope_as = cmf_log_arg2
		cmf_log_with_scope_arg = { action = bgt_log_subsidise }
	}
}
