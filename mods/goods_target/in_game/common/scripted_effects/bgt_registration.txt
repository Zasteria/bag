# Settings, registered on CMF's mod registration hook.
#
# Only setting types with CMM auto-apply are used here: bool and numeric both
# reach cmf_on_callback by themselves. The goods list this mod is really about
# is a CMM list setting, which needs an _on_changed scripted GUI of its own and
# is deliberately not part of this first step -- see the README.
# Scope: country
bgt_register = {
	cmf_register_mod = { mod_id = bgt }

	# The discount to aim for, as a positive percentage. The engine credits at
	# most 33%, so that is both the default and the ceiling.
	cmm_register_numeric_setting = {
		mod_id = bgt
		setting_id = target_discount
		tab_id = main
		group_id = goal
		default_value = 33
		min_value = 0
		max_value = 33
		step_value = 1
	}

	# Writes one line a year into CMF's log with the readings below, so the
	# numbers can be compared against the game's own construction tooltip.
	cmm_register_bool_setting = {
		mod_id = bgt
		setting_id = log_readings
		tab_id = main
		group_id = goal
		default_value = 1
	}
}
