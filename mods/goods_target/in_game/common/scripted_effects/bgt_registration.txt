# Settings, registered on CMF's mod registration hook.
#
# There is no single target discount any more: the target is a field of each
# row, because 33% off lumber and 33% off cannons are different jobs and a
# player wants them set apart.
# Scope: country
bgt_register = {
	cmf_register_mod = { mod_id = bgt }

	# Writes a line a month into CMF's log naming what is ticked. The readings
	# themselves are on screen, in the lists, so this is for watching the mod
	# rather than for reading prices.
	cmm_register_bool_setting = {
		mod_id = bgt
		setting_id = log_readings
		tab_id = main
		group_id = goal
		default_value = 1
	}

	bgt_register_lists = yes
	bgt_rebuild_lookups = yes
}
