# CMF extension points. Both merge with whatever else declares them, so
# nothing of Construction Manager's own wiring is displaced.

# New game, save load and country transfer, once per human country.
# Scope: country
cmf_on_mod_registration = {
	on_actions = {
		bgt_on_registration
	}
}

# Scope: country
bgt_on_registration = {
	effect = {
		bgt_register = yes
	}
}

# Monthly, human countries only -- prices move every month, so a yearly reading
# says nothing about what the mod would be reacting to. This is also the pulse
# Construction Manager's own dispatcher runs on, which is where the building
# half will have to live.
# Scope: country
cmf_monthly_human_country_pulse = {
	on_actions = {
		bgt_monthly_probe
	}
}

# Scope: country
#
# The probe runs unconditionally and decides inside what to log. Gating the
# whole thing on a setting read was how the last round could fail without
# leaving a trace: if that read was the fault, nothing ran, and a mod that does
# nothing looks exactly like a mod that was never called.
bgt_monthly_probe = {
	effect = {
		bgt_probe_readings = yes
	}
}
