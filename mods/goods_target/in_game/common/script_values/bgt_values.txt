# What the mod has actually done, readable from the interface.
#
# These exist because the log let us down once already: a monthly line that
# never appeared could have meant the pulse was not firing, or that the setting
# read was failing, or that CMF's log would not render what we handed it, and
# there was no way to tell which. A number on screen answers the first two
# without involving the log at all.
#
# Each guards its variable: reading one that was never set is an error, and the
# whole point of these is to work before anything else does.
# Root is the country.

# How many monthly pulses this mod has seen. Zero after a month has passed means
# the on_action never reached us.
bgt_pulses = {
	value = 0
	if = {
		limit = { has_variable = bgt_pulse_count }
		add = var:bgt_pulse_count
	}
}

# How many goods were ticked to build, as counted by the last pulse -- so this
# is the tick as script sees it, not as the widget drew it.
bgt_watched = {
	value = 0
	if = {
		limit = { has_variable = bgt_watched_count }
		add = var:bgt_watched_count
	}
}

# The same for the subsidy column.
bgt_subsidised = {
	value = 0
	if = {
		limit = { has_variable = bgt_subsidised_count }
		add = var:bgt_subsidised_count
	}
}
