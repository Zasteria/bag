# Filter entry points. Neither panel hands its filter more than the object being
# tested -- a building_type filter gets only `root`, and a location filter gets
# `root` plus `scope:target` -- so everything else comes from global variables
# that the probe widgets in the two panels write.
#
# Each panel offers two chips rather than one chip and a setting. The province
# variant matches the shovel badge the game draws, since the bonus comes from
# raw materials anywhere in the province; the location variant asks the stricter
# question of whether this very location works the material itself.

# Buildings panel of one location: which building types gain here.
# Scope: building_type
bag_rgo_has_local_bonus = {
	has_global_variable = bag_rgo_view_location
	global_var:bag_rgo_view_location = {
		province = {
			any_location_in_province = {
				bag_rgo_location_feeds_root = yes
			}
		}
	}
}

# Same, but only the location on screen counts.
# Scope: building_type
bag_rgo_has_local_bonus_here = {
	has_global_variable = bag_rgo_view_location
	global_var:bag_rgo_view_location = {
		bag_rgo_location_feeds_root = yes
	}
}

# Build panel of one building type: which locations it would gain in.
# Scope: location
bag_rgo_location_suits_build_type = {
	has_global_variable = bag_rgo_build_type
	province = {
		any_location_in_province = {
			bag_rgo_location_feeds_build_type = yes
		}
	}
}

# Same, but the material has to be worked in the location itself.
# Scope: location
bag_rgo_location_suits_build_type_here = {
	has_global_variable = bag_rgo_build_type
	bag_rgo_location_feeds_build_type = yes
}
