# A filter is handed the object it tests and little else, so the two panels
# park what their filter is missing in global variables, which any scope can
# reach. Each pair below is one probe: a query that tells the probe whether the
# stored value is already current, and the write that updates it.
#
# The build panel parks the building type being placed. The location the
# buildings panel is showing is parked by the shared probe instead --
# `bag_shared_view_location.txt`, which this mod and `where_to_produce` ship
# byte-identical because both would otherwise override the same panel.

# Scope: country
bag_rgo_build_type_is_current = {
	scope = country

	is_shown = {
		has_global_variable = bag_rgo_build_type
		global_var:bag_rgo_build_type = scope:bag_rgo_type
	}
	effect = { }
}

# Scope: country
bag_rgo_store_build_type = {
	scope = country

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		set_global_variable = { name = bag_rgo_build_type value = scope:bag_rgo_type }
	}
}
