# **Общая проба: какую локацию показывает панель производства.**
#
# Этот файл `rgo_bonus_filter` и `where_to_produce` возят побайтово одинаковым, и
# вместе с ним -- `gui/location_production_lateralview.gui`. Обоим нужен один и
# тот же факт, обоим пришлось бы перекрыть одну и ту же панель, и тот, кто
# загрузился последним, молча унёс бы пробу другого. Поэтому проба одна, глобалка
# одна (`bag_view_location`), а имена не принадлежат ни одному из модов: любой из
# них в одиночку работает, и оба вместе работают в любом порядке загрузки.
#
# **Править в обеих копиях или не править вовсе** -- `tools/check_script.py`
# роняет сборку, когда они разошлись.
#
# Фильтру со `scope = building_type` панель даёт `root` -- сам тип здания -- и
# больше ничего, а `LocationProductionView` разрешается только внутри своей
# панели. Значит локацию туда донести можно только глобалкой, и записать её
# может только виджет, стоящий в самой панели.

# Scope: country
bag_view_location_is_current = {
	scope = country

	is_shown = {
		has_global_variable = bag_view_location
		global_var:bag_view_location = scope:bag_view_loc
	}
	effect = { }
}

# Scope: country
bag_store_view_location = {
	scope = country

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		set_global_variable = { name = bag_view_location value = scope:bag_view_loc }
	}
}
