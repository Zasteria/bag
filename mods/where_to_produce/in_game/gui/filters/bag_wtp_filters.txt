# **Шаг 7: план в чужом окне, и для этого не нужно чужое окно.**
#
# Панель зданий локации набирает свои галочки через
# `WithFilterTags('building')`, а не выписывает их: любой мод, положивший сюда
# файл со `scope = building_type` и `tag = building`, получает свою галочку в том
# же ряду. Это верно и для ванильной панели, и для той, которой её перекрыл
# Glorp UI, -- значит перекрывать `glorpUI_build_location_lateralview.gui` (2876
# строк, ломается на каждом их обновлении) не нужно вовсе.
#
# Тексты галочек -- ключи `search_filter_<ключ>_name` и `_desc`.
#
# Группа 10, потому что 9 занята чипами `rgo_bonus_filter`: разные группы --
# разные подложки, и в ряду видно, чьё что. `exclusive_group = no`: галочки не
# спорят ни между собой, ни с чужими.

# Только те здания, которыми этот мод вообще умеет планировать. Список известен
# на сборке -- одно сравнение на тип, ни одного чтения с карты.
bag_wtp_planned = {
	scope = building_type
	tag = building
	group = 10
	exclusive_group = no
	trigger = { bag_wtp_type_is_planned = yes }
}

# Только те здания, которые план ставит **в этой локации**. Локацию приносит
# общая проба `bag_view_location` (`common/scripted_guis/bag_shared_view_location.txt`).
bag_wtp_plan_here = {
	scope = building_type
	tag = building
	group = 10
	exclusive_group = no
	trigger = { bag_wtp_type_in_plan_here = yes }
}
