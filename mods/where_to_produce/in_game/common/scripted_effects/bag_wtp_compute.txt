# The pass that answers the question, run when the player presses the button.
#
# Nothing here is on a pulse. Each step logs nothing of its own, but every one
# of them leaves a number the button's own description prints, which is what
# makes a failure locatable from outside: an effect that merely does nothing
# logs nothing at all.
#
# Scope: country
bag_wtp_compute = {
	bag_wtp_sync = yes
	bag_wtp_apply_method = yes
	bag_wtp_count_wanted = yes
	bag_wtp_collect_candidates = yes
	bag_wtp_clear_rows = yes
	bag_wtp_fill_rows = yes
	bag_wtp_trim_rows = yes

	# One line, no arguments. The counts are on screen instead, in the button's
	# own description, read back through script values -- `goods_target` learnt
	# that the hard way: a CMF macro pastes its arguments verbatim, which is the
	# same shape that kills a CMM list on `item = var:x`, and a log call that
	# swallows itself takes the rest of the effect with it. This line says only
	# that the pass ran, which is the one thing the log is needed for.
	cmf_log = { action = bag_wtp_log_run }
}

# What gets ranked.
#
# The locations picked in the selection window, when there are any; otherwise
# every land location of the ticked regions. The fallback is what makes the
# regions usable on their own — tick Northern Germany, press the button, and the
# whole region is answered for without opening the window at all. Picking
# anything narrows it from there.
# Scope: country
bag_wtp_collect_candidates = {
	clear_global_variable_list = bag_wtp_candidates
	set_global_variable = { name = bag_wtp_candidate_count value = 0 }

	if = {
		limit = { global_var:bag_wtp_picked_count > 0 }
		every_in_global_list = {
			variable = bag_wtp_picked_locations
			limit = { bag_wtp_is_candidate = yes }
			add_to_global_variable_list = { name = bag_wtp_candidates target = this }
			change_global_variable = { name = bag_wtp_candidate_count add = 1 }
		}
	}
	else = {
		every_in_global_list = {
			variable = bag_wtp_zone_locations
			limit = { bag_wtp_is_candidate = yes }
			add_to_global_variable_list = { name = bag_wtp_candidates target = this }
			change_global_variable = { name = bag_wtp_candidate_count add = 1 }
		}
	}
}

# Scope: country
bag_wtp_on_button = {
	if = {
		limit = { var:cmf_callback = flag:bag_wtp__run }
		bag_wtp_compute = yes
	}

	# Opening the window is a script effect setting a variable the window watches,
	# because a Mod Menu button cannot reach the interface's own variable system.
	if = {
		limit = { var:cmf_callback = flag:bag_wtp__select }
		bag_wtp_rebuild_browse = yes
		bag_wtp_open_window_effect = yes
	}

	# The two pickers auto-apply, so a change to either arrives here rather than
	# through a list's `_on_changed`. Re-running the whole pass on a picker change
	# would walk the regions on every click of the dropdown, so this only copies
	# the value out; the table is rebuilt when the button is pressed.
	if = {
		limit = {
			OR = {
				var:cmf_callback = flag:bag_wtp__method
				var:cmf_callback = flag:bag_wtp__buildable_only
			}
		}
		bag_wtp_sync = yes
	}
}

# The same thing the window's own scripted GUI does, reachable from an effect.
# Scope: country
bag_wtp_open_window_effect = {
	set_variable = { name = bag_wtp_open value = 1 }
}
