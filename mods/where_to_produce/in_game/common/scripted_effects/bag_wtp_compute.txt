# The pass that answers the question.
#
# Four steps. Nothing is on a pulse; each leaves a number the button's own
# description prints, which is what makes a failure locatable from outside, since
# an effect that merely does nothing logs nothing at all.
#
# Scope: country
bag_wtp_compute = {
	if = {
		# An urban right: a bundle of two or three goods, its own pass and its own
		# window. A right and a good are two answers to one question and only one
		# of them can be ticked, so this branch and the next never both run.
		limit = { var:bag_wtp_right_index > 0 }
		bag_wtp_collect_candidates = yes
		bag_wtp_rebuild_browse = yes
		bag_wtp_score_right = yes
		bag_wtp_clear_rows = yes
		bag_wtp_fill_rows_right = yes
	}
	else_if = {
		limit = { var:bag_wtp_good_index = 0 }
		# Nothing to answer for. Saying so beats ranking every location in the
		# world by a score of zero.
		bag_wtp_clear_rows = yes
	}
	else = {
		bag_wtp_collect_candidates = yes
		# The province count beside the buttons, rebuilt here rather than only
		# where a pick happens: it went stale between the eleventh run's second
		# and third screenshots -- «44 лок. в 6 пров.» became «44 лок. в 0 пров.»
		# with the selection untouched -- and a count that disagrees with the map
		# is a bug report about the selection every time. It walks the picked
		# locations, which this pass has walked already.
		bag_wtp_rebuild_browse = yes
		bag_wtp_score_candidates = yes
		bag_wtp_clear_rows = yes
		bag_wtp_fill_rows = yes
	}

	# One line, no arguments. The counts are on screen instead, in the button's
	# own description, read back through script values -- `goods_target` learnt
	# that the hard way: a CMF macro pastes its arguments verbatim, which is the
	# same shape that kills a CMM list on `item = var:x`, and a log call that
	# swallows itself takes the rest of the effect with it.
	cmf_log = { action = bag_wtp_log_run }
}

# What gets ranked.
#
# The locations picked on the map, when there are any; otherwise every land
# location of the ticked continents. The fallback is what makes the continents
# usable on their own — tick Europe, press the button, and the whole of it is
# answered for without opening the picker at all. Picking narrows it from there.
# Scope: country
bag_wtp_collect_candidates = {
	clear_global_variable_list = bag_wtp_candidates
	set_global_variable = { name = bag_wtp_candidate_count value = 0 }

	if = {
		limit = { global_var:bag_wtp_picked_count > 0 }
		every_in_global_list = {
			variable = bag_wtp_picked_locations
			limit = { bag_wtp_is_candidate = yes }
			add_to_global_variable_list = { name = bag_wtp_candidates target = this }
			change_global_variable = { name = bag_wtp_candidate_count add = 1 }
		}
	}
	else_if = {
		# A ticked region is the finer frame and wins over its continent.
		limit = { global_var:bag_wtp_region_count > 0 }
		bag_wtp_collect_regions = yes
	}
	else = {
		bag_wtp_collect_zone = yes
	}
}

# Scope: country
bag_wtp_on_button = {
	if = {
		limit = { var:cmf_callback = flag:bag_wtp__run }
		bag_wtp_compute = yes
		# Straight into the answer: the window is the only place it is shown.
		bag_wtp_open_result_window_effect = yes
	}

	if = {
		limit = { var:cmf_callback = flag:bag_wtp__show }
		bag_wtp_open_result_window_effect = yes
	}

	# The tick auto-applies and arrives here; the lists do not, because a list
	# reaches nothing but its own `_on_changed`.
	if = {
		limit = {
			OR = {
				var:cmf_callback = flag:bag_wtp__buildable_only
				var:cmf_callback = flag:bag_wtp__any_method
				var:cmf_callback = flag:bag_wtp__rank_by_end
			}
		}
		bag_wtp_sync = yes
		bag_wtp_refresh_goods = yes
		bag_wtp_refresh_rights = yes
	}

	# Which column the ranking sorts on is the one tick that changes the answer
	# already on screen rather than the next one, so it re-ranks where the others
	# do not. `bag_wtp_recompute_live` costs nothing against a closed window: it
	# is the same guard the map pickers run through.
	if = {
		limit = { var:cmf_callback = flag:bag_wtp__rank_by_end }
		bag_wtp_recompute_live = yes
	}

}

# Every change of the borders re-ranks, so the window answers as the plan is
# drawn rather than after another press. Only while it is open: the pass walks
# every candidate, and there is no reason to pay for it against a closed window.
#
# `bag_wtp_live_runs` counts the times this got past its own guard, and the
# window prints it. It is there because the guard is the exact place the map
# pickers used to fall over: reached in the wrong scope, `has_variable` says no
# and the effect returns having done nothing, which logs nothing. A number on
# screen that does not move after a pick says so in one glance.
# Scope: country
bag_wtp_recompute_live = {
	if = {
		limit = {
			OR = {
				has_variable = bag_wtp_result_open
				has_variable = bag_wtp_right_open
			}
		}
		change_global_variable = { name = bag_wtp_live_runs add = 1 }
		bag_wtp_compute = yes
		bag_wtp_show_results = yes
	}
}

# The results window's list, filled on opening and emptied on closing.
#
# `bag_wtp_ranked` is the answer and survives; `bag_wtp_results` is what the
# window repeats over. They are two lists rather than one because the window is
# a scripted widget and never comes down: an item stays instantiated for as long
# as it is in the datamodel, whether the window is on screen or not, so the copy
# exists only while the window is open.
#
# **The copy is ordered, and that is the whole of it.** It used to be
# `every_in_global_list`, and an unordered iterator promises nothing about the
# order it hands things back in -- the window draws its rows in the order the
# list holds them, so a ranking done in `bag_wtp_fill_rows` arrived on screen
# shuffled back into roughly map order. Each row carries the rank the pass gave
# it, and the copy sorts on that.
# Scope: country
bag_wtp_show_results = {
	clear_global_variable_list = bag_wtp_results
	clear_global_variable_list = bag_wtp_right_results

	# One list per window, and the closed one's is empty. Both windows are
	# scripted widgets and neither ever comes down, so a row of the window that
	# is not on screen is a row that is alive and costing anyway -- pointing both
	# at one list would keep fifty rows of each instantiated at all times.
	ordered_in_global_list = {
		variable = bag_wtp_ranked
		order_by = bag_wtp_rank_order
		# The row count the pass itself wrote, rather than a fifty repeated from the
		# generator: the two lists are filled in the same pass and cannot disagree.
		max = bag_wtp_show_found
		check_range_bounds = no
		if = {
			limit = { global_var:bag_wtp_right_index > 0 }
			add_to_global_variable_list = { name = bag_wtp_right_results target = this }
		}
		else = {
			add_to_global_variable_list = { name = bag_wtp_results target = this }
		}
	}
}

# Scope: country
bag_wtp_hide_results = {
	clear_global_variable_list = bag_wtp_results
	clear_global_variable_list = bag_wtp_right_results
}

# The same thing the window's own scripted GUI does, reachable from an effect --
# a Mod Menu button is script, and script cannot reach the interface's variable
# system.
#
# Which of the two windows opens is which question was asked. They share the
# ranked list, the live re-rank and the map pickers, and differ only in what a
# row draws: one method for a good, or one per good of a right's bundle.
# Scope: country
bag_wtp_open_result_window_effect = {
	bag_wtp_show_results = yes
	if = {
		limit = { var:bag_wtp_right_index > 0 }
		remove_variable = bag_wtp_result_open
		set_variable = { name = bag_wtp_right_open value = 1 }
	}
	else = {
		remove_variable = bag_wtp_right_open
		set_variable = { name = bag_wtp_result_open value = 1 }
	}
}
