# The whole interface lives in the Community Mod Framework's Mod Menu: walk down
# region, area, province, and the last list says what to build there. Nothing of
# our own is drawn, so it looks like the rest of the menu and no vanilla file is
# touched.
#
# None of the four lists is registered here. Each is registered by the effect
# that fills it, at exactly the height it needs, and all four of those run at the
# end of this. Registering them at a guessed height first would leave the item
# count and the item list disagreeing on a reload.
#
# Scope: country
wtp_register = {
	cmf_register_mod = { mod_id = wtp }

	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = half
		tab_id = plan
		group_id = display
		default_index = 1
		option_count = 3
	}
	cmm_register_global_dropdown_setting = {
		mod_id = wtp
		setting_id = sort
		tab_id = plan
		group_id = display
		default_index = 2
		option_count = 2
	}
	cmm_register_global_bool_setting = {
		mod_id = wtp
		setting_id = unlocked
		tab_id = plan
		group_id = display
		default_value = 1
	}
	cmm_register_global_bool_setting = {
		mod_id = wtp
		setting_id = filter
		tab_id = plan
		group_id = display
		default_value = 1
	}
	cmm_register_global_numeric_setting = {
		mod_id = wtp
		setting_id = shortlist_size
		tab_id = plan
		group_id = display
		default_value = 8
		min_value = 3
		max_value = 12
		step_value = 1
	}

	wtp_sync_settings = yes
	wtp_remember_display = yes
	wtp_rebuild_all = yes
}

# CMM keeps its values in a variable map that neither a script value nor an
# ordered iterator can read, so everything the ranking needs is mirrored onto
# plain globals.
# Scope: country
wtp_sync_settings = {
	cmm_sync_setting_alias = { setting = wtp__half alias = wtp_half }
	cmm_sync_setting_alias = { setting = wtp__sort alias = wtp_sort }
	cmm_sync_setting_alias = { setting = wtp__unlocked alias = wtp_unlocked }
	cmm_sync_setting_alias = { setting = wtp__filter alias = wtp_filter }
	cmm_sync_setting_alias = { setting = wtp__shortlist_size alias = wtp_rows }
}

# Scope: country
wtp_remember_display = {
	set_global_variable = { name = wtp_seen_half value = global_var:wtp_half }
	set_global_variable = { name = wtp_seen_sort value = global_var:wtp_sort }
	set_global_variable = { name = wtp_seen_unlocked value = global_var:wtp_unlocked }
	set_global_variable = { name = wtp_seen_filter value = global_var:wtp_filter }
	set_global_variable = { name = wtp_seen_rows value = global_var:wtp_rows }
}

# The Display settings arrive here; ticking a row does not, because CMM has no
# auto-apply for list settings and routes those through the `_on_changed`
# scripted GUIs instead.
#
# Every mod's callback leaf runs on every callback, ours included, so the answer
# is only rebuilt when one of ours actually moved.
# Scope: country
wtp_handle_callback = {
	wtp_sync_settings = yes

	if = {
		limit = {
			OR = {
				NOT = { global_var:wtp_seen_half = global_var:wtp_half }
				NOT = { global_var:wtp_seen_sort = global_var:wtp_sort }
				NOT = { global_var:wtp_seen_unlocked = global_var:wtp_unlocked }
				NOT = { global_var:wtp_seen_filter = global_var:wtp_filter }
				NOT = { global_var:wtp_seen_rows = global_var:wtp_rows }
			}
		}
		wtp_remember_display = yes
		wtp_fill_top = yes
	}
}
