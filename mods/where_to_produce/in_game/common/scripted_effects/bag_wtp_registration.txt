# Settings, registered on CMF's mod registration hook.
#
# Everything the player touches is here; everything that depends on the game's
# own data -- how many methods the dropdown offers, which regions exist -- is
# generated beside it, so a patch that adds either is picked up by a rebuild
# rather than by an edit.
#
# Scope: country
bag_wtp_register = {
	cmf_register_mod = { mod_id = bag_wtp }

	# Which building and which of its production methods. One dropdown rather
	# than a chain of pickers: a CMM dropdown has no item cap, and 218 options
	# sorted by the good they produce is a shorter road than good -> building ->
	# method through three lists that each have to be rebuilt when the one above
	# changes.
	bag_wtp_register_method_setting = yes

	# Off by default, and deliberately. The player plans for land he does not own
	# yet, and `can_build_building` answers for the location as it stands today --
	# an unowned location fails it. Leaving this off means the table ranks the
	# ground rather than the current border.
	cmm_register_global_bool_setting = {
		mod_id = bag_wtp
		setting_id = buildable_only
		tab_id = plan
		group_id = what
		default_value = 0
	}

	# Opens the selection window: the ticked regions' provinces, each expandable
	# into its locations. Without touching it the ranking answers for the whole
	# of the ticked regions, so this is narrowing rather than a required step.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = select
		tab_id = plan
		group_id = what
	}

	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = run
		tab_id = plan
		group_id = what
	}

	bag_wtp_register_zone_lists = yes
	bag_wtp_register_result_list = yes

	bag_wtp_init_counters = yes
	bag_wtp_sync = yes
	bag_wtp_rebuild_zone = yes
	bag_wtp_mark_zone = yes
	bag_wtp_drop_browse = yes
	bag_wtp_clear_rows = yes
	bag_wtp_trim_rows = yes
}

# Copy the two settings CMM keeps in its own variable map out to plain variables
# the rest of this mod can read. A dropdown and a bool auto-apply and reach
# `cmf_on_callback`, so this runs from there as well as at registration.
# Scope: country
bag_wtp_sync = {
	cmm_sync_setting_alias = { setting = bag_wtp__method alias = bag_wtp_method_index }
	cmm_sync_bool_alias = { setting = bag_wtp__buildable_only alias = bag_wtp_only_buildable }
}
