# Settings, registered on CMF's mod registration hook.
#
# Everything the player touches is here; everything that depends on the game's
# own data -- which goods exist, which buildings make them, what each method is
# worth -- is generated beside it, so a patch that adds any of those is picked up
# by a rebuild rather than by an edit.
#
# Scope: country
bag_wtp_register = {
	cmf_register_mod = { mod_id = bag_wtp }

	# What you want made, in the two piles goods come in. Lists rather than
	# dropdowns: CMF handles a dropdown option click through
	# `CMM_MarkDropdownSelection_<index>` and defines exactly twenty of them, so
	# the twenty-first option onwards silently keeps the old selection.
	bag_wtp_register_good_raw_list = yes
	bag_wtp_register_good_made_list = yes

	# Where to look. Continents rather than the world's seventy-seven regions:
	# the fine choosing happens on the map afterwards, and a region list six
	# groups deep was a worse way to say "Europe".
	bag_wtp_register_zone_list = yes
	bag_wtp_register_region_lists = yes

	# Off by default, and deliberately. The player plans for land he does not own
	# yet, and `can_build_building` answers for the location as it stands today --
	# an unowned location fails it. Leaving this off means the table ranks the
	# ground rather than the current border.
	cmm_register_global_bool_setting = {
		mod_id = bag_wtp
		setting_id = buildable_only
		tab_id = plan
		group_id = what
		default_value = 0
	}

	# Off by default: a recommendation three ages away is not a recommendation.
	# With it on, the table scores every method in the game, which is what to use
	# when planning several ages ahead rather than this decade.
	cmm_register_global_bool_setting = {
		mod_id = bag_wtp
		setting_id = any_method
		tab_id = plan
		group_id = what
		default_value = 0
	}

	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = run
		tab_id = plan
		group_id = what
	}

	# The answer in a window of the mod's own: one row per province, expandable
	# into its locations, naming the method and the raw materials the bonus is
	# made of. The button reopens it; pressing "run" opens it anyway.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = show
		tab_id = plan
		group_id = what
	}

	bag_wtp_init_counters = yes
	bag_wtp_sync = yes
	bag_wtp_rebuild_zone = yes
	bag_wtp_rebuild_regions = yes
	bag_wtp_refresh_goods = yes
	bag_wtp_drop_browse = yes
	bag_wtp_clear_rows = yes
}

# Copy the tick CMM keeps in its own variable map out to a plain global variable,
# which is what a location-scoped trigger can read.
# Scope: country
bag_wtp_sync = {
	cmm_sync_bool_alias = { setting = bag_wtp__buildable_only alias = bag_wtp_only_buildable }
	cmm_sync_bool_alias = { setting = bag_wtp__any_method alias = bag_wtp_any_method }
}
