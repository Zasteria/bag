# Settings, registered on CMF's mod registration hook.
#
# Everything the player touches is here; everything that depends on the game's
# own data -- which goods exist, which buildings make them, what each method is
# worth -- is generated beside it, so a patch that adds any of those is picked up
# by a rebuild rather than by an edit.
#
# Scope: country
bag_wtp_register = {
	cmf_register_mod = { mod_id = bag_wtp }

	# What you want made, in the two piles goods come in. Lists rather than
	# dropdowns: CMF handles a dropdown option click through
	# `CMM_MarkDropdownSelection_<index>` and defines exactly twenty of them, so
	# the twenty-first option onwards silently keeps the old selection.
	bag_wtp_register_good_raw_list = yes
	bag_wtp_register_good_made_list = yes

	# And the third way of naming what you want made: an urban right, which is a
	# bundle of two or three goods rather than one. It answers a different
	# question and it answers it in a window of its own -- a right's own
	# percentage is the same everywhere and re-ranks nothing, so only the bundle
	# is worth a pass. `docs/investigations/town_rights.md` has the whole of it.
	bag_wtp_register_right_common_list = yes
	bag_wtp_register_right_unique_list = yes

	# Where to look. Continents rather than the world's seventy-seven regions:
	# the fine choosing happens on the map afterwards, and a region list six
	# groups deep was a worse way to say "Europe".
	bag_wtp_register_zone_list = yes
	bag_wtp_register_region_lists = yes

	# Off by default, and deliberately. The player plans for land he does not own
	# yet, and `can_build_building` answers for the location as it stands today --
	# an unowned location fails it. Leaving this off means the table ranks the
	# ground rather than the current border.
	cmm_register_global_bool_setting = {
		mod_id = bag_wtp
		setting_id = buildable_only
		tab_id = plan
		group_id = what
		default_value = 0
	}

	# Two buttons, and they are the whole of the mod's two questions. «Считать
	# сейчас» ranks by what you could build today -- the fast answer. «Считать на
	# конец игры» ranks by what the ground gives once every advance is in, and
	# breaks the ties that leaves (0.00% everywhere, when a ladder ends early) by
	# what it gives on the way there.
	#
	# Both fill the same three columns; only the order differs. Sorting used to
	# be three buttons in the results window's header, and the twenty-first run's
	# verdict was that this is the place for it and two is the number.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = run
		tab_id = plan
		group_id = what
	}

	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = run_end
		tab_id = plan
		group_id = what
	}

	# The answer in a window of the mod's own: one row per province, expandable
	# into its locations, naming the method and the raw materials the bonus is
	# made of. The button reopens it; pressing "run" opens it anyway.
	cmm_register_button_setting = {
		mod_id = bag_wtp
		setting_id = show
		tab_id = plan
		group_id = what
	}

	# Every picker folded shut, once, the first time this player sees the page.
	# Seven region lists and two goods lists all open is a page to scroll rather
	# than a page to read; after this the player's own folding is his.
	bag_wtp_fold_pickers = yes

	bag_wtp_init_counters = yes
	bag_wtp_sync = yes
	bag_wtp_rebuild_zone = yes
	bag_wtp_rebuild_regions = yes
	bag_wtp_refresh_goods = yes
	bag_wtp_refresh_rights = yes

	# **Nothing here throws the answer away.** This runs on CMF's registration
	# hook, and the fifteenth run showed that hook firing again every time the
	# mod page is opened: it used to end with `bag_wtp_drop_browse` and
	# `bag_wtp_clear_rows`, so opening the menu to press «Открыть» wiped the
	# result that button exists to reopen, zeroed the counters beside it, and
	# took the rank off every row. On a new game there is nothing to clear --
	# `bag_wtp_init_counters` makes the numbers exist and the lists start empty.
}

# Copy the tick CMM keeps in its own variable map out to a plain global variable,
# which is what a location-scoped trigger can read.
# Scope: country
bag_wtp_sync = {
	cmm_sync_bool_alias = { setting = bag_wtp__buildable_only alias = bag_wtp_only_buildable }
	# `bag_wtp_rank_by_end` is not synced from anything: the two «Считать»
	# buttons set and clear it themselves, so no CMM setting has to agree with it.
	# It was a tick on this page for one load and three buttons in the results
	# window's header for another; a button that already says "calculate" is the
	# honest place for "calculate for what".
}
