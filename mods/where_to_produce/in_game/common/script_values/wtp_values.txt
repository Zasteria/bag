# How many buildings the answer keeps. `max` on an ordered iterator and the row
# cap both take a script value, so the Mod Menu setting reaches them through here.
# Scope: any
wtp_row_limit = {
	value = 8
	if = {
		limit = { has_global_variable = wtp_rows }
		value = global_var:wtp_rows
	}
	min = 1
	max = 12
}

# What a candidate is ranked on. One name so the selection pass has one thing to
# compare, and so the two measures cannot drift apart.
# Scope: country, with scope:wtp_cand naming the candidate
wtp_candidate_rank = {
	value = "variable_map(wtp_bonus_of|scope:wtp_cand)"
	if = {
		limit = { wtp_ranking_by_volume = yes }
		value = "variable_map(wtp_volume_of|scope:wtp_cand)"
	}
}
