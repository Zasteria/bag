# The selection window's bridges into script.
#
# The window itself is a scripted widget, and its `visible` reads a variable on
# the player rather than `GetVariableSystem`: a Mod Menu button is a script
# effect, and script cannot touch the interface's own variable system. So the
# button sets `bag_wtp_open` and the window watches it.
#
# The selection is kept twice, on purpose, and `bag_wtp_pick` / `bag_wtp_drop`
# are the only two effects allowed to write it:
#
#   - `bag_wtp_picked`, a variable on the location. This is what the interface
#     asks per row and what the map mode paints -- a map mode's `map_color` is a
#     location-scoped trigger block, so a variable on the location is a colour on
#     the map with nothing in between.
#   - `bag_wtp_picked_locations`, a global list. This is what the ranking pass
#     walks. There is no cheap "every location with variable X" iterator, and
#     walking the world to find them would cost more than keeping the list.
#
# Two records of one fact drift the moment anything else writes either half.

bag_wtp_open_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp_tree_refresh = yes
		set_variable = { name = bag_wtp_open value = 1 }
	}
}

bag_wtp_close_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		remove_variable = bag_wtp_open
		# Emptying the datamodel is what actually frees the rows. The window is
		# a scripted widget and never comes down; only the list it repeats over
		# decides how much of it is alive.
		bag_wtp_drop_browse = yes
		bag_wtp_tree_drop = yes
	}
}

# One location in or out.  `saved_scopes` is how a scripted GUI receives the
# `AddScope` its button passed in.
bag_wtp_toggle_location = {
	scope = country
	saved_scopes = { wtp_location }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp_location = {
			bag_wtp_toggle_one = yes
		}
		bag_wtp_rebuild_browse = yes
	}
}

# Everything currently browsable, in one press.
bag_wtp_pick_all = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp_tree_pick_all = yes
	}
}

# Clears the selection everywhere, not only inside what is being browsed --
# otherwise a selection made under one set of ticked regions would survive
# invisibly under another and quietly widen the next ranking.
bag_wtp_clear_all = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		every_in_global_list = {
			variable = bag_wtp_picked_locations
			remove_variable = bag_wtp_picked
		}
		clear_global_variable_list = bag_wtp_picked_locations
		set_global_variable = { name = bag_wtp_picked_count value = 0 }
		bag_wtp_drop_browse = yes
	}
}

# The results window. Opening copies the ranked answer into the list the window
# repeats over and closing empties it again: a scripted widget never comes down,
# so the datamodel is the only thing that decides how many rows are alive.
bag_wtp_open_result_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp_show_results = yes
		set_variable = { name = bag_wtp_result_open value = 1 }
	}
}

bag_wtp_close_result_window = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		remove_variable = bag_wtp_result_open
		bag_wtp_hide_results = yes
	}
}

# Expansion is kept on the province's capital, as in the selection window, but
# under a name of its own so opening a row in one window does not open it in the
# other.
bag_wtp_result_expand = {
	scope = country
	saved_scopes = { wtp_location }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp_location = {
			set_variable = { name = bag_wtp_res_expanded value = 1 }
		}
	}
}

bag_wtp_result_collapse = {
	scope = country
	saved_scopes = { wtp_location }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp_location = {
			remove_variable = bag_wtp_res_expanded
		}
	}
}

# The selection tree's bridges. Each opens one level or takes one whole
# geography; `saved_scopes` is how the row hands its own scope in.
bag_wtp_tree_open_region = {
	scope = country
	saved_scopes = { wtp_region }

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp_tree_close_area = yes
		bag_wtp_tree_fill_areas = yes
	}
}

bag_wtp_tree_open_area = {
	scope = country
	saved_scopes = { wtp_area }

	is_shown = {
		always = yes
	}

	effect = {
		bag_wtp_tree_fill_provinces = yes
		# The locations column belongs to a province of the area just closed.
		bag_wtp_tree_close_province = yes
	}
}
