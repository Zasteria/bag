# The selection window's bridges into script.
#
# The window itself is a scripted widget, and its `visible` reads a variable on
# the player rather than `GetVariableSystem`: a Mod Menu button is a script
# effect, and script cannot touch the interface's own variable system. So the
# button sets `bag_wtp_open` and the window watches it.
#
# The selection is kept twice, on purpose, and `bag_wtp_pick` / `bag_wtp_drop`
# are the only two effects allowed to write it:
#
#   - `bag_wtp_picked`, a variable on the location. This is what the interface
#     asks per row and what the map mode paints -- a map mode's `map_color` is a
#     location-scoped trigger block, so a variable on the location is a colour on
#     the map with nothing in between.
#   - `bag_wtp_picked_locations`, a global list. This is what the ranking pass
#     walks. There is no cheap "every location with variable X" iterator, and
#     walking the world to find them would cost more than keeping the list.
#
# Two records of one fact drift the moment anything else writes either half.

# One location in or out.  `saved_scopes` is how a scripted GUI receives the
# `AddScope` its button passed in.
bag_wtp_toggle_location = {
	scope = country
	saved_scopes = { wtp_location }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp_location = {
			bag_wtp_toggle_one = yes
		}
		bag_wtp_rebuild_browse = yes
		bag_wtp_recompute_live = yes
	}
}

# Clears the selection everywhere, not only inside what is being browsed --
# otherwise a selection made under one set of ticked regions would survive
# invisibly under another and quietly widen the next ranking.
bag_wtp_clear_all = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		every_in_global_list = {
			variable = bag_wtp_picked_locations
			remove_variable = bag_wtp_picked
		}
		clear_global_variable_list = bag_wtp_picked_locations
		set_global_variable = { name = bag_wtp_picked_count value = 0 }
		bag_wtp_drop_browse = yes
		bag_wtp_recompute_live = yes
	}
}

# «Count this one as a town», and its opposite, from a button on a plan row.
#
# **It is the player's answer and no plan run touches it.** The game's rank is
# only what is true today: a rural settlement he means to raise is still a rural
# settlement to `location_rank`, and the mod cannot guess which of them he will
# raise. `bag_wtp_plan_is_town` reads these two first and the rank last.
#
# Nothing is re-planned here. A rank change moves a location between two lists
# that are decided per province, so half a plan would disagree with the other
# half; the next press of «План» is what makes it true.
bag_wtp_plan_mark_town = {
	scope = country
	saved_scopes = { wtp_location }

	is_shown = {
		always = yes
	}

	effect = {
		scope:wtp_location = {
			if = {
				limit = { has_variable = bag_wtp_force_town }
				# Already forced to a town: the next click says «village» instead, and
				# the one after that gives the location back to the game's own rank.
				remove_variable = bag_wtp_force_town
				set_variable = { name = bag_wtp_force_rural value = 1 }
			}
			else_if = {
				limit = { has_variable = bag_wtp_force_rural }
				remove_variable = bag_wtp_force_rural
			}
			else = {
				set_variable = { name = bag_wtp_force_town value = 1 }
			}
		}
	}
}

# «Пересчитать», from the plan window itself.
#
# **Nothing else re-plans.** A pick on the map does not, and neither does the
# town/village button: the pass reads every recipe in the game on every location

# flip a hundred of them before he wants an answer. So the moment is his.

	is_shown = {
		always = yes
	}

# ------------------------------------------------- the settings, in the windows
#
# **Everything below was a setting on the mod's page until 2026-09-06.** His
# words: «мод внутри стал довольно громоздким и неудобным», and the rule --
# у функции на странице мода одна кнопка «открыть», всё остальное живёт внутри
# её окна. What CMM gave for free and has to be written here instead is small:
# a bool is a global to flip, a numeric is a pair of buttons with a clamp, and
# every one of them writes the same global the pass already read.

# «Считать на конец игры», in the two ranking windows.
#
# **It re-ranks on the spot.** It was two buttons -- «Считать» and «Считать на
# конец» -- and pressing one ran the pass; a switch that only changed what the
# *next* press would answer would be a switch that looks broken. The window is
# open, or this control is not on screen, so `bag_wtp_recompute_live` always
# has something to redraw.
bag_wtp_toggle_by_end = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		if = {
			limit = { has_global_variable = bag_wtp_rank_by_end }
			remove_global_variable = bag_wtp_rank_by_end
		}
		else = {
			set_global_variable = { name = bag_wtp_rank_by_end value = 1 }
		}
		bag_wtp_recompute_live = yes
	}
}

# «Только там, где здание вообще может стоять».

# The two caps, a press a step.
#
# **Three and four are the owner's own figures and cannot be derived.** The game
# has no building-slot count anywhere; what a cap is worth is only visible in
# what it buys. The bounds are CMM's old `min_value`/`max_value` written out:
# nothing else clamps them now, and a cap of zero is a plan of nothing.
bag_wtp_cap_urban_up = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		if = {
			limit = { global_var:bag_wtp_plan_cap_urban < 8 }
			change_global_variable = { name = bag_wtp_plan_cap_urban add = 1 }
		}
	}
}
