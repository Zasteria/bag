# CMF extension points. Both merge with whatever else declares them, so nothing
# of another mod's wiring is displaced.

# New game, save load and country transfer, once per human country.
# Scope: country
cmf_on_mod_registration = {
	on_actions = {
		bag_wtp_on_registration
	}
}

# Scope: country
bag_wtp_on_registration = {
	effect = {
		bag_wtp_register = yes
	}
}

# A CMM setting changed, or a button was clicked. The dropdown and the tick both
# auto-apply and arrive here; the region lists do not, because a list reaches
# nothing but its own `_on_changed` scripted GUI.
# Scope: country
cmf_on_callback = {
	on_actions = {
		bag_wtp_on_callback
	}
}

# Scope: country
bag_wtp_on_callback = {
	effect = {
		bag_wtp_on_button = yes
	}
}
