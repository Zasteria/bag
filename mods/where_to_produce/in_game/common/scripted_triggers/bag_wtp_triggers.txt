# Which locations the mod is willing to consider at all.
#
# `bag_wtp_in_zone` and `bag_wtp_can_build_something` are generated beside this,
# from the continent list and from which buildings make which good.
#
# Scope: location
bag_wtp_is_candidate = {
	# Sea zones have no province to walk, and every method's script value walks
	# the province. This is the guard, not a tidiness filter -- and `is_ownable`
	# is the whole of it: "not sea, lake or an impassable", which is also exactly
	# the ground a building can stand on.
	is_ownable = yes

	bag_wtp_in_zone = yes

	OR = {
		# The default. The player is planning for ground he does not hold yet, and
		# `can_build_building` answers for the location as it stands -- an unowned
		# one fails it, which would empty the table for exactly the case the mod
		# exists to serve.
		NOT = { has_global_variable = bag_wtp_only_buildable }

		# With the tick on, something producing the chosen good has to be startable
		# here today.
		bag_wtp_can_build_something = yes
	}
}

# Whether a province has earned a row at all.
#
# The twelfth run's table ended in a tail of provinces at 0.00% and «0/2» --
# every one of them the same building, the same method and the same output as
# the rows above, and none of them supplying a single one of the method's raw
# materials. Ranked correctly and worth nothing: they are the ground where this
# good has no reason to be made, and the thirteenth run's verdict on offering to
# show them anyway was that the offer is the litter, not the rows.
#
# There is one case where they appear regardless: a method that wants no raw
# material can never earn a bonus anywhere, so filtering on the bonus would
# empty the table rather than trim it. `_goods_all` is how many inputs the
# winning method has, and zero on both sides means there was nothing to find
# here rather than nothing found.
# Scope: location
bag_wtp_row_is_worth_it = {
	OR = {
		var:bag_wtp_bonus > 0
		var:bag_wtp_bonus_rural > 0
		# The endgame column counts too: a province that gives nothing to the
		# guild and 10% to the mill is exactly the row somebody planning for the
		# whole game came here for. `_end_bonus` is zero where that column has
		# nothing to add -- the same method won both -- and then the two above
		# have already answered.
		var:bag_wtp_end_bonus > 0
		var:bag_wtp_end_bonus_rural > 0
		# And the middle column, which is the only one with anything to say where
		# a ladder ends early: wool fine cloth is ten percent until the fifth age
		# and nothing after it.
		var:bag_wtp_mid_bonus > 0
		var:bag_wtp_mid_bonus_rural > 0
		AND = {
			var:bag_wtp_goods_all = 0
			var:bag_wtp_goods_all_rural = 0
			var:bag_wtp_end_goods_all = 0
			var:bag_wtp_end_goods_all_rural = 0
			var:bag_wtp_mid_bonus = 0
			var:bag_wtp_mid_bonus_rural = 0
		}
	}
}

# The same question for an urban right's row, asked of the whole bundle.
#
# The sixteenth run's table ended in a province at 0.00% on all three goods and
# «0/2» three times over: the bundle can be made there and none of it is made
# any better there than anywhere else. Ranked correctly by «Ценность», since
# every method still produces, and worth nothing to somebody choosing where a
# right should go.
#
# A slot whose method wants no raw material can never earn a bonus, so a row
# where every slot is like that is kept -- otherwise the filter would empty the
# table rather than trim it.
# Scope: location
bag_wtp_right_row_is_worth_it = {
	OR = {
		var:bag_wtp_r_bonus_1 > 0
		var:bag_wtp_r_bonus_2 > 0
		var:bag_wtp_r_bonus_3 > 0
		# Ranked for the last age, every bonus in the table can be zero -- no
		# recipe that survives is fed anywhere -- and then what keeps a row is
		# what the ground gives on the way there. Without this the whole table
		# empties itself exactly where the plan needs it most.
		AND = {
			has_global_variable = bag_wtp_rank_by_end
			var:bag_wtp_r_mid_total > 0
		}
		AND = {
			var:bag_wtp_r_goods_all_1 = 0
			var:bag_wtp_r_goods_all_2 = 0
			var:bag_wtp_r_goods_all_3 = 0
		}
	}
}
