# Which locations the mod is willing to consider at all.
#
# `bag_wtp_in_zone` and `bag_wtp_can_build_something` are generated beside this,
# from the continent list and from which buildings make which good.
#
# Scope: location
bag_wtp_is_candidate = {
	# Sea zones have no province to walk, and every method's script value walks
	# the province. This is the guard, not a tidiness filter -- and `is_ownable`
	# is the whole of it: "not sea, lake or an impassable", which is also exactly
	# the ground a building can stand on.
	is_ownable = yes

	bag_wtp_in_zone = yes

	OR = {
		# The default. The player is planning for ground he does not hold yet, and
		# `can_build_building` answers for the location as it stands -- an unowned
		# one fails it, which would empty the table for exactly the case the mod
		# exists to serve.
		NOT = { has_global_variable = bag_wtp_only_buildable }

		# With the tick on, something producing the chosen good has to be startable
		# here today.
		bag_wtp_can_build_something = yes
	}
}

# Whether a province has earned a row at all.
#
# The twelfth run's table ended in a tail of provinces at 0.00% and «0/2» --
# every one of them the same building, the same method and the same output as
# the rows above, and none of them supplying a single one of the method's raw
# materials. Ranked correctly and worth nothing: they are the ground where this
# good has no reason to be made.
#
# The tick brings them back, and there is one case where they come back anyway:
# a method that wants no raw material at all can never earn a bonus, so
# filtering on the bonus would empty the table rather than trim it. `_goods_all`
# is how many inputs the winning method has, and zero on both sides means there
# was nothing to find here rather than nothing found.
# Scope: location
bag_wtp_row_is_worth_it = {
	OR = {
		has_global_variable = bag_wtp_show_flat
		var:bag_wtp_bonus > 0
		var:bag_wtp_bonus_rural > 0
		AND = {
			var:bag_wtp_goods_all = 0
			var:bag_wtp_goods_all_rural = 0
		}
	}
}
