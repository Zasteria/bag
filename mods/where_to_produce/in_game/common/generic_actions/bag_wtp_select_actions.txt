# Choosing the ground on the map, using the game's own target picker.
#
# `select_trigger` inside a generic action is what draws the panel with the
# search box, the sortable list and the map highlight -- and clicking the map
# picks the same thing the list row does. Advanced Auto Build reaches it the same
# way; the three buttons in this mod's window are `action_button`s naming these.
#
# **The ownership restriction is not the engine's.** Advanced Auto Build narrows
# its picker with `interaction_source_list = { scope:actor = { every_owned_location
# ... } }`, so it can only ever offer land already held. Vanilla's own actions
# mostly give no source at all and filter with `visible`, which runs in the
# candidate's own scope -- and that is what these do. Nothing here asks who owns
# anything.
#
# Every one of them **toggles**. Picking is not a one-shot commitment that needs
# saving afterwards: the selection lives in script and each click adds or removes,
# so closing the panel is the whole of "done". That also makes it not matter
# whether the panel stays open after a pick, which is the one thing about this
# system the files in `reference/` cannot answer.

# One location.
bag_wtp_select_location = {
	type = owncountry
	show_in_gui_list = no
	show_message = no
	ai_tick = never
	automation_tick = never
	potential = { always = yes }
	allow = { always = yes }

	select_trigger = {
		looking_for_a = location
		target_flag = bag_wtp_target
		name = bag_wtp_choose_location
		map_mode = bag_wtp_selection
		bottom_widget = bag_wtp_select_footer
		column = { data = name }
		column = { data = population }
		visible = {
			is_ownable = yes
			bag_wtp_in_zone = yes
		}
		enabled = { always = yes }
		tooltip_msg_key = bag_wtp_choose_location_tt
	}

	effect = {
		scope:bag_wtp_target ?= {
			bag_wtp_toggle_one = yes
		}
		# `scope:actor` and not the scope this block opens in. A generic action's
		# effect does not run in the country's scope -- vanilla wraps every one of
		# its own in `scope:actor`, and Advanced Auto Build never touches anything
		# but `scope:target_location`. Without it `bag_wtp_recompute_live` asks
		# `has_variable = bag_wtp_result_open` of something that is not the country,
		# gets no, and returns having done nothing at all, which is the one kind of
		# failure that logs nothing.
		scope:actor = {
			bag_wtp_rebuild_browse = yes
			# The plan changed, so the answer has: re-rank while the window is open.
			bag_wtp_recompute_live = yes
		}
	}

	ai_will_do = { value = 0 }
}

# One market, all of its land locations at once.
#
# **The one picker that has to name its own candidates.** An area, a province and
# a location are ground and the panel finds them itself; a market is not drawn on
# the map, and vanilla's only `looking_for_a = market` -- the burgher capital
# mission -- fills `interaction_source_list` from
# `every_market_present_in_country`. This does the same, so the list is the
# markets the player's own land sits in, which is exactly the case it was asked
# for: one country, several markets, and a production balance wanted per market
# rather than per realm.
bag_wtp_select_market = {
	type = owncountry
	show_in_gui_list = no
	show_message = no
	ai_tick = never
	automation_tick = never
	potential = { always = yes }
	allow = { always = yes }

	select_trigger = {
		looking_for_a = market
		target_flag = bag_wtp_target
		interaction_source_list = {
			scope:actor = {
				every_market_present_in_country = {
					add_to_list = source
				}
			}
		}
		name = bag_wtp_choose_market
		none_available_msg_key = bag_wtp_no_markets
		map_mode = bag_wtp_selection
		bottom_widget = bag_wtp_select_footer
		column = { data = name }
		enabled = { always = yes }
		tooltip_msg_key = bag_wtp_choose_market_tt
	}

	effect = {
		scope:bag_wtp_target ?= {
			bag_wtp_toggle_market_scope = yes
		}
		# `scope:actor` and not the scope this block opens in -- the same reason as
		# the three above, and the same silent failure if it is left out.
		scope:actor = {
			bag_wtp_rebuild_browse = yes
			bag_wtp_recompute_live = yes
		}
	}

	ai_will_do = { value = 0 }
}
