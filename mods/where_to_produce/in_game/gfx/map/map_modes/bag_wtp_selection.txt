# The selection, painted on the map, in the base game's Geography category.
#
# `map_color` is a location-scoped trigger block, which is what makes this cheap:
# the selection is already a variable on the location, so painting it is one
# `has_variable` and nothing else.
#
# Advanced Auto Build's `eu5ab_template_coverage` is the working example this is
# built from, and the difference is the point. Its every branch opens with
# `has_owner = yes owner ?= { is_human = yes }`, so it can only ever colour land
# you already hold. That restriction is the mod's, not the engine's, and there is
# none here: the whole use is planning for ground before it is yours.
bag_wtp_selection = {
	map_color = {
		if = {
			limit = { has_variable = bag_wtp_picked }
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			# Inside the ticked regions but not chosen — the ground you are
			# choosing from, so that the selection reads against something.
			limit = { bag_wtp_in_zone = yes }
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else = { value = define:NMapColors|DEFAULT_COLOR }
	}

	tooltip_key = {
		if = {
			limit = { has_variable = bag_wtp_picked }
			value = bag_wtp_map_picked_tt
		}
		else_if = {
			limit = { bag_wtp_in_zone = yes }
			value = bag_wtp_map_browsable_tt
		}
		else = { value = bag_wtp_map_outside_tt }
	}

	legend_key = {
		desc = bag_wtp_map_legend_picked
		color = define:NMapColors|MAP_COLOR_HIGH
	}
	legend_key = {
		desc = bag_wtp_map_legend_browsable
		color = define:NMapColors|MAP_COLOR_LOW
	}

	small_map_names = location
	medium_map_names = province
	large_map_names = area
	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location
	category = geography
	# Copied from Advanced Auto Build, whose generator notes that EU5 1.3
	# enumerates the geography groups 0-3 explicitly in the map-mode selector.
	# Two mods claiming the same index would collide, and that one is the only
	# other mod known to add a geography mode.
	index = 3
	allow_allocate_hotkey = yes
	flatmap_behaviour = Always
	fill_in_impassable = no
	use_fow = no
	enable_rivers = yes
	map_markers = { all = no }

	gradient_parameters = {
		zoom_step = 2
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	flatmap_gradient_parameters = {
		zoom_step = 12
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	color_refresh_counters = { Day }
	color_and_names_refresh_counters = { LocationOwnerChanged }
}
