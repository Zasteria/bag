# The whole-map plan, painted: whether the plan filled each location or fell
# short of the cap it was allowed.
#
# `map_color` is a location-scoped trigger block, so this costs nothing beyond
# what the pass has already written -- `bag_wtp_load` is a variable on the
# location and this is two comparisons per location per repaint.
#
# **By completeness and not by good.** Forty-seven goods is forty-seven colours
# nobody can read off a map. And under the province model a full location is no
# longer a «магнит» to be worried about -- every province fills to its capacity,
# so what is worth seeing is the opposite: the ground the plan could not fill,
# because it could not feed enough different goods to make up a list. Which
# province the plan crowded most is the top of the window's table instead.
#
# `category = economy` because the geography category is full: EU5 1.3 enumerates
# its groups 0-3 and `bag_wtp_selection` already holds one of them. Glorp UI and
# Construction Manager both put a mode in economy, so an index here is a claim
# rather than an allocation -- see `docs/research/interface.md`.
bag_wtp_plan = {
	map_color = {
		if = {
			limit = { is_land = no }
			value = define:NMapColors|WATER_MAP_COLOR
		}
		else_if = {
			# Filled: the location builds its side's whole list. The two caps are
			# read the same way `bag_wtp_plan_has_room` reads them, because the
			# same two numbers decided how long the list could be.
			limit = {
				has_variable = bag_wtp_load
				OR = {
					AND = {
						location_rank = location_rank:rural_settlement
						var:bag_wtp_load >= global_var:bag_wtp_plan_cap_rural
					}
					AND = {
						NOT = { location_rank = location_rank:rural_settlement }
						var:bag_wtp_load >= global_var:bag_wtp_plan_cap_urban
					}
				}
			}
			value = rgb { 0 200 0 }
		}
		else_if = {
			# Something, but short of the cap: the ground could not feed enough
			# different goods to make up a list.
			limit = { has_variable = bag_wtp_load var:bag_wtp_load >= 2 }
			value = rgb { 255 255 0 }
		}
		else_if = {
			limit = { has_variable = bag_wtp_load var:bag_wtp_load = 1 }
			value = rgb { 255 128 0 }
		}
		else_if = {
			# In the plan's ground and given nothing at all.
			limit = { bag_wtp_in_zone = yes }
			value = rgb { 200 40 40 }
		}
		else = { value = define:NMapColors|DEFAULT_COLOR }
	}

	tooltip_key = {
		if = {
			limit = { is_land = no }
			value = bag_wtp_plan_map_water_tt
		}
		else_if = {
			limit = { has_variable = bag_wtp_load var:bag_wtp_load > 0 }
			value = bag_wtp_plan_map_built_tt
		}
		else_if = {
			limit = { bag_wtp_in_zone = yes }
			value = bag_wtp_plan_map_empty_tt
		}
		else = { value = bag_wtp_plan_map_outside_tt }
	}

	legend_key = {
		desc = bag_wtp_plan_legend_full
		color = rgb { 0 200 0 }
	}
	legend_key = {
		desc = bag_wtp_plan_legend_part
		color = rgb { 255 255 0 }
	}
	legend_key = {
		desc = bag_wtp_plan_legend_one
		color = rgb { 255 128 0 }
	}
	legend_key = {
		desc = bag_wtp_plan_legend_none
		color = rgb { 200 40 40 }
	}

	small_map_names = location
	medium_map_names = province
	large_map_names = area
	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location
	category = economy
	index = 2
	allow_allocate_hotkey = yes
	flatmap_behaviour = Always
	fill_in_impassable = no
	use_fow = no
	enable_rivers = yes
	# The game's own RGO icons stay on, and they are not decoration here: what a
	# location already digs up is half of what the plan was deciding against.
	map_markers = {
		all = no
		raw_goods_marker = yes
	}

	gradient_parameters = {
		zoom_step = 2
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	flatmap_gradient_parameters = {
		zoom_step = 12
		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}
	# A map mode repaints on the counters it names and a variable written by
	# script is not one of them, so the plan would sit on the map as it was until
	# something else moved. `Day` is what `bag_wtp_selection` uses and what makes
	# a plan appear without leaving the map mode and coming back.
	color_refresh_counters = { Day }
	color_and_names_refresh_counters = { LocationOwnerChanged }
}
